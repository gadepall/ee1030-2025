	\begin{frame}{Question}
			For what value of k do the following system of equations possess a non trivial (i.e.,
		not all zero) solution over the set of rationals Q?
		\begin{align}
			x + ky + 3z = 0
		\end{align}
		\begin{align}
			3x+ky-2z=0%3x + ky − 2z = 0
		\end{align}
		\begin{align}
			2x+3y-4z=0%2x + 3y − 4z = 0
		\end{align}
		
	\end{frame}
	
	
	\begin{frame}{Theoretical Solution}
	Given:
	\begin{align}
		\begin{myvec}{1\\k\\3}\end{myvec}^\top\vec{x} = 0 \label{1}
	\end{align}
	\begin{align}
		\begin{myvec}{3\\k\\-2}\end{myvec}^\top\vec{x} = 0\label{2}
	\end{align}
	\begin{align}
		\begin{myvec}{2\\3\\-4}\end{myvec}^\top\vec{x} = 0\label{3}
	\end{align}
	Combining \eqref{1},\eqref{2},\eqref{3}:
	\begin{align}
		A\vec{x}=0
	\end{align}
\end{frame}
	\begin{frame}{Theoretical Solution}
	where A is :
	\begin{align}
		\begin{myvec}{1&k&3\\3&k&-2\\2&3&-4}\end{myvec}
	\end{align}
	\end{frame}
	\begin{frame}{Theoretical Solution}
	For this system to have non-trivial solution these planes must intersect on a line, i.e, any two of the rows must be linearly dependent in order to have infinitely many non-trivial solutions. We simplify the matrix by using gaussian elimination:
	\begin{align}
		\begin{myvec}{1&k&3\\3&k&-2\\2&3&-4}\end{myvec}
		&\xleftrightarrow{\,R_2 \gets R_2-3R_1}
		\begin{myvec}{1&k&3\\0&-2k&-11\\2&3&-4}\end{myvec}\\
		\begin{myvec}{1&k&3\\0&-2k&-11\\2&3&-4}\end{myvec}
		&\xleftrightarrow{\,R_2 \gets R_3-2R_1}
		\begin{myvec}{1&k&3\\0&-2k&-11\\0&3-2k&-10}\end{myvec}
	\end{align}
	Now, the second and third row must be linearly dependent:
	\begin{align}
		\implies \frac{-2k}{-11}=\frac{3-2k}{-10}
	\end{align}
\end{frame}
	\begin{frame}{Theoretical Solution}
	On cross-multiplying
	\begin{align}
		20k = 22k-33
	\end{align}
	\begin{align}
		\implies k = \frac{33}{2}
	\end{align}
	\end{frame}
	
	
	\begin{frame}[fragile]
		\frametitle{C Code}
		
		\begin{lstlisting}
			double find_k() {
				// A non-trivial solution exists if the determinant of the coefficient
				// matrix is zero. The simplified determinant equation is: -2k + 33 = 0.
				double k_coefficient = -2.0;
				double constant_term = 33.0;
				// We solve for k: k = -constant_term / k_coefficient
				double k = -constant_term / k_coefficient;
				return k;
			}
		\end{lstlisting}
	\end{frame}
