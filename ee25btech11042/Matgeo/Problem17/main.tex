	\textbf{Question}:\\
	Find the points of intersection of the following parabola and circle
	\begin{align}
		y^2 &= 6ax \label{eq:parabola} \\
		x^2 + y^2 &= 16a^2 \label{eq:circle}
	\end{align}
	
	\solution
	
	We first express the equations for the parabola \eqref{eq:parabola} and the circle \eqref{eq:circle} in the standard quadratic form $g(x,y)=0$.
	\begin{align}
		y^2 - 6ax &= 0 \label{eq:P1} \\
		x^2 + y^2 - 16a^2 &= 0 \label{eq:C2}
	\end{align}
	
	In homogeneous coordinates, using the form $\vec{x}^\top C \vec{x} = 0$, where $\vec{x} = \begin{myvec}{x & y & 1}\end{myvec}^\top$, the matrices for the conics are:\\
	where,
	\begin{align}
		C = \begin{myvec}{\vec{V}& \vec{u}^\top\\\vec{u} & f}\end{myvec}
	\end{align}
	\begin{align}
		C_1 &= \begin{myvec}{0 & 0 & -3a \\ 0 & 1 & 0 \\ -3a & 0 & 0}\end{myvec} \label{eq:C1_matrix} \\
		C_2 &= \begin{myvec}{1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & -16a^2}\end{myvec} \label{eq:C2_matrix}
	\end{align}
	
	The intersection points lie on a degenerate conic formed from a linear combination $P_1 + \mu C_2 = 0$. We must find the value of $\mu$ that makes the determinant of the combined conic's matrix, $C(\mu) = C_1 + \mu C_2$, equal to zero.
	The general matrix for this combination is:
	\begin{align}
		C(\mu) = \begin{myvec}{\mu & 0 & -3a \\ 0 & 1+\mu & 0 \\ -3a & 0 & -16a^2\mu}\end{myvec}
	\end{align}
	We set the determinant to zero:
	\begin{align}
		\det(C(\mu)) &= \mu(1+\mu)(-16a^2\mu) - (-3a)(1+\mu)(-3a) \nonumber \\
		&= -16a^2\mu^2(1+\mu) - 9a^2(1+\mu) \nonumber \\
		&= -a^2(1+\mu)(16\mu^2 + 9) = 0
	\end{align}
	The only real root is obtained from the first factor, which gives $\mu = -1$.
	The equation for this degenerate conic is found by subtracting \eqref{eq:C2} from \eqref{eq:P1}:
	\begin{align}
		(y^2 - 6ax) - (x^2 + y^2 - 16a^2) &= 0 \nonumber \\
		-x^2 - 6ax + 16a^2 &= 0 \nonumber \\
		x^2 + 6ax - 16a^2 &= 0 \nonumber \\
		\implies (x+8a)(x-2a) &= 0 \label{eq:degenerate_conic}
	\end{align}
	This degenerate conic consists of two vertical lines, $x=2a$ and $x=-8a$. The line $x=-8a$ does not intersect the parabola $y^2=6ax$ in real points (assuming $a>0$). We thus consider the line $x=2a$.
	
	We find the intersection of the line $x-2a=0$ with the parabola $P_1$ using the formula $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$.
	The line and conic parameters from $P_1$ are:
	$\vec{h} = \begin{myvec}{2a \\ 0}\end{myvec}$, $\vec{m} = \begin{myvec}{0 \\ 1}\end{myvec}$, $\vec{V} = \begin{myvec}{0 & 0 \\ 0 & 1}\end{myvec}$, $\vec{u} = \begin{myvec}{-3a \\ 0}\end{myvec}$, and $f = 0$.
	
	Next, we compute the components for the parameter $\kappa_i$:
	\begin{align}
		\vec{m}^\top\vec{V}\vec{m} &= 1 \\
		g(\vec{h}) &= -12a^2 \\
		\vec{m}^\top(\vec{V}\vec{h} + \vec{u}) &= 0
	\end{align}
	
	We substitute these into the formula for $\kappa_i$:
	\begin{align}
		\kappa_i &= \frac{1}{\vec{m}^\top\vec{V}\vec{m}} \left( - \vec{m}^\top(\vec{V}\vec{h} + \vec{u}) \pm \sqrt{[\vec{m}^\top(\vec{V}\vec{h} + \vec{u})]^2 - g(\vec{h})(\vec{m}^\top\vec{V}\vec{m})} \right) \nonumber \\
		&= \frac{1}{1} \left( -(0) \pm \sqrt{(0)^2 - (-12a^2)(1)} \right) = \pm\sqrt{12a^2} = \pm 2\sqrt{3}a
	\end{align}
	This gives two values: $\kappa_1 = 2\sqrt{3}a$ and $\kappa_2 = -2\sqrt{3}a$.
	
	Finally, we find the intersection points using $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$:
	\begin{align}
		\vec{x}_1 &= \begin{myvec}{2a \\ 0}\end{myvec} + 2\sqrt{3}a \begin{myvec}{0 \\ 1}\end{myvec} = \begin{myvec}{2a \\ 2\sqrt{3}a}\end{myvec} \label{eq:sol1} \\
		\vec{x}_2 &= \begin{myvec}{2a \\ 0}\end{myvec} - 2\sqrt{3}a \begin{myvec}{0 \\ 1}\end{myvec} = \begin{myvec}{2a \\ -2\sqrt{3}a}\end{myvec} \label{eq:sol2}
	\end{align}
	
	Thus, the intersection points are $(2a, 2\sqrt{3}a)$ from \eqref{eq:sol1} and $(2a, -2\sqrt{3}a)$ from \eqref{eq:sol2}.
