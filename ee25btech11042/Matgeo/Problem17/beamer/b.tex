\begin{document}
	
	% --- Frame 1: Title Page ---
	\begin{frame}
		\titlepage
	\end{frame}
	
	% --- Frame 2: Problem Statement ---
	\begin{frame}
		\frametitle{Question}
		Find the points of intersection of the following parabola and circle and plot the region:
		\begin{align}
			y^2 \le 6ax \\
			x^2 + y^2 \le 16a^2
		\end{align}
		\begin{figure}
			\includegraphics[width=0.6\textwidth]{figs/Figure_2.png}
			\caption{Region bounded by the parabola and circle.}
		\end{figure}
	\end{frame}
	
	% --- Frame 3: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		We express the boundaries in the standard quadratic form $g(x,y)=0$:
		\begin{align}
			y^2 - 6ax &= 0 \label{eq:P1} \\
			x^2 + y^2 - 16a^2 &= 0 \label{eq:C2}
		\end{align}
		\pause
		In homogeneous coordinates, using the form $\vec{x}^\top C \vec{x} = 0$, where $\vec{x} = \begin{myvec}{x & y & 1}\end{myvec}^\top$, the conic matrices are:
		\begin{align}
			C_1 &= \begin{myvec}{0 & 0 & -3a \\ 0 & 1 & 0 \\ -3a & 0 & 0}\end{myvec} \\
			C_2 &= \begin{myvec}{1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & -16a^2}\end{myvec}
		\end{align}
	\end{frame}
	
	% --- Frame 4: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		The intersection points lie on a degenerate conic $C(\mu) = C_1 + \mu C_2$. We find $\mu$ by solving $\det(C(\mu)) = 0$.
		\begin{align}
			\det \begin{myvec}{\mu & 0 & -3a \\ 0 & 1+\mu & 0 \\ -3a & 0 & -16a^2\mu}\end{myvec} &= 0 \nonumber \\
			\implies (\mu)(1+\mu)(-16a^2\mu - \frac{9a^2}{\mu}) &= 0 \nonumber \\
			-a^2(1+\mu)(16\mu^2 + 9) &= 0
		\end{align}
		\pause
		The only real root is $\mu = -1$.
	\end{frame}
	
	% --- Frame 5: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		The equation is found by subtracting \eqref{eq:C2} from \eqref{eq:P1}:
		\begin{align}
			(y^{2}-6ax)-(x^{2}+y^{2}-16a^{2})=0 \nonumber \\
			\implies -x^{2} - 6ax + 16a^{2} = 0
		\end{align}
		\pause
		Using the general conic matrix representation $\begin{myvec}{x & y}\end{myvec} Q \begin{myvec}{x \\ y}\end{myvec} + L^{\top} \begin{myvec}{x \\ y}\end{myvec} + C = 0$, we have:
		\begin{align}
			Q = \begin{myvec}{ -1 & 0 \\ 0 & 0 }\end{myvec}, \quad
			L = \begin{myvec}{ -6a \\ 0 }\end{myvec}, \quad
			C = 16a^{2}.
		\end{align}
		\pause
		The full matrix form is:
		\begin{align}
			\boxed{
				\begin{myvec}{x & y}\end{myvec}
				\begin{myvec}{ -1 & 0 \\ 0 & 0 }\end{myvec}
				\begin{myvec}{x \\ y}\end{myvec}
				+
				\begin{myvec}{-6a & 0}\end{myvec}
				\begin{myvec}{x \\ y}\end{myvec}
				+ 16a^{2} = 0.
			}
		\end{align}
	\end{frame}
	
	% --- Frame 6: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		Multiplying by -1 gives $x^{2} + 6ax - 16a^{2} = 0$, which factors into two linear equations:
		\begin{align}
			(x + 8a)(x - 2a) = 0
		\end{align}
		This represents two vertical lines: $x + 8a = 0$ and $x - 2a = 0$.
		\pause
		In matrix form, the lines are:
		\begin{align}
			\boxed{
				\begin{myvec}{1 & 0}\end{myvec}
				\begin{myvec}{x \\ y}\end{myvec} + 8a = 0
				\quad \text{and} \quad
				\begin{myvec}{1 & 0}\end{myvec}
				\begin{myvec}{x \\ y}\end{myvec} - 2a = 0.
			}
		\end{align}
		\pause
		The line $x=-8a$ does not intersect the parabola $y^2=6ax$ in real points (for $a>0$). We only consider the line $\mathbf{x=2a}$.
	\end{frame}
	
	% --- Frame 7: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		We find the intersection of the line $x-2a=0$ with the parabola $P_1$ using the formula $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$.
		\pause
		
		The parameters for the line and conic are:
		\[ \vec{h} = \begin{myvec}{2a \\ 0}\end{myvec}, \vec{m} = \begin{myvec}{0 \\ 1}\end{myvec}, \vec{V} = \begin{myvec}{0 & 0 \\ 0 & 1}\end{myvec}, \vec{u} = \begin{myvec}{-3a \\ 0}\end{myvec}, f = 0. \]
		
		\pause
		The components for the parameter $\kappa_i$ are:
		\begin{align}
			\vec{m}^\top\vec{V}\vec{m} &= 1 \\
			g(\vec{h}) &= -12a^2 \\
			\vec{m}^\top(\vec{V}\vec{h} + \vec{u}) &= 0
		\end{align}
		\pause
		Substituting these values gives:
		\begin{align}
			\kappa_i &= \frac{1}{1} \left( -(0) \pm \sqrt{(0)^2 - (-12a^2)(1)} \right) = \pm\sqrt{12a^2} \nonumber \\
			&= \pm 2\sqrt{3}a
		\end{align}
	\end{frame}
	
	% --- Frame 8: Solution Step ---
	\begin{frame}
		\frametitle{Theoretical solution}
		Finally, we find the intersection points using $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$:
		\begin{align}
			\vec{x}_1 &= \begin{myvec}{2a \\ 0}\end{myvec} + 2\sqrt{3}a \begin{myvec}{0 \\ 1}\end{myvec} = \begin{myvec}{2a \\ 2\sqrt{3}a}\end{myvec} \\
			\vec{x}_2 &= \begin{myvec}{2a \\ 0}\end{myvec} - 2\sqrt{3}a \begin{myvec}{0 \\ 1}\end{myvec} = \begin{myvec}{2a \\ -2\sqrt{3}a}\end{myvec}
		\end{align}
		\pause
		
		The intersection points are:
		
		\textbf{$(2a, 2\sqrt{3}a)$ and $(2a, -2\sqrt{3}a)$}
		
		\begin{figure}
			\includegraphics[width=0.55\textwidth]{figs/Figure_1.png}
			\caption{Plot showing the intersection points.}
		\end{figure}
	\end{frame}
