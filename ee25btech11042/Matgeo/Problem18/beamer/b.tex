	\begin{frame}{Question}
		\large
		A circle touches the line $y=x$ at a point P such that $OP=4\sqrt{2}$, where O is the origin. The circle contains the point $(-10,2)$ in its interior and the length of its chord on the line $x+y=0$ is $6\sqrt{2}$.
		
		\
		
		Determine the equation of the circle.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		The tangent line is $y=x$, or $x-y=0$. A point $\vec{P}$ on this line is represented parametrically.
		\begin{align}
			\vec{P} = k\begin{myvec}{1 \\ 1}\end{myvec}
		\end{align}
		The distance from the origin is $\norm{\vec{P}} = 4\sqrt{2}$.
		\begin{align}
			|k|\norm{\begin{myvec}{1 \\ 1}\end{myvec}} = |k|\sqrt{2} &= 4\sqrt{2} \nonumber \\
			\implies |k| &= 4
		\end{align}
		The two possible points are $\vec{P}_1=4\begin{myvec}{1 \\ 1}\end{myvec}$ and $\vec{P}_2=-4\begin{myvec}{1 \\ 1}\end{myvec}$.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		Let the center be $\vec{C} = \begin{myvec}{h \\ k}\end{myvec}$. $\vec{P}$ is the foot of the perpendicular from $\vec{C}$ to the line $\vec{n}^\top\vec{x}=0$ where $\vec{n}=\begin{myvec}{1 \\ -1}\end{myvec}$.
		\begin{align}
			\vec{P} &= \vec{C} - \frac{\vec{n}^\top\vec{C}}{\norm{\vec{n}}^2}\vec{n} = \begin{myvec}{h \\ k}\end{myvec} - \frac{h-k}{2}\begin{myvec}{1 \\ -1}\end{myvec} = \frac{h+k}{2}\begin{myvec}{1 \\ 1}\end{myvec}
		\end{align}
		Comparing this with the two possibilities for $\vec{P}$:
		\begin{align}
			\text{Case 1: } \frac{h+k}{2} = 4 \implies h+k = 8 \\
			\text{Case 2: } \frac{h+k}{2} = -4 \implies h+k = -8
		\end{align}
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		The chord lies on $x+y=0$. The distance from $\vec{C}$ to this line is $d_{chord}$.
		\begin{align}
			d_{chord} = \frac{\norm{h+k}}{\sqrt{1^2+1^2}} = \frac{\norm{h+k}}{\sqrt{2}}
		\end{align}
		With half-chord length $L=3\sqrt{2}$, the radius $r$ is given by:
		\begin{align}
			r^2 = d_{chord}^2 + L^2 = \frac{(h+k)^2}{2} + 18
		\end{align}
		The radius is also $r = \norm{\vec{C}-\vec{P}}$.
		\begin{align}
			r^2 = \norm{\frac{1}{2}\begin{myvec}{h-k \\ k-h}\end{myvec}}^2 = \frac{(h-k)^2}{2}
		\end{align}
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		Equating the two expressions for $r^2$:
		\begin{align}
			\frac{(h-k)^2}{2} &= \frac{(h+k)^2}{2} + 18 \nonumber \\
			(h-k)^2 &= (h+k)^2 + 36 \implies -4hk = 36 \implies hk = -9
		\end{align}
		Now we solve for the center in each case.
		
		For Case 1, $h+k=8$ and $hk=-9$. 
		\begin{align}
			h(8-h)=-9 \implies h^2-8h-9=0 \implies (h-9)(h+1)=0
		\end{align}
		This gives centers $\vec{C}_A = \begin{myvec}{9 \\ -1}\end{myvec}$ and $\vec{C}_B = \begin{myvec}{-1 \\ 9}\end{myvec}$.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		For Case 2, $h+k=-8$ and $hk=-9$. 
		\begin{align}
			h(-8-h)=-9 \implies h^2+8h-9=0 \implies (h+9)(h-1)=0
		\end{align}
		This gives centers $\vec{C}_C = \begin{myvec}{1 \\ -9}\end{myvec}$ and $\vec{C}_D = \begin{myvec}{-9 \\ 1}\end{myvec}$.
		
		For any of these centers, the radius squared is $r^2 = \frac{(h-k)^2}{2} = \frac{(\pm 10)^2}{2} = 50$.
		\end{frame}
			
			\begin{frame}{Theoretical solution}
				The point $\vec{Q} = \begin{myvec}{-10 \\ 2}\end{myvec}$ is in the circle's interior, so we check $\norm{\vec{Q}-\vec{C}}^2 < 50$.
				\begin{align}
					\norm{\vec{Q}-\vec{C}_A}^2 &= \norm{\begin{myvec}{-19 \\ 3}\end{myvec}}^2 = 370 > 50 \\
					\norm{\vec{Q}-\vec{C}_B}^2 &= \norm{\begin{myvec}{-9 \\ -7}\end{myvec}}^2 = 130 > 50 \\
					\norm{\vec{Q}-\vec{C}_C}^2 &= \norm{\begin{myvec}{-11 \\ 11}\end{myvec}}^2 = 242 > 50 \\
					\norm{\vec{Q}-\vec{C}_D}^2 &= \norm{\begin{myvec}{-1 \\ 1}\end{myvec}}^2 = 2 < 50
				\end{align}
				The correct center is $\vec{C}_D = \begin{myvec}{-9 \\ 1}\end{myvec}$.
			\end{frame}
			
			\begin{frame}{Theoretical solution}
				The equation of the circle is:
				\begin{align}
					(x+9)^2 + (y-1)^2 = 50
				\end{align}
				Expanding this gives:
				\begin{align}
					x^2+y^2+18x-2y+32=0
				\end{align}
				In the matrix form $\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f = 0$:
				\begin{align}
					\begin{myvec}{x & y}\end{myvec} \begin{myvec}{1 & 0 \\ 0 & 1}\end{myvec} \begin{myvec}{x \\ y}\end{myvec} + 2\begin{myvec}{9 & -1}\end{myvec} \begin{myvec}{x \\ y}\end{myvec} + 32 = 0
				\end{align}
			\end{frame}
