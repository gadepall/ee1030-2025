	\begin{frame}[fragile]{Question}
		
		The solution to system of equations
		\begin{align}
			\begin{myvec}{2 & 5 \\ -4 & 3}\end{myvec}
			\begin{myvec}{x \\ y}\end{myvec}
			=
			\begin{myvec}{2 \\ -30}\end{myvec}
		\end{align}
		is
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		We can write the given system in the form of an augmented matrix and solve using row reduction.
		\begin{align}
			\myaugvec{2}{2 & 5 & 2 \\ -4 & 3 & -30}
		\end{align}
		
		Applying the row operation $R_2 \to R_2 + 2R_1$:
		\begin{align}
			\myaugvec{2}{2 & 5 & 2 \\ 0 & 13 & -26}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		Applying the row operation $R_2 \to \frac{1}{13}R_2$:
		\begin{align}
			\myaugvec{2}{2 & 5 & 2 \\ 0 & 1 & -2}
		\end{align}
		
		Applying the row operation $R_1 \to R_1 - 5R_2$:
		\begin{align}
			\myaugvec{2}{2 & 0 & 12 \\ 0 & 1 & -2}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		Applying the row operation $R_1 \to \frac{1}{2}R_1$:
		\begin{align}
			\myaugvec{2}{1 & 0 & 6 \\ 0 & 1 & -2}
		\end{align}
		
		From the reduced row echelon form, we get the solution:
		\begin{align}
			x &= 6 \\
			y &= -2
		\end{align}
		
		The solution vector is
		\begin{align}
			\vec{x} = \begin{myvec}{6 \\ -2}\end{myvec}
		\end{align}
		
	\end{frame}
