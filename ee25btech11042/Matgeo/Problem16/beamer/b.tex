	\begin{frame}{Question}
		\large
		A straight line L is perpendicular to the line $5x - y + 1 = 0$.
		The area of the triangle formed by line L and the coordinate axes is 5.
		
		\
		
		Find the equation of the line L.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		Let the given line be $L_1$ and the required perpendicular line be $L$.
		The equation of the given line $L_1$ is $5x - y = -1$.
		In matrix form $\vec{n}^\top\vec{x} = c$, its parameters are:
		\begin{align}
			\vec{n}_1 = \begin{myvec}{5 \\ -1}\end{myvec}, \quad \vec{x} = \begin{myvec}{x \\ y}\end{myvec}, \quad c_1 = -1
		\end{align}
		
		
		Let the required line $L$ have the equation $\vec{n}_2^\top\vec{x} = c_2$.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		Since the lines are perpendicular, the dot product of their normal vectors must be zero:
		\begin{align} 
			\vec{n}_1^\top\vec{n}_2 = 0 
		\end{align}
		
		
		Let's assume the normal vector for line L is $\vec{n}_2 = \begin{myvec}{1 \\ m}\end{myvec}$.
		We solve for $m$:
		\begin{align}
			\vec{n}_1^\top\vec{n}_2 = \begin{myvec}{5 & -1}\end{myvec}\begin{myvec}{1 \\ m}\end{myvec} &= 5 - m = 0 \\
			\implies m &= 5
		\end{align}
		
		
		Therefore, the normal vector for line L is $\vec{n}_2 = \begin{myvec}{1 \\ 5}\end{myvec}$, and its equation is:
		\begin{align} 
			x + 5y = c_2 
		\end{align}
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		We find the intercepts by solving the intersection of line L ($x+5y=c_2$) with each axis using Gaussian elimination.
		
		\
		
		x-intercept ($\vec{a}$): Intersection with x-axis ($y=0$)
		\begin{align}
			\myaugvec{2}{1 & 5 & c_2 \\ 0 & 1 & 0} \xleftrightarrow{\,R_1 \to R_1 - 5R_2} \myaugvec{2}{1 & 0 & c_2 \\ 0 & 1 & 0}
		\end{align}
		This gives the x-intercept vector $\vec{a} = \begin{myvec}{c_2 \\ 0}\end{myvec}$.
		
		\
		
		y-intercept ($\vec{b}$): Intersection with y-axis ($x=0$)
		\begin{align}
			\myaugvec{2}{1 & 5 & c_2 \\ 1 & 0 & 0} &\xleftrightarrow{\,R_1 \leftrightarrow R_2} \myaugvec{2}{1 & 0 & 0 \\ 1 & 5 & c_2} \\
			&\xleftrightarrow{\,R_2 \to R_2 - R_1} \myaugvec{2}{1 & 0 & 0 \\ 0 & 5 & c_2} \\
			&\xleftrightarrow{\,R_2 \to \frac{1}{5}R_2} \myaugvec{2}{1 & 0 & 0 \\ 0 & 1 & c_2/5}
		\end{align}
		This gives the y-intercept vector $\vec{b} = \begin{myvec}{0 \\ c_2/5}\end{myvec}$.
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		The area of the triangle formed by the origin, $\vec{a}$, and $\vec{b}$ is given as 5.
		
		Since the intercepts lie on the axes, the angle $\theta$ between them is $90^\circ$.
		\begin{align}
			\text{Area} = \frac{1}{2}\norm{\vec{a}}\norm{\vec{b}}\sin(90^\circ) &= 5 \\
			\frac{1}{2} \left(\sqrt{c_2^2}\right) \left(\sqrt{(c_2/5)^2}\right) (1) &= 5 \\
			\frac{1}{2} |c_2| \left|\frac{c_2}{5}\right|
			&= 5 \\
			\frac{|c_2^2|}{10} &= 5 \\
			c_2^2 = 50 \implies c_2 &= \pm 5\sqrt{2}
		\end{align}
	\end{frame}
	
	\begin{frame}{Theoretical solution}
		There are two possible lines that satisfy the given conditions.
		Substituting the values of $c_2$ into the equation, the possible equations for line L are:\\
		
		\
		
		\large $x + 5y = 5\sqrt{2}$\\
		
		\large $x + 5y = -5\sqrt{2}$
		
		\
		
		In matrix form:
		\begin{align}
			\begin{myvec}{1 & 5}\end{myvec}\begin{myvec}{x \\ y}\end{myvec} &= 5\sqrt{2} \\
			\begin{myvec}{1 & 5}\end{myvec}\begin{myvec}{x \\ y}\end{myvec} &= -5\sqrt{2}
		\end{align}
	\end{frame}
