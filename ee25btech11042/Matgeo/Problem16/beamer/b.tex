	\begin{frame}{Question}
		\large
		A straight line L is perpendicular to the line $5x - y + 1 = 0$. The area of the triangle formed by line L and the coordinate axes is 5.
		
		\textbf{Find the equation of the line L.}
	\end{frame}
	
	\begin{frame}{Step 1: Define the Lines}
		Let the given line be $L_1$ and the required perpendicular line be $L$.
		
		The equation of the given line $L_1$ is $5x - y = -1$.
		 
		
		In matrix form $\vec{n}^\top\vec{x} = c$, its parameters are:
		\begin{align}
			\vec{n}_1 = \begin{myvec}{5 \\ -1}\end{myvec}, \quad \vec{x} = \begin{myvec}{x \\ y}\end{myvec}, \quad c_1 = -1
		\end{align}
		
		 
		Let the required line $L$ have the equation $\vec{n}_2^\top\vec{x} = c_2$.
	\end{frame}
	
	\begin{frame}{Step 2: Find the Normal Vector of Line L}
		Since the lines are perpendicular, the dot product of their normal vectors must be zero:
		\[ \vec{n}_1^\top\vec{n}_2 = 0 \]
		
		 
		Let's assume the normal vector for line L is $\vec{n}_2 = \begin{myvec}{1 \\ m}\end{myvec}$. We solve for $m$:
		\begin{align}
			\vec{n}_1^\top\vec{n}_2 = \begin{myvec}{5 & -1}\end{myvec}\begin{myvec}{1 \\ m}\end{myvec} &= 5 - m = 0 \\
			\implies m &= 5
		\end{align}
		
		 
		Therefore, the normal vector for line L is $\vec{n}_2 = \begin{myvec}{1 \\ 5}\end{myvec}$, and its equation is:
		\[ x + 5y = c_2 \]
	\end{frame}
	
	\begin{frame}{Step 3: Finding the Intercepts}
		The line L forms a triangle with the coordinate axes.
		\begin{itemize}
			\item The y-axis equation is $x=0$, or $\vec{e_1}^\top\vec{x} = 0$, where $\vec{e_1} = \begin{myvec}{1 \\ 0}\end{myvec}$.
			\item The x-axis equation is $y=0$, or $\vec{e_2}^\top\vec{x} = 0$, where $\vec{e_2} = \begin{myvec}{0 \\ 1}\end{myvec}$.
		\end{itemize}
		
		 
		We find the intercepts by solving for the intersection of line L with each axis.
	\end{frame}
	
	\begin{frame}{Step 4: Calculating the x-intercept ($\vec{a}$)}
		The x-intercept, $\vec{a}$, is the intersection of line L ($x + 5y = c_2$) and the x-axis ($y=0$).
		
		We solve the stacked system:
		\begin{align}
			\begin{myvec}{1 & 5 \\ 0 & 1}\end{myvec}\vec{x} = \begin{myvec}{c_2 \\ 0}\end{myvec}
		\end{align}
		
		 
		The intercept vector $\vec{a}$ is found by inverting the matrix:
		\begin{align}
			\vec{a} = \begin{myvec}{1 & 5 \\ 0 & 1}\end{myvec}^{-1} \begin{myvec}{c_2 \\ 0}\end{myvec} = \begin{myvec}{1 & -5 \\ 0 & 1}\end{myvec}\begin{myvec}{c_2 \\ 0}\end{myvec} = \begin{myvec}{c_2 \\ 0}\end{myvec}
		\end{align}
	\end{frame}
	
	\begin{frame}{Step 5: Calculating the y-intercept ($\vec{b}$)}
		The y-intercept, $\vec{b}$, is the intersection of line L ($x + 5y = c_2$) and the y-axis ($x=0$).
		
		We solve the stacked system:
		\begin{align}
			\begin{myvec}{1 & 5 \\ 1 & 0}\end{myvec}\vec{x} = \begin{myvec}{c_2 \\ 0}\end{myvec}
		\end{align}
		
		 
		The intercept vector $\vec{b}$ is:
		\begin{align}
			\vec{b} = \begin{myvec}{1 & 5 \\ 1 & 0}\end{myvec}^{-1} \begin{myvec}{c_2 \\ 0}\end{myvec} = \frac{1}{-5}\begin{myvec}{0 & -5 \\ -1 & 1}\end{myvec} \begin{myvec}{c_2 \\ 0}\end{myvec} = \begin{myvec}{0 \\ c_2/5}\end{myvec}
		\end{align}
	\end{frame}
	
	
	\begin{frame}{Step 6: Using the Area to Find $c_2$}
		The area of the triangle formed by the origin, $\vec{a}$, and $\vec{b}$ is given as 5.
		
		Since the intercepts lie on the axes, the angle $\theta$ between them is $90^\circ$.
		
		\begin{align}
			\text{Area} = \frac{1}{2}\norm{\vec{a}}\norm{\vec{b}}\sin(90^\circ) &= 5 \\
			\frac{1}{2} \left(\sqrt{c_2^2}\right) \left(\sqrt{(c_2/5)^2}\right) (1) &= 5 \\
			\frac{1}{2} |c_2| \left|\frac{c_2}{5}\right| &= 5 \\
			\frac{|c_2^2|}{10} &= 5 \\
			c_2^2 = 50 \implies c_2 &= \pm 5\sqrt{2}
		\end{align}
	\end{frame}
	
	\begin{frame}{Solution: The Equations of Line L}
		There are two possible lines that satisfy the given conditions.
		
		Substituting the values of $c_2$ into the equation $\vec{n}_2^\top\vec{x} = c_2$, the possible equations for line L are:\\
		
		
		
			 \large $x + 5y = 5\sqrt{2}$\\
			
		\large $x + 5y = -5\sqrt{2}$
		
		
		
		In matrix form:
		\begin{align}
			\begin{myvec}{1 & 5}\end{myvec}\begin{myvec}{x \\ y}\end{myvec} &= 5\sqrt{2} \\
			\begin{myvec}{1 & 5}\end{myvec}\begin{myvec}{x \\ y}\end{myvec} &= -5\sqrt{2}
		\end{align}
	\end{frame}
