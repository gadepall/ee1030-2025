	\begin{frame}{Question}
		
		If the area of a triangle with the vertices $\vec{A}\brak{k, 0}$, $\vec{B}\brak{2, 0}$ and $\vec{C}\brak{0, -2}$ is 2 square units, find the value of k.
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		Given the vertices:
		\begin{align}
			\vec{A} = \begin{myvec}{k\\0} \end{myvec} , \vec{B} = \begin{myvec}{2\\0} \end{myvec}, \vec{C} = \begin{myvec}{0\\-2} \end{myvec}
		\end{align}
		
		We define two side vectors originating from vertex $\vec{C}$.
		
		Vector $\vec{a}$ (from $\vec{C}$ to $\vec{B}$):
		\begin{align}
			\vec{a} = \vec{B}-\vec{C} = \begin{myvec}{2 - 0\\0 - (-2)} \end{myvec} = \begin{myvec}{2\\2} \end{myvec}
		\end{align}
		Vector $\vec{b}$ (from $\vec{C}$ to $\vec{A}$):
		\begin{align}
			\vec{b} = \vec{A}-\vec{C} = \begin{myvec}{k - 0\\0 - (-2)} \end{myvec} = \begin{myvec}{k\\2} \end{myvec}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		The area is given by $\Delta = \frac{1}{2}\norm{\vec{b}\times\vec{a}}$.
		
		Since the area is 2, we have:
		\begin{align}
			\frac{1}{2}\norm{\vec{b}\times\vec{a}} = 2 \implies \norm{\vec{b}\times\vec{a}} = 4
		\end{align}
		
		To compute the cross product, we represent the vectors in 3D space:
		\begin{align}
			\vec{b} = \begin{myvec}{k\\2\\0} \end{myvec} && \vec{a} = \begin{myvec}{2 \\2\\0} \end{myvec}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		Using the formula $\vec{A}\times\vec{B} = \begin{myvec}{|\vec{A}_{23} & \vec{B}_{23}|\\|\vec{A}_{31} & \vec{B}_{31}|\\|\vec{A}_{12} & \vec{B}_{12}|} \end{myvec}$:
		
		\begin{align}
			|\vec{b}_{23} \text{ } \vec{a}_{23}| &= \begin{mydet}{2 & 2\\ 0 & 0} \end{mydet} = 0 \\
			|\vec{b}_{31} \text{ } \vec{a}_{31}| &= \begin{mydet}{0 & 0\\ k & 2} \end{mydet} = 0 \\
			|\vec{b}_{12} \text{ } \vec{a}_{12}| &= \begin{mydet}{k & 2\\ 2 & 2} \end{mydet} = 2k - 4
		\end{align}
		
		The cross product vector is:
		\begin{align}
			\vec{b}\times\vec{a} = \begin{myvec}{0 \\ 0 \\ 2k - 4} \end{myvec}
		\end{align}
		
		The norm is $\norm{\vec{b}\times\vec{a}} = \sqrt{0^2+0^2+(2k-4)^2} = |2k-4|$.
		
	\end{frame}
	
	\begin{frame}{Theoretical Solution}
		
		We now solve for k using the area condition:
		\begin{align}
			|2k - 4| = 4
		\end{align}
		
		This absolute value equation yields two possibilities:
		
		Case 1:
		\begin{align}
			2k - 4 = 4 \implies 2k = 8 \implies k = 4
		\end{align}
		
		Case 2:
		\begin{align}
			2k - 4 = -4 \implies 2k = 0 \implies k = 0
		\end{align}
		
		The possible values for k are $0$ or $4$.
		
	\end{frame}
