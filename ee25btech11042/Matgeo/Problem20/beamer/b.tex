	\begin{frame}{Question}
		
		The points $\vec{A}\brak{2, 9}$, $\vec{B}\brak{a, 5}$ and $\vec{C}\brak{5, 5}$ are the vertices of a triangle $\triangle\vec{ABC}$ right angled at $\vec{B}$.
		
		Find the values of a and hence the area of $\triangle\vec{ABC}$.
		
	\end{frame}
	
	\begin{frame}{Solution: Setting up the Vectors}
		
		Given the vertices:
		\begin{align}
			\vec{A} = \begin{myvec}{2\\9} \end{myvec} , \vec{B} = \begin{myvec}{a\\5} \end{myvec}, \vec{C} = \begin{myvec}{5\\5} \end{myvec}
		\end{align}
		
		We define the side vectors originating from the right-angled vertex $\vec{B}$.
		
		Vector $\vec{c}$ (from $\vec{B}$ to $\vec{A}$):
		\begin{align}
			\vec{c} = \vec{A}-\vec{B} = \begin{myvec}{2 -a\\9 - 5} \end{myvec} = \begin{myvec}{2 -a\\4} \end{myvec}
		\end{align}
		Vector $\vec{a}$ (from $\vec{B}$ to $\vec{C}$):
		\begin{align}
			\vec{a} = \vec{C}-\vec{B} = \begin{myvec}{5 - a\\5 - 5} \end{myvec} = \begin{myvec}{5 - a \\0} \end{myvec}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Solution: Finding 'a' using Orthogonality}
		
		Since the triangle is right-angled at $\vec{B}$, the vectors $\vec{c}$ and $\vec{a}$ are orthogonal. Their inner product must be zero.
		
		\begin{align}
			\vec{c}^T\vec{a} = \begin{myvec}{2 -a & 4} \end{myvec}\begin{myvec}{5 - a \\0} \end{myvec} = 0
		\end{align}
		
		Solving the equation:
		\begin{align}
			\brak{2-a}\brak{5-a} + (4)(0) &= 0 \\
			\brak{2-a}\brak{5-a} &= 0
		\end{align}
		
		This gives two possible values: $a = 2$ or $a = 5$.
		
		If $a = 5$, vertex $\vec{B}$ becomes $(5,5)$, which is the same as vertex $\vec{C}$. This would not form a triangle.
		
		Therefore, we must have $a = 2$.
		
	\end{frame}
	
	\begin{frame}{Solution: Calculating the Area (Setup)}
		
		With $a = 2$, the side vectors are:
		\begin{align}
			\vec{c} &= \begin{myvec}{2 - 2\\4} \end{myvec} = \begin{myvec}{0\\4} \end{myvec} \\
			\vec{a} &= \begin{myvec}{5 - 2\\0} \end{myvec} = \begin{myvec}{3 \\0} \end{myvec}
		\end{align}
		
		The area is calculated using the cross product formula:
		\begin{align}
			\Delta = \frac{1}{2}\norm{\vec{c}\times\vec{a}}
		\end{align}
		
		In 3D space, the vectors are:
		\begin{align}
			\vec{c} = \begin{myvec}{0\\4\\0} \end{myvec} && \vec{a} = \begin{myvec}{3 \\0\\0} \end{myvec}
		\end{align}
		
	\end{frame}
	
	\begin{frame}{Solution: Calculating the Cross Product}
		
		Using the component-wise formula $\vec{A}\times\vec{B} = \begin{myvec}{|\vec{A}_{23} & \vec{B}_{23}|\\|\vec{A}_{31} & \vec{B}_{31}|\\|\vec{A}_{12} & \vec{B}_{12}|} \end{myvec}$:
		
		\begin{align}
			|\vec{c}_{23} \text{ } \vec{a}_{23}| &= \begin{mydet}{4 & 0\\ 0 & 0} \end{mydet} = 0 \\
			|\vec{c}_{31} \text{ } \vec{a}_{31}| &= \begin{mydet}{0 & 0\\ 0 & 3} \end{mydet} = 0 \\
			|\vec{c}_{12} \text{ } \vec{a}_{12}| &= \begin{mydet}{0 & 4\\ 3 & 0} \end{mydet} = 0 - 12 = -12
		\end{align}
		
		The cross product vector is:
		\begin{align}
			\vec{c}\times\vec{a} = \begin{myvec}{0 \\ 0 \\ -12} \end{myvec}
		\end{align}
		
		The norm is $\norm{\vec{c}\times\vec{a}} = \sqrt{0^2+0^2+(-12)^2} = 12$.
		
	\end{frame}
	
	\begin{frame}{Solution: Final Area}
		
		Substituting the norm into the area formula:
		\begin{align}
			\Delta = \frac{1}{2}\norm{\vec{c}\times\vec{a}} = \frac{1}{2}(12) = 6
		\end{align}
		
		The area of the triangle is $6$ square units.
		
	\end{frame}
