	\begin{frame}{Question}
	Solve the following system of rational equations
	\begin{align}
		\frac{10}{x+y}+\frac{2}{x-y} = 4
	\end{align}
	\begin{align}
		\frac{15}{x+y}-\frac{5}{x-y} = -2
	\end{align}
	
	\end{frame}
	
	
	\begin{frame}{Theoretical Solution}
			We convert the system of rational equations from \eqref{eq:rational_1} and \eqref{eq:rational_2} into the standard quadratic form for conic sections.
		
		For the first equation \eqref{eq:rational_1}, we multiply by $(x+y)(x-y)$:
		\begin{align}
			10(x-y) + 2(x+y) &= 4(x^2 - y^2) \nonumber \\
			12x - 8y &= 4x^2 - 4y^2 \nonumber \\
			\implies x^2 - y^2 - 3x + 2y &= 0 \label{eq:H1}
		\end{align}
		
		For the second equation \eqref{eq:rational_2}, we do the same:
		\begin{align}
			15(x-y) - 5(x+y) &= -2(x^2 - y^2) \nonumber \\
			10x - 20y &= -2x^2 + 2y^2 \nonumber \\
			\implies x^2 - y^2 + 5x - 10y &= 0 \label{eq:H2}
		\end{align}
	\end{frame}
	\begin{frame}{Theoretical Solution}
	 	The intersection points of the two hyperbolas from \eqref{eq:H1} and \eqref{eq:H2} lie on a degenerate conic (a pair of lines). We find this by creating a linear combination of the two conics, $H_1 + \mu H_2 = 0$, and finding $\mu$ such that the determinant of the resulting conic matrix is zero.
	 
	 The parameters for each hyperbola in the form $\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f = 0$ are:
	 \begin{align}
	 	\text{For } H_1: \quad \vec{V}_1 = \begin{myvec}{1 & 0 \\ 0 & -1}\end{myvec}, \quad \vec{u}_1 = \begin{myvec}{-3/2 \\ 1}\end{myvec}, \quad f_1 = 0 \\
	 	\text{For } H_2: \quad \vec{V}_2 = \begin{myvec}{1 & 0 \\ 0 & -1}\end{myvec}, \quad \vec{u}_2 = \begin{myvec}{5/2 \\ -5}\end{myvec}, \quad f_2 = 0
	 \end{align}
	 
	 The condition for a degenerate conic is:
	 \begin{align}
	 	\begin{vmatrix} \vec{V}_1 + \mu\vec{V}_2 & \vec{u}_1 + \mu\vec{u}_2 \\ (\vec{u}_1 + \mu\vec{u}_2)^\top & f_1 + \mu f_2 \end{vmatrix} = 0
	 \end{align}
	\end{frame}
		\begin{frame}{Theoretical Solution}
		
Since $\vec{V}_1 = \vec{V}_2$, choosing $\mu = -1$ makes the quadratic term $\vec{V}_1 - \vec{V}_2$ a zero matrix, which guarantees a degenerate conic. The equation for this conic is found by subtracting \eqref{eq:H2} from \eqref{eq:H1}:
\begin{align}
	(x^2 - y^2 - 3x + 2y) - (x^2 - y^2 + 5x - 10y) &= 0 \nonumber \\
	-8x + 12y &= 0 \nonumber \\
	\implies y &= \frac{2}{3}x \label{eq:common_chord}
\end{align}
This linear equation is the common chord passing through the intersection points.


We now find the intersection of the common chord from \eqref{eq:common_chord} with the first hyperbola from \eqref{eq:H1} using the formula $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$.
	\end{frame}
		\begin{frame}{Theoretical Solution}
		First, we define the line and conic parameters:
		
		\textbf{Line}: From \eqref{eq:common_chord}, the line passes through the origin, so $\vec{h} = \begin{myvec}{0 \\ 0}\end{myvec}$. The direction vector is $\vec{m} = \begin{myvec}{3 \\ 2}\end{myvec}$.\\
		\textbf{Conic}: We use the parameters for $H_1$, so $\vec{V} = \vec{V}_1$, $\vec{u} = \vec{u}_1$, and $f = f_1$.
		
		
		$\kappa_i$:
		\begin{align}
			\vec{m}^\top\vec{V}\vec{m} &= \begin{myvec}{3 & 2}\end{myvec}\begin{myvec}{1 & 0 \\ 0 & -1}\end{myvec}\begin{myvec}{3 \\ 2}\end{myvec} = 5 \\
			g(\vec{h}) &= \vec{h}^\top\vec{V}\vec{h} + 2\vec{u}^\top\vec{h} + f = 0 \\
			\vec{m}^\top(\vec{V}\vec{h} + \vec{u}) &= \vec{m}^\top\vec{u} = \begin{myvec}{3 & 2}\end{myvec}\begin{myvec}{-3/2 \\ 1}\end{myvec} = -\frac{5}{2}
		\end{align}
		
	\end{frame}	
	\begin{frame}{Theoretical Solution}
		We substitute these into the formula for $\kappa_i$:
	\begin{align}
		\kappa_i &= \frac{1}{\vec{m}^\top\vec{V}\vec{m}} \left( - \vec{m}^\top(\vec{V}\vec{h} + \vec{u}) \pm \sqrt{[\vec{m}^\top(\vec{V}\vec{h} + \vec{u})]^2 - g(\vec{h})(\vec{m}^\top\vec{V}\vec{m})} \right) \nonumber \\
		&= \frac{1}{5} \left( -(-\frac{5}{2}) \pm \sqrt{(-\frac{5}{2})^2 - (0)(5)} \right) = \frac{1}{5} \left( \frac{5}{2} \pm \frac{5}{2} \right)
	\end{align}
	This gives two values: $\kappa_1 = 1$ and $\kappa_2 = 0$.
	
	\end{frame}
	\begin{frame}{Theoretical Solution}
			Finally, we find the intersection points using $\vec{x}_i = \vec{h} + \kappa_i\vec{m}$:
		\begin{align}
			\vec{x}_1 &= \begin{myvec}{0 \\ 0}\end{myvec} + 1 \begin{myvec}{3 \\ 2}\end{myvec} = \begin{myvec}{3 \\ 2}\end{myvec} \label{eq:sol1} \\
			\vec{x}_2 &= \begin{myvec}{0 \\ 0}\end{myvec} + 0 \begin{myvec}{3 \\ 2}\end{myvec} = \begin{myvec}{0 \\ 0}\end{myvec} \label{eq:sol2}
		\end{align}
		
		The intersection points are $(3,2)$ from \eqref{eq:sol1} and $(0,0)$ from \eqref{eq:sol2}. The point $(0,0)$ is extraneous because it causes division by zero in the original equations \eqref{eq:rational_1} and \eqref{eq:rational_2}. Thus, the only valid solution is $x=3, y=2$.
		
	\end{frame}

# --- Set up the plotting grid ---
# Generate a grid of points to evaluate the equations on
x_vals = np.linspace(-10, 15, 500)
y_vals = np.linspace(-10, 15, 500)
X, Y = np.meshgrid(x_vals, y_vals)
# --- Define the hyperbola equations ---
# Equation is x^T V x + 2u^T x + f = 0
# For a point (x,y), the vector is [x, y]
# So x^T V x becomes V[0,0]*x^2 + V[1,1]*y^2
# and 2u^T x becomes 2*(u[0]*x + u[1]*y)
eq1 = V1[0,0]*X**2 + V1[1,1]*Y**2 + 2*(u1[0]*X + u1[1]*Y) + f1
