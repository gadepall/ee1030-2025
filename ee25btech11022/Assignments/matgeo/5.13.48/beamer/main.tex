\begin{frame}
\frametitle{Question}
If $\vec{A}$ and $\vec{B}$ are square matrices of equal degree, then which one is correct among the followings? 
\begin{enumerate}
    \item $\vec{A} + \vec{B} = \vec{B} + \vec{A}$
    \item $\vec{A} + \vec{B} = \vec{A} - \vec{B}$
    \item $\vec{A} - \vec{B} = \vec{B} - \vec{A}$
    \item $\vec{A}\vec{B} = \vec{B}\vec{A}$
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Solution}
For any two matrices $\vec{A}$ and $\vec{B}$ of the same order, addition is commutative
 Therefore, $\vec{A}+\vec{B} = \vec{B}+\vec{A}$. \\
 let
\begin{align}
\vec{A} = \myvec{2 &1 \\3 &6},\vec{B} = \myvec{5 &3 \\ 8 &2}     
\end{align}

\begin{align}
 \vec{A}+\vec{B} &= \myvec{2+5 &3+1 \\3+8 &6+2} \\  
 &=\myvec{7 &4 \\11 &8}
\end{align}
\begin{align}
 \vec{B}+\vec{A} &= \myvec{5+2 &1+3 \\8+3 &2+6} \\  
 &=\myvec{7 &4 \\11 &8}
\end{align}
Therefore, $\vec{A}+\vec{B} = \vec{B}+\vec{A}$.
\end{frame}
