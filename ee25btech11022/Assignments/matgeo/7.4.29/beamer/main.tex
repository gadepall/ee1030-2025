\begin{frame}
\frametitle{Question}
If one of the diameters of the circle $x^2 + y^2 - 2x - 6y + 6 = 0$ is a chord to the circle with the center $\brak{2,1}$,then the radius of the circle is 
\begin{enumerate}
     \item $\sqrt{3}$
     \item $\sqrt{2}$
     \item $3$
     \item $2$
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Solution}
Given the circle equation:
\begin{align}
x^2 + y^2 - 2x - 6y + 6 = 0 \\
\mathbf{x}^T \mathbf{x} + 2\mathbf{u}^T \mathbf{x} + f = 0,\text{where}
\mathbf{x} = \myvec{x \\ y}, \quad \mathbf{u} = \myvec{g \\ f}
\end{align}
From the given equation:
\begin{align}
2\mathbf{u}^T \mathbf{x} = -2x - 6y \implies \mathbf{u} = \myvec{-1 \\ -3}, \quad f = 6
\end{align}
Center of the circle is:
\begin{align}
\mathbf{c_1} = -\mathbf{u} = \myvec{1 \\ 3}
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}
Radius of the circle is:
\begin{align}
r_1 =\sqrt{\mathbf{u}^T \mathbf{u} - f} =\sqrt{(-1)^2 + (-3)^2 - 6} =\sqrt{4} = 2
\end{align}
Given the second circle has center:
\begin{align}
\mathbf{c_2} = \myvec{2 \\ 1}
\end{align}
Assuming the diameter chord line of the first circle with equation $y=3$,
\begin{align}
\mathbf{n} = \myvec{0 \\ 1}, \quad c = 3
\end{align}
Distance from second circle center \(\mathbf{c_2}\) to this chord:
\begin{align}
\frac{|\mathbf{n}^T \mathbf{c_2} - c|}{\|\mathbf{n}\|} = \frac{|1 - 3|}{1} = 2
\end{align}
Therefore, the radius of the second circle  r=2.
\end{frame}

// code.c
#include <stdio.h>

void get_circle_data(double *c1, double *r1, double *c2, double *r2, double *chord_y) {
    // Circle 1 center (1,3), radius 2
    c1[0] = 1; c1[1] = 3;
    *r1 = 2;

    // Circle 2 center (2,1), unknown radius
    c2[0] = 2; c2[1] = 1;
    *r2 = 2;  // radius determined by chord distance

    // chord line y=3
    *chord_y = 3;
}

get_circle_data = lib.get_circle_data
get_circle_data.argtypes = [
    np.ctypeslib.ndpointer(dtype=np.float64, shape=(2,)),
    ctypes.POINTER(ctypes.c_double),
    np.ctypeslib.ndpointer(dtype=np.float64, shape=(2,)),
    ctypes.POINTER(ctypes.c_double),
    ctypes.POINTER(ctypes.c_double)
]
get_circle_data.restype = None
def get_data():
    c1 = np.zeros(2, dtype=np.float64)
    r1 = ctypes.c_double()
    c2 = np.zeros(2, dtype=np.float64)
    r2 = ctypes.c_double()
    chord_y = ctypes.c_double()

    get_circle_data(c1, ctypes.byref(r1), c2, ctypes.byref(r2), ctypes.byref(chord_y))
    return c1, r1.value, c2, r2.value, chord_y.value
