\begin{frame}
\frametitle{Question}
Find the equations of the lines through the point of intersection of the line  $x - y + 1 = 0$ and $2x - 3y + 5 = 0$ and whose distance from the point $\brak{3, 2}$ is $\frac{7}{5}$. 
\end{frame}

\begin{frame}
\frametitle{Solution}
Given lines are 
\begin{align}
\myvec{ 1 & -1 } \myvec{ x \\ y } + 1 = 0 ,
\myvec{ 2 & -3 } \myvec{ x \\ y } + 5 = 0  \\
\myvec{ 1 & -1 \\ 2 & -3 } \myvec{ x \\ y } = \myvec{ -1 \\ -5 }
\end{align}
Write as augmented matrix \\
\begin{align}
\augvec{2}{1}{1 & -1 & -1\\ 2 & -3 & -5} \xrightarrow{R_2 \rightarrow R_2-2R_1}\augvec{2}{13}{1 & -1 & -1\\ 0 & -1 & -3}
\end{align}
From second row:  $-y = -3 \implies y = 3$ \\ 
Substitute in first row:  $x - 3 = -1 \implies x = 2$ \\
Intersection point of given lines is $\myvec{2 \\ 3}$ 
\end{frame}

\begin{frame}
\frametitle{Solution}
General line through $\brak{x_0,y_0}$ with normal vector \myvec{a \\ b} is
\begin{align}
\myvec{a & b}\brak{\myvec{x \\ y} -\myvec{2 \\ 3}} = 0 
\end{align}
Distance from point $\myvec{3 \\ 2}$  to line is
\begin{align}
Distance = \frac{\lvert\myvec{a & b}\brak{\myvec{3 \\ 2}-\myvec{2 \\ 3}}\rvert}{\sqrt{\myvec{a & b}\myvec{a \\ b}}} =\frac{7}{5} \\
\frac{\lvert\myvec{a & b}\brak{\myvec{3-2 \\ 2-3}-\myvec{2 \\ 3}}\rvert}{\sqrt{
\myvec{a & b}\myvec{a \\ b}}} =\frac{7}{5} 
\end{align}
\end{frame}

\begin{frame}
\frametitle{Solution}
$\frac{|a - b|}{\sqrt{a^2 + b^2}} = \frac{7}{5}$ \\
Let t = $\frac{a}{b}\brak{b \neq 0}$ \\
\begin{align}
\frac{|t - 1|}{\sqrt{t^2 + 1}} = \frac{7}{5} \\
\end{align}
Squaring  on both sides
\begin{align}
25 (t - 1)^2 &= 49 (t^2 + 1) \\
25 (t^2 - 2 t + 1) &= 49 t^2 + 49 \\
25 t^2 - 50 t + 25 &= 49 t^2 + 49 \\
24 t^2 + 50 t + 24 &= 0 \\
12 t^2 + 25 t + 12 &= 0 \\
t = \frac{-25\pm \sqrt{25^2-4\times12\times12}}{2\times12} =\frac{-25\pm 7}{24} 
\end{align}
\end{frame}

\begin{frame}
\frametitle{Solution}
$t_1 = -\frac{3}{4},t_2 = -1$ \\
Normal vectors $\brak{let b=1}$:
\begin{align}
\myvec{a \\ b} =\myvec{\frac{-3}{4} \\ 1},\myvec{-1 \\ 1}
\end{align}
Equation constants \quad c = -a$\brak{2}$ - b$\brak{3}$
\begin{align}
c_1 =\brak{-2}\times\brak{-\frac{3}{4}} -3\times1 =\frac{3}{2} -3 =\frac{-3}{2} \\
c_2 = \brak{-2}\times\brak{-1} - 3\times1 = 2 -3= -1 
\end{align}

required equations of line are  
\begin{align}
\myvec{\frac{-3}{4} & 1} \myvec{x \\ y} -\frac{3}{2} = 0 \\
\myvec{-1 & 1} \myvec{x \\ y} - 1 = 0 \\
\end{align}
\end{frame}

// code.c
#include <stdio.h>
#include <math.h>

// Solve quadratic ax^2 + bx + c = 0
void solve_quadratic(double a, double b, double c, double *r1, double *r2) {
    double d = b*b - 4*a*c;
    if (d < 0) {
        *r1 = NAN;
        *r2 = NAN;
        return;
    }
    *r1 = (-b + sqrt(d)) / (2*a);
    *r2 = (-b - sqrt(d)) / (2*a);
}
\end{lstlisting}

void get_line_params(double *a1, double *a2, double *c1, double *c2) {
    double A = 1.0, B = 25.0/12.0, C = 1.0;
    double r1, r2;
    solve_quadratic(A, B, C, &r1, &r2);

    a1[0] = r1;
    a2[0] = r2;

    // Constants c = 2a + 3, intersection point fixed at (2,3)
    c1[0] = 2*r1 + 3;
    c2[0] = 2*r2 + 3;
}
