\begin{frame}
\frametitle{Question}
Find the equation of the line passing through the point $\brak{1, -3, 2}$ and parallel to the line  $\vec{r} = \brak{2 + \lambda}\hat{i} + \lambda\hat{j} + \brak{-1+\lambda}\hat{k}$
\end{frame}

\begin{frame}
\frametitle{Solution}
Given line is
\begin{align}
 \vec{r} = \myvec{2+\lambda\\\lambda\\-1+\lambda} \quad \vec{a}=\myvec{1\\-3\\2}
\end{align}
The vector equation of given line is given by
\begin{align}
\vec{r}=\myvec{2\\0\\-1}+\lambda\myvec{1\\1\\1}
\end{align}
The direction vectors of given line are
\begin{align}
    \vec{m}=\myvec{1\\1\\1}
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}
The lines with direction vectors $\vec{m}$ and $\vec{n}$ are parallel if
\begin{align}
\vec{m}=\vec{n}  \implies  \vec{n}=\myvec{1\\1\\1}
\end{align}
The equation of a line is given by
\begin{align}
\vec{r}=\vec{a}+t\vec{n}
\end{align}
\begin{align}
\vec{r}=\myvec{1\\-3\\2}+t\myvec{1\\1\\1}
\end{align}
\end{frame}

// code.c
#include <stdio.h>

void line_points(double* xs, double* ys, double* zs, int n) {
    double x0 = 1, y0 = -3, z0 = 2;  // given point
    double dx = 1, dy = 1, dz = 1;   // direction vector
    for (int i = 0; i < n; ++i) {
        double lambda = (i - n/2);   // range [-n/2, n/2]
        xs[i] = x0 + lambda * dx;
        ys[i] = y0 + lambda * dy;
        zs[i] = z0 + lambda * dz;
    }
}

void normal_vector(double* normal) {
    normal[0] = 1;
    normal[1] = 1;
    normal[2] = 1;
}
