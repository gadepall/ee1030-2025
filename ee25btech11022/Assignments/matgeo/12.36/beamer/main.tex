\begin{frame}
\frametitle{Question}
Given the regression model:$Y_i = \alpha + \beta x_i + \epsilon_i$ with data:  \\ \begin{align}
\begin{tabular}{|c|c|}\hline
\text{Year} & \text{Production} \\ \hline
2001 & 30 \\ \hline
2002 & 35 \\ \hline
2003 & 36 \\ \hline
2004 & 32 \\ \hline
2005 & 37 \\ \hline
2006 & 40 \\ \hline
\end{tabular}
\end{align}
\end{frame}

\begin{frame}
\frametitle{Solution}
Let
\begin{align}
x_i = \myvec{1 \\ 2 \\ 3 \\ 4 \\ 5 \\ 6}, Y = \myvec{30 \\ 35 \\ 36 \\ 32 \\ 37 \\ 40}
\end{align}

Let
\begin{align}
X = \myvec{1 & 1 \\1 & 2 \\1 & 3 \\1 & 4 \\1 & 5 \\1 & 6}
\end{align}
\end{frame}

\begin{frame}
\frametitle{Solution}
\begin{align}
X^T X =\myvec{6 & 21 \\21 & 91},X^T Y =\myvec{210 \\761} \\
\brak{X^T X}^{-1} =\myvec{0.87 & -0.20 \\-0.20 & 0.06} \\
\myvec{\alpha \\\beta}= \brak{X^T X}^{-1}\brak{X^T Y} \\
=\myvec{0.87 & -0.20 \\-0.20 & 0.06} \\
=\myvec{210 \\761}=\myvec{29.80 \\1.49}
\end{align}
Thus, $\alpha + \beta = 31.29$
\end{frame}

#include <stdio.h>

void get_data(double *years, double *production, int *n) {
    // Fill arrays with dataset, n=6
    double local_years[6] = {2001, 2002, 2003, 2004, 2005, 2006};
    double local_prod[6] = {30, 35, 36, 32, 37, 40};
    for(int i=0; i<6; i++) {
        years[i] = local_years[i];
        production[i] = local_prod[i];
    }
    *n = 6;
}

void get_stats(double *years, double *production, int n, double *out) {
    // Linear regression using normal equations, years mapped to 1,2,...,6
    double xsum=0, ysum=0, x2sum=0, xysum=0;
    for(int i=0; i<n; i++) {
        double x = i+1;
        double y = production[i];
        xsum += x;
        ysum += y;
        x2sum += x*x;
        xysum += x*y;
    }
    double beta = (n*xysum - xsum*ysum) / (n*x2sum - xsum*xsum);
    double alpha = (ysum - beta*xsum) / n;
    out[0] = alpha;
    out[1] = beta;
    out[2] = alpha + beta;
}
