\begin{frame}
\frametitle{Question}
The equation of directrix of parabola $y^2+4y+4x+2=0$
\begin{enumerate}
     \item $x=-1$
     \item $x=\frac{-3}{2}$
     \item $x=1$
     \item $x=\frac{3}{2}$
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Solution}
Given the parabola equation:
\begin{align}
y^2 + 4y + 4x + 2 = 0
\end{align}
The equation of a conic with directrix $\vec{n}^\top \vec{x} = c$, eccentricity $e$ and focus $\vec{F}$ is given by
\begin{align}
\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f = 0,\vec{x} = \myvec{x \\ y},
\mathbf{V} =\myvec{0 & 0 \\0 & 1},\vec{u} =\myvec{2 \\ 2},f = 2
\end{align}
Find vertex $\vec{c}$ from
\begin{align}
\mathbf{V} \vec{c} &= -\vec{u}  \\
\myvec{0 & 0 \\0 & 1}\myvec{c_x \\c_y} &=-\myvec{2 \\ 2} \\
\myvec{0 \\c_y} &=-\myvec{2 \\ 2} \\
c_y &= -2
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}
Put $c_y=-2$ into parabola to find  $c_x$:
\begin{align}
(-2)^2 + 4(-2) + 4 c_x + 2 = 0  \Rightarrow  c_x = \frac{1}{2}
\vec{c} = \myvec{ \frac{1}{2} \\ -2 }
\end{align}

Eigenvector for zero eigenvalue of $\vec{V}$:
\begin{align}
\vec{a} =\myvec{1 \\ 0}
p = \frac{\|\vec{u}\|^2 - f}{4 \, \vec{u}^T \vec{a}} = \frac{4 + 4 - 2}{8} = \frac{6}{8} = \frac{3}{4}
\end{align}
Equation of directrix (line perpendicular to $\vec{a}$,distance p from vertex):
\begin{align}
\vec{n}^T \vec{x} = \vec{n}^T \vec{c} + p,\text{where}\vec{n}=\vec{a}=\myvec{1\\0}
\vec{x} = \frac{1}{2} + \frac{3}{4} = \frac{5}{4}
\end{align}
Therefore, the equation of the directrix is approximately:
$\vec{x} = \frac{3}{2}$
\end{frame}

// code.c
#include <stdio.h>

void get_parabola_data(double *vertex, double *p, double *directrix_x) {
    // vertex (1/2, -2)
    vertex[0] = 0.5;
    vertex[1] = -2.0;

    // parameter p = 3/4
    *p = 0.75;

    // directrix x = vertex_x + p
    *directrix_x = vertex[0] + *p;
}

    vertex = np.zeros(2, dtype=np.float64)
    p = ctypes.c_double()
    directrix_x = ctypes.c_double()

    return vertex, p.value, directrix_x.value
