\begin{frame}
\frametitle{Question}
If \brak{-4,3} and \brak{4,3} are two vertices of an equilateral triangle. Find the coordinates of the third vertex, given that the origin lies in the interior of the triangle.
\end{frame}
\begin{frame}
\frametitle{Solution}
Let 
\begin{align}
 \Vec{A} = \myvec{-4 \\ 3},\Vec{B} = \myvec{4 \\ 3} and  \Vec{C} = \myvec{x \\ y}
 \end{align}be the vertices of eqilateral triangle.\\
 The direction vector of the segment joining A and B is given by:
\begin{align}
\vec{B} - \vec{A} = \myvec{4 - (-4) \\ 3 - 3)} = \myvec{8 \\ 0} 
\end{align}
The side length of equilateral triangle is 
\begin{align} 
\|\vec{B-A}\| &= \sqrt{\brak{\vec{B}-\vec{A}}^\top \brak{\vec{B}-\vec{A}}}  \\
&=\sqrt{\myvec{8 \ 0}\myvec{8 \\ 0}} \\
&=\sqrt{8(8)+0(0)} \\
&=8
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}
Let $\vec{M}$ be the midpoint of $\vec{A}$ and $\vec{B}$
\begin{align}
\vec{M} &=\frac{\vec{A} + \vec{B}}{2} \\
&= \frac{\myvec{-4 \\ 3} + \myvec{4 \\ 3}}{2} \\
&= \frac{\myvec{0 \\ 6}}{2} \\
&= \myvec{0 \\ 3}
\end{align}
Height of equilateral triangle 
\begin{align} 
h = \frac{\sqrt{3}}{2} \cdot (8) = 4\sqrt{3}
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}

The point $\vec{C}$ lies in the perpendicular bisector drawn from $\vec{M}$. Therefore,the possible values of $\vec{C}$ are 
\begin{align}
    \vec{C} = \myvec{0 \\ 3+4\sqrt{3}} (or) \myvec{0 \\ 3-4\sqrt{3}} 
\end{align}
Since the origin is inside the triangle, choose the lower vertex:
\begin{align}
\vec{C} = \myvec{0 \\ 3-4\sqrt{3}}
\end{align}
\end{frame}

// code.c
#include <stdio.h>
#include <math.h>

void find_third_vertex(double *x, double *y) {
    // Given two vertices
    double Ax = -4, Ay = 3;
    double Bx = 4,  By = 3;

    // Side length AB
    double AB = sqrt((Bx - Ax)*(Bx - Ax) + (By - Ay)*(By - Ay));

    // Midpoint M
    double Mx = (Ax + Bx) / 2.0;
    double My = (Ay + By) / 2.0;

    // Height for equilateral triangle
    double h = (sqrt(3) / 2.0) * AB;
 \end{lstlisting}
\end{frame}
