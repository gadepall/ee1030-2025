\begin{frame}
\frametitle{Question}
Passing through the points $\brak{3,4,-7}$ and $\brak{1,-1,6}$.
\end{frame}

\begin{frame}
\frametitle{Solution}
let Passing through the points $\vec{A} =\myvec{3 \\4 \\ -7}$ and $\vec{B} =\myvec{1 \\ -1 \\ 6}$. \\
\myvec{3 & 4 & -7 \\ 1 & -1 & 6}$\mathbf{n}$ =\myvec{1 \\ 1} \\
Writing the augmented matrix:
\begin{align}
\myvec{3 & 4 & -7 & 1 \\ 1 & -1 & 6 & 1} \xleftrightarrow{R_1 \leftarrow R_1 - 3R_2} \myvec{0 & 7 & -25 & -2 \\ 1 & -1 & 6 & 1}\\
\xleftrightarrow{R_1 \leftrightarrow R_2}\myvec{1 & -1 & 6 & 1 \\ 0 & 7 & -25 & -2}
\end{align}
\begin{align}
n_1 - n_2 + 6 n_3 = 1 \\
7 n_2 - 25 n_3 = -2 
\end{align}
From third equation: 
\begin{align}
n_2 = \frac{-2 + 25 n_3}{7}
\end{align}
\end{frame}
\begin{frame}
\frametitle{Solution}
Substitute in second equation: 
\begin{align}
n_1 = \frac{5 - 17 n_3}{7} 
\end{align}
Let $n_3 = t$ , then
\begin{align}
\mathbf{n} = \myvec{\frac{5 - 17 t}{7} \\ \frac{-2 + 25 t}{7} \\ t}
\end{align}
let $t = 1$, multiply with $7$ to clear denominators:
\begin{align}
\mathbf{n} = \myvec{-12 \\ 23 \\ 7}
\end{align}

Thus, the equation of the line is:
$\myvec{-12 & 23 & 7}\mathbf{x}$ = 7
\end{frame}
