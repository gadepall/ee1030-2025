\textbf{Question}:passing through the points $\brak{3,4,-7}$ and $\brak{1,-1,6}$.

\textbf{solution}:let Passing through the points
\begin{align}
\vec{A} =\myvec{3 \\4 \\ -7} ,\vec{B} =\myvec{1 \\ -1 \\ 6} 
\end{align}
The direction vector of line passing through $\vec{A}$ and $\vec{B}$ is
\begin{align}
\vec{B} -\vec{A} &=\myvec{1-3 \\ -1-4 \\  6-(-7)} \\
    &= \myvec{-2 \\ -5 \\ 13}
\end{align}
The equation of line is 
\begin{align}
    r\brak{t} = \myvec{3 \\4 \\ -7} + t\myvec{-2 \\ -5 \\ 13}
\end{align}
