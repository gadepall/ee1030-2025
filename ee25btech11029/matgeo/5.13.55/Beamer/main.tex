\begin{frame}{Question:}


\noindent $\vec{A}=\myvec{1 & 0 & 0\\0 & 1 & 1\\0 & -2 & 4}$ and $\vec{I}=\myvec{1 & 0 & 0\\0 & 1 & 0\\0 & 0 &1}$ and $\vec{A}^{-1} = \frac{1}{6}(\vec{A}^2 + c\vec{A} + d\vec{I})$, then value of c and d are
\begin{enumerate}
\begin{multicols}{4}
    \item \brak{-6,-11}
    \item \brak{6,11}
    \item \brak{-6,11}
    \item \brak{6,-11}
\end{multicols}
\end{enumerate}
\end{frame}

\begin{frame}{Theoretical Solution}
\noindent 
 Given the matrices
\begin{align}
A = \myvec{ 1 & 0 & 0 \\ 0 & 1 & 1 \\ 0 & -2 & 4 }
\quad \text{and} \quad
I = \myvec{ 1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1 }
\end{align}
and the relation $\vec{A}^{-1} = \frac{1}{6}(\vec{A}^2 + c\vec{A} + d\vec{I})$

First we Isolate the Unknowns\\
Starting with the given relation:
\begin{align}
\vec{A}^{-1} = \frac{1}{6}\brak{\Vec{A}^2 + c\vec{A} + d\vec{I}}
\end{align}
Multiply both sides by 6 and rearrange to isolate the terms with the unknown scalars.
\begin{align}
6\vec{A}^{-1} = \vec{A}^2 + c\vec{A} + d\vec{I}\\
c\vec{A} + d\vec{I} = 6\vec{A}^{-1} - \vec{A}^2 
\end{align}
\end{frame}
\begin{frame}{Theoretical Solution}
We find the inverse using Gauss-Jordan elimination on the augmented matrix $\augvec{1}{1}{\vec{A}&\vec{I}}$.
\begin{align}
\augvec{3}{3}{
1 & 0 & 0 & 1 & 0 & 0 \\
0 & 1 & 1 & 0 & 1 & 0 \\
0 & -2 & 4 & 0 & 0 & 1
}
\xrightarrow{R_3 \to R_3 + 2R_2}
\augvec{3}{3}{
1 & 0 & 0 & 1 & 0 & 0 \\
0 & 1 & 1 & 0 & 1 & 0 \\
0 & 0 & 6 & 0 & 2 & 1
}
\xrightarrow{R_2 \to R_2 - \frac{1}{6}R_3}
\augvec{3}{3}{
1 & 0 & 0 & 1 & 0 & 0 \\
0 & 1 & 0 & 0 & 2/3 & -1/6 \\
0 & 0 & 6 & 0 & 2 & 1
}
\end{align}
After scaling the last row, we get $\augvec{1}{1}{\vec{A}&\vec{I}}$. The resulting inverse matrix is:
\begin{align}
\vec{A}^{-1} = \myvec{ 1 & 0 & 0 \\ 0 & 2/3 & -1/6 \\ 0 & 1/3 & 1/6 }
\end{align}

Now we Calculate the Square Matrix, $\vec{A}^2$
We compute $\vec{A}^2$ by matrix multiplication.

\end{frame}
\begin{frame}{Theoretical Solution}
\begin{align}
\vec{A}^2 = \vec{A.A} = \myvec{ 1 & 0 & 0 \\ 0 & 1 & 1 \\ 0 & -2 & 4 } \myvec{ 1 & 0 & 0 \\ 0 & 1 & 1 \\ 0 & -2 & 4 } = \myvec{ 1 & 0 & 0 \\ 0 & -1 & 5 \\ 0 & -10 & 14 }
\end{align}
Substituting the calculated matrices into Equation \brak{2}:
\begin{align*}
6\vec{A}^{-1} - \vec{A}^2 &= 6\myvec{ 1 & 0 & 0 \\ 0 & 2/3 & -1/6 \\ 0 & 1/3 & 1/6 } - \myvec{ 1 & 0 & 0 \\ 0 & -1 & 5 \\ 0 & -10 & 14 } \\
&= \myvec{ 6 & 0 & 0 \\ 0 & 4 & -1 \\ 0 & 2 & 1 } - \myvec{ 1 & 0 & 0 \\ 0 & -1 & 5 \\ 0 & -10 & 14 } \\
&= \myvec{ 5 & 0 & 0 \\ 0 & 5 & -6 \\ 0 & 12 & -13 }
\end{align*}
    
\end{frame}
\begin{frame}{Theoretical Solution}
Now, set the left side of Equation \brak{2} equal to this result:
\begin{align}
c \myvec{ 1 & 0 & 0 \\ 0 & 1 & 1 \\ 0 & -2 & 4 } + d \myvec{ 1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1 } = \myvec{ 5 & 0 & 0 \\ 0 & 5 & -6 \\ 0 & 12 & -13 }\\
\myvec{ c+d & 0 & 0 \\ 0 & c+d & c \\ 0 & -2c & 4c+d } = \myvec{ 5 & 0 & 0 \\ 0 & 5 & -6 \\ 0 & 12 & -13 }
\end{align}
By equating corresponding elements, we can find the values of $c$ and $d$.

 From element (2, 3): $c = -6$\\ From element (2, 2): $c + d = 5 \implies -6 + d = 5 \implies d = 11$\\
The solution is therefore third option$(c, d) = (-6, 11)$.
\end{frame}

void multiplyMatrices(double mat1[N][N], double mat2[N][N], double result[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            result[i][j] = 0;
            for (int k = 0; k < N; k++) {
                result[i][j] += mat1[i][k] * mat2[k][j];
            }
        }
    }
}

bool invertMatrix(double mat[N][N], double inverse[N][N]) {
    double augvec[N][N * 2];

    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            augvec[i][j] = mat[i][j];
        }
        for (int j = 0; j < N; j++) {
            augvec[i][j + N] = (i == j) ? 1.0 : 0.0;
        }
    }

    void scalarMultiply(double scalar, double mat[N][N], double result[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            result[i][j] = mat[i][j] * scalar;
        }
    }
}

void subtractMatrices(double mat1[N][N], double mat2[N][N], double result[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            result[i][j] = mat1[i][j] - mat2[i][j];
        }
    }
}
