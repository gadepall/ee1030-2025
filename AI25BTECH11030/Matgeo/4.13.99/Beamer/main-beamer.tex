\section{Question}
\begin{frame}{Question}
\textbf{Question}:
Two planes are given:
\begin{align}
P_{1}:\;10x + 15y + 12z - 60 &= 0,\\
P_{2}:-2x + 5y + 4z - 20 &= 0.
\end{align}
Which of the following straight lines can be an edge of some tetrahedron whose two faces lie on $P_{1}$ and $P_{2}$?
\begin{enumerate}[label=(\alph*)]
  \item $\displaystyle \frac{x-1}{0} = \frac{y-1}{-5} = \frac{z-1}{4}$
  \item $\displaystyle \frac{x-6}{-5} = \frac{y-2}{3} = \frac{z}{-5}$
  \item $\displaystyle \frac{x}{-2} = \frac{y-4}{5} = \frac{z}{4}$
  \item $\displaystyle \frac{x}{1} = \frac{y-4}{-2} = \frac{z}{3}$
\end{enumerate}
\end{frame}

\section{Solution}
\begin{frame}[fragile]
    \frametitle{Solution}
The two planes are:
\begin{align}
P_{1}:&\;\vec{n}_{1}^\top \vec{r} = d_{1}, 
\quad \vec{n}_{1} = \myvec{10 \\ 15 \\ 12},\; d_{1} = 60,\\
P_{2}:&\;\vec{n}_{2}^\top \vec{r} = d_{2}, 
\quad \vec{n}_{2} = \myvec{-2 \\ 5 \\ 4},\; d_{2} = 20.
\end{align}
A general line is
\begin{align}
\vec{r}(t) = \vec{r}_{0} + t\,\vec{d}.
\end{align}

\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
 For a line to be an edge of a tetrahedron whose two faces lie on $P_1$ and $P_2$, it must satisfy one of the conditions:
\begin{align}
t_1 &= \frac{d_1 - \vec{n}_1^\top\vec{r}_0}{\vec{n}_1^\top\vec{d}},&
t_2 &= \frac{d_2 - \vec{n}_2^\top\vec{r}_0}{\vec{n}_2^\top\vec{d}},
\end{align}
and then either\\
  1. $\vec{n}_1^\top\vec{r}_0=d_1$ and $\vec{n}_1^\top\vec{d}=0$ i.e. the line lies on plane $P_{1}$,or \\
  2.  $\vec{n}_2^\top\vec{r}_0=d_2$ and $\vec{n}_2^\top\vec{d}=0$ i.e. the line lies on plane $P_{2}$,or \\
  3.  $t_1\neq t_2$ i.e. the line intersects both planes on two different points.

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
We test the options:\\
  a) $\displaystyle \vec{r}_0=\myvec{1\\1\\1},\,\vec{d}=\myvec{0\\-5\\4}$. 
    \(\vec{n}_1^\top\vec{r}_0=37,\;\vec{n}_2^\top\vec{r}_0=7,\;
    \vec{n}_1^\top\vec{d}=-27,\;\vec{n}_2^\top\vec{d}=-9,\)
    \[
      t_1=\tfrac{60-37}{-27}=-\tfrac{23}{27},\;
      t_2=\tfrac{20-7}{-9}=-\tfrac{13}{9},\quad t_1\neq t_2\ (\text{valid}).
    \]
  b) $\displaystyle \vec{r}_0=\myvec{6\\2\\0},\,\vec{d}=\myvec{-5\\3\\-5}$. 
    \(\vec{n}_1^\top\vec{r}_0=90,\;\vec{n}_2^\top\vec{r}_0=-2,\;
    \vec{n}_1^\top\vec{d}=-65,\;\vec{n}_2^\top\vec{d}=5,\)
    \[
      t_1=\tfrac{60-90}{-65}=\tfrac{6}{13},\;
      t_2=\tfrac{20-(-2)}{5}=\tfrac{22}{5},\quad t_1\neq t_2\ (\text{valid}).
    \]
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
    c) $\displaystyle \vec{r}_0=\myvec{0\\4\\0},\,\vec{d}=\myvec{-2\\5\\4}$. 
    \(\vec{n}_1^\top\vec{r}_0=60,\;\vec{n}_2^\top\vec{r}_0=20,\;
    \vec{n}_1^\top\vec{d}=103,\;\vec{n}_2^\top\vec{d}=45,\)
    \[
      t_1=\tfrac{60-60}{103}=0,\;
      t_2=\tfrac{20-20}{45}=0,\quad t_1=t_2\ (\text{invalid}).
    \]
  d)  $\displaystyle \vec{r}_0=\myvec{0\\4\\0},\,\vec{d}=\myvec{1\\-2\\3}$. 
    \(\vec{n}_2^\top\vec{r}_0=20,\;\vec{n}_2^\top\vec{d}=0,\)
    \[
      \myvec{-2 \ 5 \ 4 }\vec{r}=20\quad\forall\,t,
      \quad\text{(the line lies on }P_2\text{) valid}.
    \]
Therefore, the valid edges are \textbf{(A), (B), and (D)}.

\end{frame}

    double norm = sqrt(v[0]*v[0] + v[1]*v[1] + v[2]*v[2]);

static int check_line_on_plane(double r0[3], double d[3], double n[3], double d_plane, double tol) {
    double n_d = 0.0, n_r0 = 0.0;
    for (int i=0; i<3; i++) {
        n_d += n[i] * d[i];
        n_r0 += n[i] * r0[i];
    }
    return (fabs(n_d) < tol && fabs(n_r0 - d_plane) < tol);

void prepare_line_points(double r0[3], double d[3], double t_start, double t_end, int num_points, double *output) {
    double step = (t_end - t_start) / (num_points - 1);
    for (int i=0; i < num_points; i++) {
        double t = t_start + i * step;
        for (int j=0; j<3; j++) {
            output[i*3 + j] = r0[j] + t * d[j];
        }
    }
}
