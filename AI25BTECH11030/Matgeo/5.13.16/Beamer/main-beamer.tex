\section{Question}
\begin{frame}{Question}
\textbf{Question}:
Let \( k \) be an integer such that the triangle with vertices 
\[
\vec{A} = \myvec{ k \\ -3k }, \quad 
\vec{B} = \myvec{ 5 \\ k }, \quad 
\vec{C} = \myvec{ -k \\ 2 }
\]
has area 28 sq. units. Find the orthocenter of the triangle.

\end{frame}

\section{Solution}
\begin{frame}[fragile]
    \frametitle{Solution}
\begin{align}
\text{Area} &= \frac{1}{2} \left| \vec{M} \right| = 28
\end{align}

\begin{align}
\text{where} \quad
\vec{M} &= \myvec{
\vec{B} - \vec{A} & \vec{C} - \vec{A}
}
= \myvec{
5-k & -2k \\
4k & 2 + 3k
}
\end{align}

\begin{align}
\det(\vec{M}) &= (5-k)(2+3k) - (-2k)(4k) = 10 + 13k + 5k^2
\end{align}

\begin{align}
10 + 13k + 5k^2 &= \pm 56
\end{align}
After solving,
\begin{align}
k &= 2
\end{align}

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
\begin{align}
\text{For } k=2, \quad \vec{A} = \myvec{2 \\ -6}, \quad \vec{B} = \myvec{5 \\ 2}, \quad \vec{C} = \myvec{-2 \\ 2}
\end{align}

The equation of the altitude through \( \vec{A} \) is:
\begin{align}
\vec{n}^T \vec{x} = 1
\end{align}

This altitude passes through \( \vec{A} \), so
\begin{align}
\vec{n}^T \vec{A} = 1
\end{align}
\begin{align}
\vec{n}^\top \myvec{2 \\ -6} = 1
\end{align}

and the normal vector \( \vec{n} \) of the altitude must be parallel to \( \vec{B} - \vec{C} \):
\begin{align}
\vec{n} = t \myvec{7 \\ 0}
\end{align}
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
Substitute to get:
\begin{align}
t \myvec{7 \\ 0}^\top \myvec{2 \\ -6} = 1 \implies 14 t = 1 \implies t = \frac{1}{14}
\end{align}

Hence,
\begin{align}
\vec{n} = \myvec{\frac{1}{2} \\ 0}
\end{align}

The equation of altitude through \( \vec{A} \) is:
\begin{align}
\myvec{\frac{1}{2} & 0} \vec{x} = 1
\end{align}

\vspace{1em}

The equation of the altitude through \( \vec{B} \) is:
\begin{align}
\vec{n}^T \vec{x} = 1
\end{align}
This altitude passes through \( \vec{B} \), so
\begin{align}
\vec{n}^T \vec{B} = 1
\end{align}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
\begin{align}
\vec{n}^\top \myvec{5 \\ 2} = 1
\end{align}

and the normal vector \( \vec{n} \) must be parallel to \( \vec{A} - \vec{C} \):
\begin{align}
\vec{n} = t \myvec{4 \\ -8}
\end{align}

Substitute to get:
\begin{align}
t \myvec{4 \\ -8}^\top \myvec{5 \\ 2} = 1 \implies t (20 - 16) = 1 \implies 4 t = 1 \implies t = \frac{1}{4}
\end{align}

Hence,
\begin{align}
\vec{n} = \myvec{1 \\ -2}
\end{align}
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
The equation of altitude through \( \vec{B} \) is:
\begin{align}
\myvec{1 & -2} \vec{x} = 1
\end{align}

Combining equations (2.13) and (2.20):
\begin{align}
\myvec{\frac{1}{2} & 0 \\ 1 & -2} \vec{x} = \myvec{1 \\ 1}
\end{align}

Writing the augmented matrix:
\begin{align}
\myaugvec{2}{
\frac{1}{2} & 0 & 1 \\
1 & -2  & 1
}
\end{align}

Perform the row operation \( R_2 \to R_2 - 2 R_1 \):
\begin{align}
\myaugvec{2}{
\frac{1}{2} & 0 & 1 \\
0 & -2 & -1
}
\end{align}

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
Therefore, the solution vector is:
\begin{align}
\vec{x} = \myvec{2 \\ \frac{1}{2}}
\end{align}

\textbf{Answer:} The orthocentre of the triangle is
\[
\boxed{
\vec{x} = \myvec{2 \\ \frac{1}{2}}
}
\]
\end{frame}

#include "matfun.h"

void foot_of_perpendicular(const double *P, const double *Q, const double *R, double *foot) {
    double QR[2] = {R[0]-Q[0], R[1]-Q[1]};
    double QP[2] = {P[0]-Q[0], P[1]-Q[1]};
    double dot_QP_QR = QP[0]*QR[0] + QP[1]*QR[1];
    double dot_QR_QR = QR[0]*QR[0] + QR[1]*QR[1];
    double t = dot_QP_QR / dot_QR_QR;
    foot[0] = Q[0] + t * QR[0];
    foot[1] = Q[1] + t * QR[1];
}

void orthocenter(const double *A, const double *B, const double *C, double *O) {
    double footA[2], footB[2];
    foot_of_perpendicular(A, B, C, footA);
    foot_of_perpendicular(B, A, C, footB);

\end{lstlisting}

        return;

import numpy as np
import matplotlib.pyplot as plt

A = np.array([2, -6])
B = np.array([5, 2])
C = np.array([-2, 2])

def foot_of_perpendicular(P, Q, R):
    """
    Calculate foot of perpendicular from point P to line QR.
    
    Returns the point (x,y) on line QR closest to P.
    """
    QR = R - Q
    QP = P - Q
    t = np.dot(QP, QR) / np.dot(QR, QR)
    foot = Q + t * QR
    return foot
\end{lstlisting}

# Compute orthocenter as intersection of two altitudes
# Use line parameter form: for line segment vertex-foot
def line_params(P, Q):
    # Line vector:
    v = Q - P
    # Line coefficients: a*x + b*y = c form
    a = v[1]
    b = -v[0]
    c = a*P[0] + b*P[1]
    return a, b, c
