\textbf{Question}:Let \( k \) be an integer such that the triangle with vertices 
\[
\vec{A} = \myvec{ k \\ -3k }, \quad 
\vec{B} = \myvec{ 5 \\ k }, \quad 
\vec{C} = \myvec{ -k \\ 2 }
\]
has area 28 sq. units. Find the orthocenter of the triangle.


\textbf{Solution}: \\
\begin{align}
\text{Area} &= \frac{1}{2} \left| \vec{M} \right| = 28
\end{align}

\begin{align}
\text{where} \quad
\vec{M} &= \myvec{
\vec{B} - \vec{A} & \vec{C} - \vec{A}
}
= \myvec{
5-k & -2k \\
4k & 2 + 3k
}
\end{align}

\begin{align}
\det(\vec{M}) &= (5-k)(2+3k) - (-2k)(4k) = 10 + 13k + 5k^2
\end{align}

\begin{align}
10 + 13k + 5k^2 &= \pm 56
\end{align}
After solving,
\begin{align}
k &= 2
\end{align}
\begin{align}
\text{For } k=2, \quad \vec{A} &= \myvec{2 \\ -6}, \quad \vec{B} = \myvec{5 \\ 2}, \quad \vec{C} = \myvec{-2 \\ 2}
\end{align}
The equation of the altitude through \( \vec{A} \) is:
\begin{align}
\vec{x}^T \vec{n} = 1
\end{align}

This altitude passes through \( \vec{A} \), so
\begin{align}
\vec{A}^T \vec{n} = 1
\end{align}
\begin{align}
\myvec{2 & -6} \vec{n} = 1
\end{align}
and is perpendicular to \( \vec{B}-\vec{C} \), so
\begin{align}
(\vec{B} - \vec{C})^T \vec{n} = 0
\end{align}
\begin{align}
\myvec{7 & 0} \vec{n} = 0
\end{align}
The augmented matrix is:
\begin{align}
\myaugvec{2}{
7 & 0 & 0 \\
2 & -6 & 1
}
\end{align}

Row reduction steps:

First, \( R_2 \to R_2 - \frac{2}{7} R_1 \):
\begin{align}
\myaugvec{2}{
7 & 0 & 0 \\
0 & -6 & 1
}
\end{align}

\begin{align}
\vec{n} = \myvec{0 \\ -\frac{1}{6}}
\end{align}


The equation of the altitude through \( \vec{B} \) is:
\begin{align}
\vec{x}^T \vec{n} = 1
\end{align}
This altitude passes through \( \vec{B} \), so
\begin{align}
\vec{B}^T \vec{n} = 1
\end{align}
\begin{align}
\myvec{5 & 2} \vec{n} = 1
\end{align}
and is perpendicular to \( \vec{A}-\vec{C} \), so
\begin{align}
(\vec{A} - \vec{C})^T \vec{n} = 0
\end{align}
\begin{align}
\myvec{4 & -8} \vec{n} = 0
\end{align}
The augmented matrix is:
\begin{align}
\myaugvec{2}{
5 & 2 & 1 \\
4 & -8 & 0
}
\end{align}
Row reduction steps:
First, \( R_2 \to R_2 - \frac{4}{5} R_1 \):
\begin{align}
\myaugvec{2}{
5 & 2 & 1 \\
0 & -\frac{48}{5} & -\frac{4}{5}
}
\end{align}
Next, \( R_2 \to -\frac{5}{48} R_2 \):
\begin{align}
\myaugvec{2}{
5 & 2 & 1 \\
0 & 1 & \frac{1}{12}
}
\end{align}
Back substitute: \( R_1 \to R_1 - 2 R_2 \)
\begin{align}
\myaugvec{2}{
5 & 0 & \frac{5}{6} \\
0 & 1 & \frac{1}{12}
}
\end{align}
\begin{align}
\vec{n} = \myvec{\frac{1}{6} \\ \frac{1}{12}}
\end{align}

First altitude:
\begin{align}
\myvec{0 & -\frac{1}{6}} \vec{x} = 1
\end{align}

Second altitude:
\begin{align}
\myvec{\frac{1}{6} & \frac{1}{12}} \vec{x} = 1
\end{align}

Combining both equations:
\begin{align}
\myvec{0 & -\frac{1}{6} \\ \frac{1}{6} & \frac{1}{12}} \vec{x} = \myvec{1 \\ 1}
\end{align}

Writing the augmented matrix:
\begin{align}
\myaugvec{2}{
\frac{1}{6} & \frac{1}{12} & 1 \\
0 & -\frac{1}{6}  & 1
}
\end{align}

Perform the row operation \( R_1 \to R_1 + \frac{1}{2} R_2 \):
\begin{align}
\myaugvec{2}{
\frac{1}{6} & 0 & \frac{3}{2} \\
0 & -\frac{1}{6} & 1
}
\end{align}

Therefore, the solution vector is:
\begin{align}
\vec{x} = \myvec{9 \\ -6}
\end{align}
\textbf{Answer:} The orthocentre of the triangle is
\[
\boxed{
\vec{x}=\myvec{9 \\ -6}
}
\]
