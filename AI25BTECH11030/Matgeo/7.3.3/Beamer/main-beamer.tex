\section{Question}
\begin{frame}{Question}
\textbf{Question}:
The equation of a circle with origin as centre and passing through the vertices of an equilateral triangle whose median is of length $3a$ is
\begin{enumerate}
\item $x^2 + y^2 = 9a^2$
\item $x^2 + y^2 = 16a^2$
\item $x^2 + y^2 = 4a^2$
\item $x^2 + y^2 = a^2$
\end{enumerate}



\end{frame}

\section{Solution}
\begin{frame}[fragile]
    \frametitle{Solution}
Let the vertices of the equilateral triangle be $\vec{A}$, $\vec{B}$, and $\vec{C}$.And the radius of circle be r.

Since the circle is centered at the origin, and the radius is r:
\begin{align}
\vec{u} = -\myvec{0 \\ 0} = \myvec{0 \\ 0}
\end{align}
\begin{align}
f=-r^2
\end{align}

Since the circle passes through $\vec{A}$:
\begin{align}
\norm{\vec{x}^\top\vec{x}} + 2\vec{u}^T\vec{x} + f &= 0\\
\norm{\vec{x}^\top\vec{x}}+ 2\myvec{0 & 0}\vec{x} + f &= 0\\
\norm{\vec{x}^\top\vec{x}} - r^2 &= 0\\
\end{align}

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
For an equilateral triangle inscribed in a circle, let us place the vertices as:
\begin{align}
\vec{A} &= \myvec{r \\ 0}\\
\vec{B} &= \myvec{r\cos\theta \\ r\sin\theta}\\
\vec{C} &= \myvec{r\cos\theta \\ -r\sin\theta}
\end{align}

The median from vertex $\vec{A}$ goes to the midpoint of $\vec{B}$ and $\vec{C}$:
\begin{align}
\text{Midpoint} &= \frac{\vec{B} + \vec{C}}{2} = \myvec{r\cos\theta \\ 0}
\end{align}

The median vector is:
\begin{align}
\vec{m} &= \myvec{r\cos\theta \\ 0} - \myvec{r \\ 0} = \myvec{r(\cos\theta - 1) \\ 0}
\end{align}

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
Given that the median length is $3a$:
\begin{align}
\norm{\vec{m}^\top\vec{m}} &= (3a)^2 = 9a^2\\
r^2(\cos\theta - 1)^2 &= 9a^2
\end{align}

For an equilateral triangle, $\cos\theta = \cos\left(\frac{2\pi}{3}\right) = -\frac{1}{2}$:
\begin{align}
(\cos\theta - 1) &= -\frac{3}{2}\\
r^2  \frac{9}{4} &= 9a^2\\
r^2 &= 4a^2
\end{align}



Therefore, the equation of the circle is:
\begin{align}
\norm{\vec{x}^\top\vec{x}} + 2\myvec{0 & 0}\vec{x} - 4a^2 &= 0\\
\norm{\vec{x}^\top\vec{x}} - 4a^2 &= 0
\end{align}
\end{frame}

#include <math.h>
#include "matfun.h"

#define MEDIAN_COEFF 3.0

double compute_cos_theta(void) {
    return cos(2.0 * M_PI / 3.0);
}

double compute_coefficient_sq(void) {
    double cos_theta = compute_cos_theta();
    double diff = cos_theta - 1.0;
    return diff * diff;
}
\end{lstlisting}

double compute_r_squared_coeff(void) {
    double num = MEDIAN_COEFF * MEDIAN_COEFF;            
    double denom = compute_coefficient_sq();             
    return num / denom;                                  
}

import numpy as np
import matplotlib.pyplot as plt

# Given data
median_length_coefficient = 3  # coefficient of 'a' in median length

# Matrix approach to find radius
theta = 2 * np.pi / 3
cos_theta = np.cos(theta)
cos_theta_minus_1 = cos_theta - 1  # -3/2
coefficient_squared = cos_theta_minus_1**2  # 9/4

# Solve for r^2 coefficient: r^2 * (9/4) = 9a^2  =>  r^2 = 4a^2
r_squared_coefficient = (median_length_coefficient**2) / coefficient_squared  # 4
r = 2  # since a = 1 for plotting
