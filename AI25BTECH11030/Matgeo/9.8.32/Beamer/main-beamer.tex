\section{Question}
\begin{frame}{Question}
\textbf{Question}:
The centre of the circle passing through the point $(0,1)$ and touching the curve $y = x^2$ at $(2,4)$ is

\begin{enumerate}[label=(\alph*)]
\item $\left(\frac{-16}{5},\; \frac{27}{10}\right)$
\item $\left(\frac{-16}{7},\; \frac{53}{10}\right)$
\item $\left(\frac{-16}{5},\; \frac{53}{10}\right)$
\item none of these
\end{enumerate}

\end{frame}

\section{Solution}
\begin{frame}[fragile]
    \frametitle{Solution}
\textbf{Input Variables:  }\\
Parameters of the Circle:\\
Where $\vec{c}$ is the centre of circle.
\begin{equation}
\vec{V}_1 = \myvec{1 & 0 \\ 0 & 1}, \, \vec{u}_1 = -\vec{c}
\end{equation}

Parameters of the Parabola:
\begin{equation}
\vec{V}_2 = \myvec{1 & 0 \\ 0 & 0}, \, \vec{u}_2 = \myvec{0 \\ -\frac{1}{2}}, \, f_2 = 0
\end{equation}

\begin{equation}
\vec{P} = \myvec{0 \\ 1}, \, \vec{Q}= \myvec{2 \\ 4}
\end{equation}
\end{frame}


\begin{frame}[fragile]
    \frametitle{Solution}
\textbf{Step 1:} Find one equation for $\vec{c}$\\
Points P and Q lie on the circle so,
\begin{equation}
\vec{P}^\top \vec{V}_1 \vec{P} - 2 \vec{c}^\top \vec{P} + f_1 = 0,
\end{equation}

\begin{equation}
\vec{Q}^\top \vec{V}_1 \vec{Q} - 2 \vec{c}^\top \vec{Q} + f_1 = 0,
\end{equation}

Substracting (0.4)-(0.5),

\begin{equation}
(\vec{P}^\top \vec{V}_1 \vec{P} - \vec{Q}^\top \vec{V}_1 \vec{Q}) 
- 2 \vec{c}^\top (\vec{P} - \vec{Q}) = 0,
\end{equation}

\begin{equation}
2 \vec{c}^\top (\vec{P} - \vec{Q}) = \vec{P}^\top \vec{V}_1 \vec{P} - \vec{Q}^\top \vec{V}_1 \vec{Q}.
\end{equation}

\begin{equation}
-2 \vec{c}^\top \myvec{2 \\ 3} = 
\myvec{0 & 1} 
\myvec{1 & 0 \\ 0 & 1} 
\myvec{0 \\ 1} - 
\myvec{2 & 4} 
\myvec{1 & 0 \\ 0 & 1} 
\myvec{2 \\ 4}.
\end{equation}

\begin{equation}
\myvec{4 \ 6}\vec{c} = 19
\end{equation} 
\end{frame}


\begin{frame}[fragile]
    \frametitle{Solution}

\textbf{Step 2:} Find the equation of tangent through (2,4) at Parabola
\begin{equation}
(\vec{V}_2 \vec{Q} + \vec{u}_2)^\top \vec{x} + \vec{u}_2^\top \vec{Q} + f_2 = 0,
\end{equation}

\begin{equation}
\biggl(
\myvec{1 & 0 \\ 0 & 0}
\myvec{2 \\ 4}
+
\myvec{0 \\ -\tfrac{1}{2}}
\biggr)^\top \vec{x}
+
\myvec{0 \\ -\tfrac{1}{2}}^\top
\myvec{2 \\ 4}
+ 0 = 0,
\end{equation}

\begin{equation}
\myvec{2 \ -\tfrac{1}{2}} \vec{x} = 2.
\end{equation}

\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
\textbf{Step 3:} Find the equation of normal to circle at (2,4)\\
Let the equation of normal to circle at (2,4) be:
\begin{equation}
\vec{n}^\top\vec{x} = 1
\end{equation}

The normal is perpendicular to the previously calculated tangent in (0.12)
\begin{equation}
\vec{n}^\top \myvec{2 \\ -\frac{1}{2}} = 0
\end{equation}
\quad It passes through (2,4)
\begin{equation}
\vec{n}^\top \myvec{2 \\ 4} = 1
\end{equation}

Row Reduction Steps:
\begin{equation}
\myaugvec{2}{
2 & -\frac{1}{2} & 0 \\
2 & 4 & 1
}
\end{equation}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Solution}
Perform the row operation \( R_2 \to R_2 - R_1\):
\begin{equation}
\myaugvec{2}{
2 & -\frac{1}{2} & 0 \\
0 & \frac{9}{2} & 1
}
\end{equation}
Perform the row operation \( R_1 \to R_1 + \frac{R_2}{9}\):
\begin{equation}
\myaugvec{2}{
2 & 0 & \frac{1}{9} \\
0 & \frac{9}{2} & 1
}
\end{equation}
\begin{equation}
\vec{n}=\frac{1}{18}\myvec{1 \\ 4}
\end{equation}

Equation of normal is :
\begin{equation}
\myvec{1 \ 4}\vec{x} = 18
\end{equation}
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
\textbf{Step 4:} Find $\vec{c}$\\

Centre lies on the normal (2.20)
\begin{equation}
\myvec{1 & 4}\vec{c} = 18
\end{equation}
from equation (2.9)
\begin{equation}
\myvec{4 & 6}\vec{c} = 19
\end{equation}
Row reduction steps:
\begin{equation}
\myaugvec{2}{
1 & 4 & 18 \\
4 & 6 & 19
}
\end{equation}
Perform the row operation \( R_2 \to R_2 - 4R_2\):
\begin{equation}
\myaugvec{2}{
1 & 4 & 18 \\
0 & -10 & -53
}
\end{equation}
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
Perform the row operation \( R_1 \to R_1 +\frac{2}{5}R_2\):
\begin{equation}
\myaugvec{2}{
1 & 0 & -\frac{16}{5} \\
0 & -10 & -53
}
\end{equation}

\begin{equation}
\vec{c} = \myvec{ -\frac{16}{5} \\ \frac{53}{10} }
\end{equation}

\textbf{Final Answer:}\\
The centre of the circle is $\myvec{ -\frac{16}{5} \\ \frac{53}{10} }$.

\end{frame}

// Solve 2x2 system Ax = b using simple method
int solve_2x2(double A[4], double b[2], double x[2]) {
    // A = [a11, a12, a21, a22]  (row-major)
    double det = A[0]*A[3] - A[1]*A[2];
    
    if (fabs(det) < 1e-10) return 0;  // No solution
    
    x[0] = (b[0]*A[3] - b[1]*A[1]) / det;
    x[1] = (A[0]*b[1] - A[2]*b[0]) / det;
    
    return 1;  // Success
}

// Solve the circle-parabola problem
int solve_problem(double center[2], double *radius) {
    // System: 4cx + 6cy = 19, cx + 4cy = 18
    double A[4] = {4.0, 6.0, 1.0, 4.0};
    double b[2] = {19.0, 18.0};
    
    if (!solve_2x2(A, b, center)) return 0;
    
    // Calculate radius from center to (0,1)
    *radius = distance(center[0], center[1], 0.0, 1.0);
    
    return 1;
}

import numpy as np
import matplotlib.pyplot as plt

def solve_circle_parabola_problem():
    """Find the center of the circle passing through (0,1) and touching y = x² at (2,4)"""
    
    # Given points
    P = np.array([0, 1])
    Q = np.array([2, 4])
    
    # Set up system of equations
    # Constraint 1: Equal distances from center to P and Q
    # 4cx + 6cy = 19
    # Constraint 2: Center-to-Q line is normal to parabola at Q  
    # cx + 4cy = 18

\end{lstlisting}

\begin{lstlisting}[language=Python]
    A = np.array([[4, 6], [1, 4]])
    b = np.array([19, 18])
    
    # Solve for center
    center = np.linalg.solve(A, b)
    radius = np.linalg.norm(P - center)
    
    return center, radius

def plot_solution():
    """Plot the parabola, circle, and key points with complete circle visible"""
    center, radius = solve_circle_parabola_problem()
