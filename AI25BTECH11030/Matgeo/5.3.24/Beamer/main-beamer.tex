\section{Question}
\begin{frame}{Question}
\textbf{Question}:
Solve the following pair of linear equations:
\begin{align}
   3x - 5y = 4 
\end{align}
\begin{align}
    2y + 7 = 9x
\end{align}
\end{frame}

\section{Solution}
\begin{frame}[fragile]
    \frametitle{Solution}
The equation of a line:
\begin{align}
\vec{n^T} \vec{x} = c
\end{align}

Line \(L\):
\begin{align}
\myvec{3 & -5} \vec{x} = 4
\end{align}

Line \(K\):
\begin{align}
\myvec{9 & -2} \vec{x} = 7
\end{align}

Combining both the equations:
\begin{align}
\myvec{3 & -5 \\ 9 & -2} \vec{x} = \myvec{4 \\ 7}
\end{align}

Writing the augmented matrix:
\begin{align}
\myaugvec{2}{
3 & -5 & 4 \\
9 & -2 & 7
}
\end{align}

\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
Perform the row operation \( R_2 \to R_2 - 3 R_1 \):
\begin{align}
\myaugvec{2}{
3 & -5 & 4 \\
0 & 13 & -5
}
\end{align}

Perform the row operation \( R_2 \to \frac{1}{13} R_2 \):
\begin{align}
\myaugvec{2}{
3 & -5 & 4 \\
0 & 1 & -\frac{5}{13}
}
\end{align}

Back substitute using \( R_1 \to R_1 + 5 R_2 \):
\begin{align}
\myaugvec{2}{
3 & 0 & \frac{27}{13} \\
0 & 1 & -\frac{5}{13}
}
\end{align}

Therefore, the solution vector is:
\begin{align}
\vec{x} = \myvec{\frac{9}{13} \\ -\frac{5}{13}}
\end{align}
\end{frame}
\begin{frame}[fragile]
    \frametitle{Solution}
\textbf{Final Answer:} Therefore, the solution vector is:
\[
\boxed{
\vec{x} = \myvec{\frac{9}{13} \\ -\frac{5}{13}}
}
\]


\end{frame}

\begin{lstlisting}[language=C]
include "matfun.h"

// Solve 2x2 system using Cramer's rule (appropriate for this fixed size)
void solve_linear_system(double A[2][2], double B[2], double X[2]) {
    double det = A[0][0]*A[1][1] - A[0][1]*A[1][0];
    if (det == 0) {
        // No unique solution
        X[0] = X[1] = 0;
        return;
    }
    X[0] = (B[0]*A[1][1] - B[1]*A[0][1]) / det;
    X[1] = (A[0][0]*B[1] - A[1][0]*B[0]) / det;
}


\end{lstlisting}
\end{frame}
