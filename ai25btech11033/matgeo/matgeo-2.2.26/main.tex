\textbf{Question}

\noindent Find the angle between the pair of lines given by

$\overrightarrow{\vec{r}} = 3\hat{i} + 2\hat{j} - 4\hat{k} +\lambda (\,\hat{i} + 2\hat{j} + 2\hat{k}\,)$

and
$\overrightarrow{\vec{r}}=2\hat{j}+ \mu (\,3\hat{i} + 2\hat{j} + 6\hat{k}\,)$

\textbf{solution}:\\
Let\\
\begin{align}
    \vec{A}=\myvec{1\\2\\2} \vec{B}=\myvec{3\\2\\2}
\end{align} 
The angle btween the lines equals the angle btween a and b


    
\begin{align}
\cos \theta=\frac{a^T b}{\|\vec{a}\|\|\vec{b}\|}
\end{align}

Compute the pieces:
    
\begin{align}
\vec{a}^T\vec{b} = \myvec{1&2&3} \myvec{3\\2\\6}
\end{align}
\begin{align}
\vec{a}^T\vec{b}=1.3+2.2+2.6+=9
\end{align}
Thus,

\begin{align}
\cos \theta = \frac{19}{3.7}=\frac{19}{21}
\end{align}

Then:
\begin{align}
\theta = \cos^{-1}\left( \frac{19}{21} \right)
\end{align}
