\begin{frame}{Question}
Find the equations of the lines parallel to axes and passing through $(2, 3)$

\end{frame}

\begin{frame}{Solution}
The given point is  
\begin{align}
A = \begin{pmatrix}2 \\ 3\end{pmatrix}.
\end{align}


\textbf{Case 1: Line parallel to the $x$-axis}

 A line parallel to the $x$-axis has direction vector  
\begin{align}
\vec{d} = \begin{pmatrix}1 \\ 0\end{pmatrix}.
\end{align}

 A normal vector to this line is therefore  

\begin{align}
\vec{n} = \begin{pmatrix}0 \\ 1\end{pmatrix}.
\end{align}
\end{frame}

\begin{frame}{Solution}
Equation of a line in normal form is  
\begin{align}
\vec{n}^T(\vec{x} - A) = 0.
\end{align}

Substituting,
\begin{align}
\begin{pmatrix}0 & 1\end{pmatrix}
\left(\vec{x} - \begin{pmatrix}2 \\ 3\end{pmatrix}\right) &= 0
\end{align}
\begin{align}
\begin{pmatrix}0 & 1\end{pmatrix}\vec{x} &= 3
\end{align}

\end{frame}

\begin{frame}{Solution}
\textbf{Case 2: Line parallel to the $y$-axis}

 A line parallel to the $y$-axis has direction vector  
\begin{align}
\vec{d} = \begin{pmatrix}0 \\ 1\end{pmatrix}.
\end{align}
 A normal vector to this line is therefore  
\begin{align}
\vec{n} = \begin{pmatrix}1 \\ 0\end{pmatrix}.
\end{align}

Equation of the line:
\begin{align}
\begin{pmatrix}1 & 0\end{pmatrix}
\left(\vec{x} - \begin{pmatrix}2 \\ 3\end{pmatrix}\right) &= 0 \\[6pt]
\begin{pmatrix}1 & 0\end{pmatrix}\vec{x} &= 2
\end{align}
\end{frame}

\begin{frame}{Solution}
\textbf{Case 3: General line through $(2,3)$}

Let the normal vector be  
\begin{align}
\vec{n} = \begin{pmatrix}a \\ b\end{pmatrix}, \quad (a,b)\neq (0,0).
\end{align}

Equation:

\begin{align}
\vec{n}^T(\vec{x} - A) &= 0 \\[6pt]
\begin{pmatrix}a & b\end{pmatrix}
\left(\vec{x} - \begin{pmatrix}2 \\ 3\end{pmatrix}\right) &= 0 \\[6pt]
\begin{pmatrix}a & b\end{pmatrix}\vec{x} &= 2a+3b
\end{align}
\end{frame}
