\begin{frame}{Question}
 Solve the following system of linear equations.
 \begin{align}
 x-3y-3=0\\
 3x-9y-2=0
 \end{align}
\end{frame}

\begin{frame}{Solution}
The given equations of lines are,
\begin{align}
 \myvec{1 & -3}\vec{x} = 3 \\ 
 \myvec{3 & -9}\vec{x} = 2
\end{align}

In matrix form,
\begin{align}
\myvec{1 & -3 \\ 3 & -9}\vec{x} = \myvec{3 \\ 2}
\end{align}
\end{frame}

\begin{frame}{Solution}
So the augmented matrix is
\begin{align}
\augvec{2}{1}{1 & -3 & 3 \\ 3 & -9 & 2}
\end{align}

Performing row operations:
\begin{align}
R_2 \to R_2 - 3R_1 
\implies \augvec{2}{1}{1 & -3 & 3 \\ 0 & 0 & -7}
\end{align}

This corresponds to the equation
\begin{align}
0x + 0y = -7,
\end{align}
which is a contradiction.
\end{frame}

\begin{frame}{Solution}
Hence, the system has \textbf{no solution}. The two equations can be written as
\begin{align}
x - 3y = 3\\
\quad x - 3y = \tfrac{2}{3}
\end{align}
which represent two parallel lines with the same slope but different intercepts. 

Therefore,The given system of equations is \textbf{inconsistent}, and the two lines are parallel.
\end{frame}


\begin{frame}[fragile]{ Code (C)}
\begin{block}{Part 1}
\begin{lstlisting}[language=C, basicstyle=\ttfamily\small, keywordstyle=\color{blue}, frame=single]
#include <stdio.h>

int solve(int a1, int b1, int c1, int a2, int b2, int c2, float* x, float* y) {
    int det = a1 * b2 - a2 * b1;

    if (det == 0) {
        if (a1 * c2 == a2 * c1 && b1 * c2 == b2 * c1)
            return 2; // coincident
        else
            return 1; // parallel
    } else {
        *x = (float)(c1 * b2 - c2 * b1) / det;
        *y = (float)(a1 * c2 - a2 * c1) / det;
        return 0; // unique solution
    }
}
\end{lstlisting}
\end{block}
\end{frame}
