\begin{frame}{Question}
For what value of $k$ $(k>0)$ is the area of the triangle with vertices $(-2,6)$, $(2k,4)$ and $(2k+1,10)$ equal to $52$ sq. units?
\end{frame}

\begin{frame}{Solution}
Vertices of triangle are
\begin{align}
A
\begin{pmatrix}
x_1 \\[4pt] y_1
\end{pmatrix}
=
\begin{pmatrix}
-2 \\[4pt] 6
\end{pmatrix}
\end{align}

\begin{align}
B
\begin{pmatrix}
x_2 \\[4pt] y_2
\end{pmatrix}
=
\begin{pmatrix}
2k \\[4pt] 4
\end{pmatrix}
\end{align}

\begin{align}
C 
\begin{pmatrix}
x_3 \\[4pt] y_3
\end{pmatrix}
=
\begin{pmatrix}
2k+1 \\[4pt] 10
\end{pmatrix}
\end{align}


\end{frame}

\begin{frame}{Solution}
\begin{align}
\text{Area} &= \frac{1}{2}
\left|
\begin{vmatrix}
x_1 & y_1 & 1 \\
x_2 & y_2 & 1 \\
x_3 & y_3 & 1
\end{vmatrix}
\right| \\[6pt]
&= \frac{1}{2}
\left|
\begin{vmatrix}
-2 & 6 & 1 \\
2k & 4 & 1 \\
2k+1 & 10 & 1
\end{vmatrix}
\right| \\[6pt]
&= \frac{1}{2}
\left|
-2\begin{vmatrix}4 & 1 \\ 10 & 1\end{vmatrix}
-6\begin{vmatrix}2k & 1 \\ 2k+1 & 1\end{vmatrix}
+\begin{vmatrix}2k & 4 \\ 2k+1 & 10\end{vmatrix}
\right| \\[6pt]
&= \frac{1}{2}
\Big[(-2)(4-10) -6(2k-(2k+1)) + (20k-8k-4)\Big] \\[6pt]
&= \frac{1}{2}|12k+14|
\end{align}
\end{frame}

\begin{frame}{Solution}
\text{Given area of triangle} = 52 sq.units, 
\begin{align}
\frac{1}{2}|12k+14| &= 52 \\[6pt]
|12k+14| &= 104.
\end{align}

Now consider the two cases from the absolute value,

\textbf{Case 1:}
\begin{align}
12k+14 &= 104 \\
12k &= 90 \\
k &= \frac{90}{12} = \frac{15}{2}.
\end{align}


\end{frame}

\begin{frame}{Solution}

\textbf{Case 2:}
\begin{align}
12k+14 &= -104 \\
12k &= -118 \\
k &= -\frac{118}{12} = -\frac{59}{6}.
\end{align}

Because the problem requires \(k>0\), we discard the negative solution.
\begin{center}
\begin{align}
\boxed{k = \dfrac{15}{2}}
\end{align}
\end{center}
\end{frame}

#include <stdio.h>

#ifdef _WIN32
#define EXPORT __declspec(dllexport)
#else
#define EXPORT
#endif

// Function to compute the positive value of k
EXPORT double solve_k_positive() {
    double k1 = (104.0 - 14.0) / 12.0;   // 7.5
    double k2 = (-104.0 - 14.0) / 12.0;  // -9.8333

    return (k1 > 0) ? k1 : k2;
}
\end{lstlisting}
\end{block}
\end{frame}

lib.solve_k.restype = None
\end{lstlisting}
