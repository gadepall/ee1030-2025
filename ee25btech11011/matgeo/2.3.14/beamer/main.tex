\begin{frame}{Question}
If a line makes an angle of $30^\circ$ with the positive direction of the X-axis, $120^\circ$  with the positive direction of the Y-axis, then find the angle which it makes with the positive direction of the Z-axis
\end{frame}

\begin{frame}{Solution}
Let the direction (unit) vector of the line be
\begin{align}  
d = \begin{bmatrix} a \\ b \\ c \end{bmatrix}.
\end{align}

Condition with $x$-axis:
\begin{align}
d^T e_x &= \cos 30^\circ 
\end{align}

\begin{align}
\begin{bmatrix} a & b & c \end{bmatrix}
\begin{bmatrix} 1 \\ 0 \\ 0 \end{bmatrix} &= \cos 30^\circ 
\end{align}

\begin{align}
a &= \tfrac{\sqrt{3}}{2} 
\end{align}
\end{frame}

\begin{frame}{Solution}
Condition with $y$-axis:
\begin{align}
d^T e_y &= \cos 120^\circ 
\end{align}


\begin{align}
\begin{bmatrix} a & b & c \end{bmatrix}
\begin{bmatrix} 0 \\ 1 \\ 0 \end{bmatrix} &= \cos 120^\circ 
\end{align}


\begin{align}
b &= -\tfrac{1}{2} 
\end{align}

Now using normalization:
\begin{align}
d^T d &= 1 
\end{align}

\begin{align}
\begin{bmatrix} a & b & c \end{bmatrix}
\begin{bmatrix} a \\ b \\ c \end{bmatrix} & = 1 
\end{align}
\end{frame}

\begin{frame}{Solution}
\begin{align}
a^2 + b^2 + c^2 &= 1 \\
\left(\tfrac{\sqrt{3}}{2}\right)^2 + \left(-\tfrac{1}{2}\right)^2 + c^2 &= 1 \\
\tfrac{3}{4} + \tfrac{1}{4} + c^2 &= 1 
\end{align}

\begin{align}
c^2 &= 0 \quad \Rightarrow \quad c = 0
\end{align}

Hence,
\begin{align}
d = \begin{bmatrix} \tfrac{\sqrt{3}}{2} \\[6pt] -\tfrac{1}{2} \\[6pt] 0 \end{bmatrix}
\end{align}
\end{frame}



\begin{frame}{Solution}
Angle with $z$-axis:
\begin{align}
d^T e_z &= \cos \theta 
\end{align}

\begin{align}
\begin{bmatrix} \tfrac{\sqrt{3}}{2} & -\tfrac{1}{2} & 0 \end{bmatrix}
\begin{bmatrix} 0 \\ 0 \\ 1 \end{bmatrix} &= \cos \theta 
\end{align}
\begin{align}
0 &= \cos \theta 
\end{align}

\begin{align}
\theta &= 90^\circ
\end{align}

Therefore, the line makes angles of $30^\circ$, $120^\circ$, and $90^\circ$ with the $x$-, $y$-, and $z$-axes, respectively.


\end{frame}

#include <stdio.h>
#include <math.h>

#define PI 3.14159265358979323846

// Vector struct
typedef struct {
    double x, y, z;
} Vector3;

// Dot product of two vectors
double dotProduct(Vector3 a, Vector3 b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}

// Normalize a vector to unit length
Vector3 normalize(Vector3 v) {
    double length = sqrt(dotProduct(v, v));
    Vector3 result = {v.x / length, v.y / length, v.z / length};
    return result;
}

// Create vector from direction cosines (angles in degrees)
Vector3 fromDirectionCosines(double angle_x_deg, double angle_y_deg) {
    double a = cos(angle_x_deg * PI / 180.0);
    double b = cos(angle_y_deg * PI / 180.0);

    // Since a^2 + b^2 + c^2 = 1, solve for c
    double c_squared = 1.0 - (a*a + b*b);
    double c = 0.0;
\end{lstlisting}

    }
    Vector3 v = {a, b, c};
    return normalize(v);
}

int main() {
