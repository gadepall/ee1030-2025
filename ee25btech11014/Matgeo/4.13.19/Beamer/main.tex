\begin{frame}{Question} If the line $2x + y = k$ passes through the point which divides the line segment joining the points $\bark{1, 1}$ and $\brak{2, 4}$ in the ratio $3:2$, then $k$ equals
\begin{multicols}
\begin{enumerate}
    \item $\frac{29}{5}$
    \item $5$
    \item $6$
    \item $\frac{11}{5}$
\end{enumerate}
\end{multicols}
\end{frame}

\begin{frame}{Solution}
Let the point $\vec{C}$ divide the line segment joining $\vec{A}$ and $\vec{B}$ in the ratio $3:2$ where
\begin{align}
     \vec{A}=\myvec{1\\1}\\
    \vec{B}=\myvec{2\\4}\\
    \vec{C}=\myvec{x\\y}
\end{align}
\end{frame}

\begin{frame}{Formula}
By Section formula,
\begin{align}
\brak{\lambda +1}\vec{C}=\lambda\vec{B}+\vec{A}.\\  
    Here \brak{\lambda=\frac{3}{2}}\\
      \frac{5}{2}\myvec{x\\y}=\frac{3}{2}\myvec{2\\4}+\myvec{1\\1}\\
    \frac{5}{2}\myvec{x\\y}=\myvec{3\\6}+\myvec{1\\1}\\
    \myvec{x\\y}=\frac{2}{5}\myvec{4\\7}\\
    \therefore\vec{C}=\myvec{\frac{8}{5}\\\frac{14}{5}}.
\end{align}
\end{frame}

\begin{frame}{Solution}
The equation of the line can be represented as:
\begin{align}
    \vec{n^T}\vec{x}=c_1\\
 where  \ \vec{n^T}=\myvec{2&1},\vec{x}=\myvec{x\\y} \ and \ c_1=k.
\end{align}
As $\vec{C}$ lies on the above line,
\begin{align}
  \vec{n^T}\vec{C}=c_1\\
 \myvec{2&1}\myvec{\frac{8}{5}\\\frac{14}{5}}=k\\
 k=\frac{16}{5}+\frac{14}{5}=6.\\
 \therefore k=6.
\end{align}
\end{frame}

 \begin{frame}[fragile]
\frametitle{C Code }
\begin{lstlisting}
#include <stdio.h>
// sol[0] = x, sol[1] = y, sol[2] = k
void find_k(double sol[3]) {
   
    double x1 = 1, y1 = 1;
    double x2 = 2, y2 = 4;
    int m = 3, n = 2;

    // Internal division point (x, y)
    double x = (m*x2 + n*x1) / (double)(m+n);
    double y = (m*y2 + n*y1) / (double)(m+n);

    double k = 2*x + y;

    // Store results
    sol[0] = x;
    sol[1] = y;
    sol[2] = k;
}
\end{lstlisting}
\end{frame}
