\begin{frame}{Theoretical Solution}
We can calculate the magnitude of sides of the triangle using norms:
\begin{align}
    \brak{\vec{A}-\vec{B}}=\myvec{-4\\0}-\myvec{0\\3}=\myvec{-4\\3}
\end{align}
\begin{align}
 \brak{\vec{B}-\vec{C}}=\myvec{0\\3}-\myvec{4\\0}=\myvec{-4\\3}
 \end{align}
 \begin{align}
 \brak{\vec{A}-\vec{C}}=\myvec{-4\\0}-\myvec{4\\0}=\myvec{-8\\0}
 \end{align}
\end{frame}

\begin{frame}{Formulae}
\begin{align}
\left|\vec{A}-\vec{B} \right| &= \sqrt{ \brak{{\vec{A}-\vec{B}}}^T \cdot \brak{\vec{A}-\vec{B}}}\\
 \left|\vec{B}-\vec{C} \right| &= \sqrt{ \brak{{\vec{B}-\vec{C}}}^T \cdot \brak{\vec{B}-\vec{C}}}\\
 \left|\vec{A}-\vec{C} \right| &= \sqrt{ \brak{{\vec{A}-\vec{C}}}^T \cdot\brak{ \vec{A}-\vec{C}}}
\end{align}
\end{frame}

\begin{frame}{Calculation}
\begin{align}
  \brak{{\vec{A}-\vec{B}}}^T\cdot\brak{ \vec{A}-\vec{B}}= \myvec{-4 & -3}\myvec{-4\\-3}\\
\left|\vec{A}-\vec{B} \right| &= \sqrt{ \brak{-4}^2+\brak{-3}^2}\\
\left|\vec{A}-\vec{B} \right| &= 5\\
\brak{{\vec{B}-\vec{C}}}^T\cdot\brak{ \vec{B}-\vec{C}}= \myvec{-4 & 3}\myvec{-4\\3}\\
  \left|\vec{B}-\vec{C} \right| &=\sqrt{ \brak{-4}^2+\brak{3}^2}\\
\left|\vec{B}-\vec{C} \right| &= 5\\
\end{align}
\end{frame}

\begin{frame}{Theoritical solution}
\begin{align}
  \brak{{\vec{A}-\vec{C}}}^T\cdot\brak{ \vec{A}-\vec{C}}= \myvec{-8 & 0}\myvec{-8\\0}\\
 \left|\vec{A}-\vec{C} \right| &=\sqrt{ \brak{-8}^2+\brak{0}^2}\\
\left|\vec{A}-\vec{C} \right| &= 8\\
 As \left|\vec{A}-\vec{B}\right|^2+ \left|\vec{B}-\vec{C}\right|^2=25+25=50\\
 and \left|\vec{A}-\vec{C}\right|^2=64
 \end{align}
 \begin{center}
      Hence, the triangle is \textbf{not RIGHT} angled.
 \end{center}
\end{frame}

\begin{frame}{Formulae}
Angles at the vertices \vec{A},\vec{B} and \vec{C}. 
\begin{align}
    \cos\brak{\angle A}=\frac{\brak{\vec{A}-\vec{B}}^T \cdot \brak{ \vec{A}-\vec{C}}}{\left|\vec{A}-\vec{B} \right|\left|\vec{A}-\vec{C} \right|}
\end{align}    
\begin{align}
   \cos\brak{\angle B}=\frac{\brak{\vec{B}-\vec{A}}^T \cdot \brak{ \vec{B}-\vec{C}}}{\left|\vec{B}-\vec{A} \right|\left|\vec{B}-\vec{C} \right|}
\end{align}
\begin{align}
    \cos\brak{\angle C}=\frac{\brak{\vec{C}-\vec{A}}^T \cdot \brak{ \vec{C}-\vec{B}}}{\left|\vec{C}-\vec{A} \right|\left|\vec{C}-\vec{B} \right|}
\end{align}    
\end{frame}

\begin{frame}{Calculation}
\begin{align}
\brak{\vec{A}-\vec{B}}^T \cdot \brak{\vec{A}-\vec{C}}= 
\myvec{-4 & 3}
\myvec{-8 \\ 0}= 32\\
\cos(\angle A) = \frac{32}{5 \cdot 8} = 0.8\\
\Rightarrow \angle A = \cos^{-1}(0.8) \approx 36.87^\circ\\
\brak{\vec{B}-\vec{A}}^T \cdot \brak{\vec{B}-\vec{C}} = 
\myvec{-4 & -3}
\myvec{4 \\
-3}
= -16 + 9 = -7\\
\cos(\angle B) = \frac{-7}{5 \cdot 5 } = -0.28\\
\Rightarrow \angle B = \cos^{-1}(-0.28) \approx 106.26^\circ
\end{align}
\end{frame}

\begin{align}
    \brak{\vec{C}-\vec{A}}^T \cdot \brak{\vec{C}-\vec{B}} = 
\myvec{-8 & 0}
\myvec{-4 \\
3}= 32\\
\cos(\angle C) = \frac{32}{8 \cdot 5} = 0.8\\
\Rightarrow \angle C = \cos^{-1}(0.8) \approx 36.87^\circ
\end{align}
\begin{align}
\therefore \angle A \approx 36.87^\circ, \quad 
\angle B \approx 106.26^\circ, \quad 
\angle C \approx 36.87^\circ
\end{align}

  // --- Compute side vectors ---
    double AB[2] = {B[0] - A[0], B[1] - A[1]};
    double BC[2] = {C[0] - B[0], C[1] - B[1]};
    double CA[2] = {A[0] - C[0], A[1] - C[1]};

    // --- Compute side lengths using dot product (matrix method) ---
    sides[0] = sqrt(AB[0]*AB[0] + AB[1]*AB[1]); // AB
    sides[1] = sqrt(BC[0]*BC[0] + BC[1]*BC[1]); // BC
