\textbf{Question:}
Find the area of the quardrilateral \textit{ABCD}, whose vertices are $\Vec{A}\brak{-3,-1},\Vec{B}\brak{-2,-4},\Vec{C}\brak{4,-1}$ and $\Vec{D}\brak{3,4}$

\textbf{Solution:}Given the points,
\begin{table}[H]    
  \centering
  \input{Table/table}
  \caption{Variables Used}
  \label{tab:2.7.26}
\end{table}

 We can say that, 
 \begin{align}
Area \brak{\textit{ABCD}}=Area\brak{\textit{ABC}}+Area\brak{\textit{ADC}}
 \end{align}

 Formulae:
 \begin{align}
\textit{Area}\brak{\triangle ABC}
= \frac{1}{2} \left\| \brak{\Vec{B}-\Vec{A}} \times \brak{\Vec{B}-\Vec{C}} \right\|
\\
\textit{Area}\brak{\triangle ADC}
= \frac{1}{2} \left\| \brak{\Vec{D}-\Vec{A}} \times \brak{\Vec{D}-\Vec{C}} \right\|
\end{align}

 \begin{align}
 \brak{\Vec{B}-\Vec{A}}=\myvec{-2\\-4}-\myvec{-3\\-1}=\myvec{1\\-3}\\ 
  \brak{\Vec{B}-\Vec{C}}=\myvec{-2\\-4}-\myvec{4\\-1}=\myvec{-6\\-3}
 \end{align}
 
\begin{align}
 \left\|\brak{\Vec{B}-\Vec{A}} \times \brak{\Vec{B}-\Vec{C}}\right\|=\left\| \myvec{1\\-3}\times \myvec{-6\\-3}\right\| =\left|-3-\brak{-6}\brak{-3}\right|=\left|-3-18\right|=21
\end{align}

 and\begin{align}
      \brak{\Vec{D}-\Vec{A}}=\myvec{3\\4}-\myvec{-3\\-1}=\myvec{6\\5}\\
       \brak{\Vec{D}-\Vec{C}}=\myvec{3\\4}-\myvec{4\\-1}=\myvec{-1\\5}
 \end{align}
 \begin{align}
 \left\|\brak{\Vec{D}-\Vec{A}} \times \brak{\Vec{D}-\Vec{C}}\right\|=\left\|\myvec{6\\5} \times \myvec{-1\\5} \right\| =\left|\brak{6}\brak{5}-5\brak{-1}\right|=\brak{30+5}=35
\end{align}
\begin{align}
    Area \brak{\textit{ABCD}}=\frac{1}{2}\left|\brak{-21}\right|+\frac{1}{2}\left|\brak{35}\right|=\frac{56}{2}=28 sq.units
\end{align}

\begin{center}
$\therefore$ The Area of the quadrilateral \textit{ABCD} is 28 sq.units.
\end{center}
