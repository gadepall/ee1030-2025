\begin{frame}{Question}Find the area of the quardrilateral ABCD, whose vertices are
\Vec{A}\brak{−3, −1} , \Vec{B} \brak{−2, −4} , \Vec{C} \brak{4, −1} and \Vec{D} \brak{3, 4}.
\end{frame}

\begin{frame}{Formulae}
 We can say that,
\begin{align}
Area \brak{ABCD} = Area \brak{ABC} + Area \brak{ADC} \\
Area\brak{\triangle ABC}= {\frac{1}{2}} \left\| \brak{\Vec{B}-\Vec{A}} \times \brak{\Vec{B}-\Vec{C}} \right\|\\
Area\brak{\triangle ADC}= {\frac{1}{2}} \left\| \brak{\Vec{D}-\Vec{A}} \times \brak{\Vec{D}-\Vec{C}} \right\|
\end{align}
\end{frame}

\begin{frame}{solution}

  \begin{align}
 \brak{\Vec{B}-\Vec{A}}=\myvec{-2\\-4}-\myvec{-3\\-1}=\myvec{1\\-3}\\ 
  \brak{\Vec{B}-\Vec{C}}=\myvec{-2\\-4}-\myvec{4\\-1}=\myvec{-6\\-3}
 \end{align}
 
\begin{align}
 \left\|\brak{\Vec{B}-\Vec{A}} \times \brak{\Vec{B}-\Vec{C}}\right\|=\left\| \myvec{1\\-3}\times \myvec{-6\\-3}\right\|\\ \implies\left|-3-\brak{-6}\brak{-3}\right|=\left|-3-18\right|=21
\end{align}

 and\begin{align}
      \brak{\Vec{D}-\Vec{A}}=\myvec{3\\4}-\myvec{-3\\-1}=\myvec{6\\5}\\
       \brak{\Vec{D}-\Vec{C}}=\myvec{3\\4}-\myvec{4\\-1}=\myvec{-1\\5}
 \end{align}

\end{frame}
\begin{frame}{Solution}
 \begin{align}
\left\|\brak{\Vec{D}-\Vec{A}} \times \brak{\Vec{D}-\Vec{C}}\right\|=\left\|\myvec{6\\5} \times \myvec{-1\\5} \right\|\\ \implies\left|\brak{6}\brak{5}-5\brak{-1}\right|=30+5=35
\end{align}

\begin{align}
    Area \brak{\textit{ABCD}}=\frac{1}{2} \left|\brak{-21}\right|+\frac{1}{2}\left|\brak{35}\right|=\frac{56}{2}=28 sq.units
\end{align}
\begin{center}
$\therefore$ The Area of the quadrilateral \textit{ABCD} is 28 sq.units.
\end{center}
 \end{frame}

#include <stdio.h>
#include <math.h>

// Function to calculate area of quadrilateral using shoelace formula
float quadArea(int x[], int y[], int n) {
    float area = 0.0;
    int i, j;

    for (i = 0; i < n; i++) {
        j = (i + 1) % n;  // Next vertex index
        area += (x[i] * y[j]) - (y[i] * x[j]);
    }
\end{lstlisting}

float findQuadArea(int x1, int y1, int x2, int y2,
                   int x3, int y3, int x4, int y4) {
    int x[4] = {x1, x2, x3, x4};
    int y[4] = {y1, y2, y3, y4};
    return quadArea(x, y, 4);
}
