\textbf{Question}:
Write the equation of the lines for which $\tan\theta = \frac{1}{2}$, where $\theta$ is the inclination of the line, and
\begin{enumerate}
\item[(a)] y intercept $- \frac{3}{2}$
\item[(b)] x intercept $4$
\end{enumerate}
\textbf{Solution 1: }
\begin{table}[H]    
  \centering
  \input{tables/table.tex}
  \caption{Variables Used}
  \label{tab:4.3.49}
\end{table}
Let $\vec{B}$ be a point on the line 
\begin{align}
\vec{A}=\myvec{
                0
                \\
                 -\frac{3}{2}  }\\
\text{Let }\vec{M}=\myvec{
                           1
                           \\
                            m
                            }\\
            \vec{M}=\myvec{
                           1
                           \\
                            \frac{1}{2}
                            }\\
            (\vec{B}-\vec{A})=\vec{M}\\
            \vec{B}=\myvec{
                           1
                           \\
                           -1
                            }
\end{align}
Let eq of line be
\begin{align}
\vec{n^T}\vec{x}=1
\end{align}
As $\vec{A}$ and $\vec{B}$ lie on the line 
\begin{align}
\vec{n^T}\vec{A}=1\\
\vec{n^T}\vec{B}=1
\end{align}
From eq $(0.7)$ and $(0.8)$
\begin{align}
\myvec{0  &  -\frac{3}{2}
          \\
      1 & -1 }
n     
= \myvec{
           1
           \\
            1
               }
\end{align}
In augmented matrix form
\begin{align}
\begin{bmatrix}
0 & -\frac{3}{2}  & \big| & 1 \\
1&-1 & \big| & 1
\end{bmatrix}
\end{align}
Swap $R_{1}$ and $R_{2}$
\begin{align}
\begin{bmatrix}
1 &-1 & \big| & 1\\
0 & -\frac{3}{2} & \big| & 1 
\end{bmatrix}
\end{align}
$R_2 \rightarrow {-\frac{2R_2}{3}}$
\begin{align}
\begin{bmatrix}
1 &-1 & \big| &1\\
0 & 1 & \big| & -\frac{2}{3} 
\end{bmatrix}
\end{align}
$R_1 \rightarrow {R_1+R_2}$
\begin{align}
    \begin{bmatrix}
            1 &0 & \big| & \frac{1}{3}\\
            0 & 1 & \big| &-\frac{2}{3} 
    \end{bmatrix}
\end{align}
\begin{align}
    \vec{n}=\myvec{\frac{1}{3}
                \\
                -\frac{2}{3}
\end{align}
     

hence eq of line is 
\begin{align}
  \vec{n^T}\vec{x}=1\\
  \myvec{\frac{1}{3}&-\frac{2}{3}}\vec{x}=1
\end{align}

Let $\vec{B}$ be a point on the line 
\begin{align}
\vec{A}=\myvec{
                4
                \\
                 0  }\\
\text{Let }\vec{M}=\myvec{
                           1
                           \\
                            m
                            }\\
            \vec{M}=\myvec{
                           1
                           \\
                            \frac{1}{2}
                            }\\
            (\vec{B}-\vec{A})=\vec{M}\\
            \vec{B}=\myvec{
                           5
                           \\
                           \frac{1}{2}
                            }
\end{align}
Let eq of line be
\begin{align}
\vec{n^T}\vec{x}=1
\end{align}
As $\vec{A}$ and $\vec{B}$ lie on the line 
\begin{align}
\vec{n^T}\vec{A}=1\\
\vec{n^T}\vec{B}=1
\end{align}
From eq $(0.22)$ and $(0.23)$
\begin{align}
\myvec{4  &  0
          \\
      5 & \frac{1}{2} }
\vec{n}    
= \myvec{
           1
           \\
            1
               }
\end{align}
In augmented matrix form
\begin{align}
\begin{bmatrix}
4 & 0 & \big| & 1 \\
5 &\frac{1}{2} & \big| & 1
\end{bmatrix}
\end{align}
$R_1 \rightarrow {\frac{R_1}{4}}$
\begin{align}
\begin{bmatrix}
1 & 0` & \big| & \frac{1}{4}\\
5 & \frac{1}{2} & \big| & 1 
\end{bmatrix}
\end{align}
$R_2 \rightarrow {R_2-5R_1}$
\begin{align}
\begin{bmatrix}
1 &0 & \big| & \frac{1}{4}\\
0 & \frac{1}{2} & \big| & -\frac{1}{4} 
\end{bmatrix}
\end{align}
$R_2 \rightarrow {2R_2}$
\begin{align}
\begin{bmatrix}
1 &0 & \big| & \frac{1}{4}\\
0 & 1 & \big| & -\frac{1}{2} 
\end{bmatrix}
\\
\vec{n}=\myvec{\frac{1}{4}
                \\
                -\frac{1}{2}}
\end{align}
hence eq of line is 
\begin{align}
    \vec{n^T}\vec{x}=1\\
    \myvec{\frac{1}{4}&-\frac{1}{2}}\vec{x}=1
\end{align}
