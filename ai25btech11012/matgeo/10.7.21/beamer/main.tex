% Question frame
\begin{frame}
\frametitle{Question}
Find the points on the curve $x^2$ + $y^2$ - 2x - 3 = 0 at which the tangents are parallel
to the x-axis.
\end{frame}


% Solution steps
\begin{frame}
\frametitle{Solution}
For the given curve 

\begin{align}
    \textbf{u} = \begin{bmatrix}-1\\0\end{bmatrix}\\
    f = -3
\end{align}


\end{frame}
\begin{frame}
\frametitle{Solution}
For a circle the points of contact are given by the formula :

\begin{align}
\textbf{q}_{ij} = \left (\pm r \frac{\textbf{n}_j}{{\lVert\textbf{n}_j\rVert}} - \textbf{u}\right)
\end{align}
Where i,j = 1,2

Since tangents are parallel to x - axis ,the normal is given as 
\begin{align}
  \textbf{n} =\begin{bmatrix} 0\\1\end{bmatrix}
\end{align}

\end{frame}


\begin{frame}
\frametitle{Solution}
Solving 0.5 we get :

\begin{align}
 \textbf{q} = (\pm 2\begin{bmatrix}0\\1\end{bmatrix} - \begin{bmatrix}-1\\0\end{bmatrix})\\
= \begin{bmatrix}1\\-2\end{bmatrix} , \begin{bmatrix}1\\2\end{bmatrix}
\end{align}


Hence the two points of contact are 
\begin{align}
    \begin{bmatrix}1\\2\end{bmatrix} \quad and \quad \begin{bmatrix}1\\-2\end{bmatrix}
\end{align}

\end{frame}
