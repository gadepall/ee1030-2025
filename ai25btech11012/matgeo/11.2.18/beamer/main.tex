\begin{frame}
\frametitle{Question}
The line segment joining the mid-points of any two sides of a triangle is parallel to
the third side and is half of it.
\end{frame}


% Solution steps
\begin{frame}
Let ABC be the triangle ,\\
\textbf{E }and \textbf{F} be the midpoints of AB and BC respectively .


\begin{align}
  \textbf{E}  = \frac{A + B}{2}\\
  \textbf{F}  = \frac{B + C}{2}
\end{align}
\end{frame}


\begin{frame}
\frametitle{Solution}
The line joining these two points is  


\begin{align}
 \textbf{EF}=  \frac{C - A}{2}
\end{align}

The line AC is
\begin{align}
\textbf{AC} = \textbf{C} - \textbf{A}
\end{align}

\end{frame}


\begin{frame}
Hence ,
\begin{align}
    \textbf{EF} = \frac{1}{2}\textbf{AC}
\end{align}
Therefore the line segment joining the mid-points of any two sides of a triangle is parallel to
the third side and is half of it

\end{frame}
