\textbf{Question}:\newline
Let $\vec{a} = \hat{\vec{i}} + 2\hat{\vec{j}} + \hat{\vec{k}}$, $\vec{b} = \hat{\vec{i}} - \hat{\vec{j}} + \hat{\vec{k}}$ and $\vec{c} = \hat{\vec{i}} + \hat{\vec{j}} - \hat{\vec{k}}$. A vector in the plane of $\vec{a}$ and $\vec{b}$ whose projection on $\vec{c}$ is $\frac{1}{\sqrt{3}}$, is
\begin{multicols}{4}
\begin{enumerate}
\item 4$\hat{\vec{i}} - \hat{\vec{j}} + 4\hat{\vec{k}}$
\item 3$\hat{\vec{i}} + \hat{\vec{j}} - 3\hat{\vec{k}}$
\item $2\hat{\vec{i}} + \hat{\vec{j}} - 2\hat{\vec{k}}$
\item $4\hat{\vec{i}} + \hat{\vec{j}} - 4\hat{\vec{k}}$
\end{enumerate}
\end{multicols}

\textbf{Solution: }

Given
\begin{align}
\vec{a}=\myvec{1\\2\\1},\vec{b}=\myvec{1\\-1\\1},\vec{c}=\myvec{1\\1\\-1}.
\end{align}

Let $\vec{r}$ be coplanar to $\vec{a}$ and $\vec{b}$
\begin{align}
 \vec{r}&=\vec{a}+t\vec{b} 
 \end{align}
Given the projection of $\vec{r}$ on $\vec{c}$ is $\frac{1}{\sqrt{3}}$
\begin{align}
\frac{\vec{r}\cdot\vec{c}}{\norm{\vec{c}}} = \frac{1}{\sqrt{3}} 
\end{align}
\begin{align}
\vec{r}\cdot\vec{c}&=\brak{\vec{a}+t\vec{b}}\cdot\vec{c}\\
&=\vec{a}\cdot\vec{c}+t\brak{\vec{b}\cdot\vec{c}} \\
&=\vec{a}^\top\vec{c}+t\brak{\vec{b}^\top\vec{c}}
\end{align}
On substituting the given values,
\begin{align}
\vec{r}\cdot\vec{c}=&\myvec{1&2&1}\myvec{1\\1\\-1}+t\brak{\myvec{1&-1&1}\myvec{1\\1\\-1}}\\
&=1+2-1+t\brak{1-1-1}\\
&=2-t 
\end{align}
\begin{align}
\norm{\vec{c}}^2=\vec{c}^\top\vec{c}&=\myvec{1&1&-1}\myvec{1\\1\\-1} \\
&=1+1+1=3  
\end{align}
\begin{align}
\frac{|2-t|}{\sqrt{3}} = \frac{1}{\sqrt{3}}
\end{align}
yielding
\begin{align}
 \brak{2-t}=\pm 1 \\
 \implies t=1\ \text{or}\ 3
\end{align}
When t=1, we have
\begin{align}
 \vec{r}=\myvec{2 \\ 1 \\2}
\end{align}
When t=3, we have
\begin{align}
 \vec{r}=\myvec{4 \\ -1 \\4}
\end{align}
Hence Option(1) is the correct answer
