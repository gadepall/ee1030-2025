\textbf{Question}:\\
Find the coordinates of the point at which the circles $x^2+y^2-4x-2y=-4$ and $x^2+y^2-12x-8y=-36$ touch each other. Also find equations of the common tangents touching the circles in the distinct points.\\    
\bigskip

\textbf{Solution}:\\
Circle $\brak{C_1}$:
\begin{align}
    x^2+y^2-4x-2y+4=0
\end{align}
\begin{align}
    \implies u=\myvec{-2\\-1}\text{ and }f=4  
\end{align}
\begin{align}
    \implies \text{Centre}\brak{\vec{c_1}} = \myvec{2\\1} \text{ and } r_1=\sqrt{\norm{u}^2-f}=1
\end{align}
Circle $\brak{C_2}$:
\begin{align}
    x^2+y^2-12x-8y+36=0
\end{align}
\begin{align}
    \implies u=\myvec{-6\\-4}\text{ and }f=36  
\end{align}
\begin{align}
    \implies \text{Centre}\brak{\vec{c_2}} = \myvec{6\\4} \text{ and } r_2=\sqrt{\norm{u}^2-f}=4
\end{align}
This point of contact$\brak{\vec{P}}$ lies on the line segment connecting the two centers, $\vec{c_1}$ and $\vec{c_2}$,and divides it in the ratio of their radii, $r_1:r_2$.
\begin{align}
    \vec{P} = \frac{1}{r_1+r_2}\myvec{c_1 & c_2}\myvec{r_2\\r_1}
\end{align}
\begin{align}
    \vec{P} = \frac{1}{5}\myvec{2 & 6\\1 & 4}\myvec{4\\1}
\end{align}
\begin{align}
    \vec{P} = \myvec{14/5\\8/5}
\end{align}
Let $\vec{n}^\top \vec{x}=c$ be the equation of common tangent to both the circles.\\
\begin{align}
    \frac{\abs{\vec{n}^\top\vec{c_1}-c}}{\norm{\vec{n}}} = r_1\implies \vec{n}^\top\vec{c_1} = c\pm\norm{\vec{n}}
\end{align}
\begin{align}
    \frac{\abs{\vec{n}^\top\vec{c_2}-c}}{\norm{\vec{n}}} = r_2\implies \vec{n}^\top\vec{c_2} = c\pm4\norm{\vec{n}}
\end{align}
\begin{align}
    \vec{n}^\top\brak{\vec{c_2}-\vec{c_1}}=\pm3\norm{\vec{n}}
\end{align}
substitute $\vec{n} = \myvec{-m\\1}$ where m is the slope of the line.
\begin{align}
    \implies  \myvec{-m & 1}\myvec{4\\3} = \pm3\sqrt{1+m^2}
\end{align}
\begin{align}
    \implies 16m ^ 2 - 24m + 9 = 9(1 + m ^ 2)
\end{align}
\begin{align}
    \implies 7m ^ 2 - 24m = 0
\end{align}
\begin{align}
    m=0 \text{ and } m=24/7\implies \vec{n}=\myvec{0\\1}\text{ and } \vec{n} = \myvec{-24\\7}
\end{align}
To find the tangent line ,substitute $\vec{n} = \myvec{-24\\7}$ in equation (10), we get
\begin{align}
    c = -66\text{ and } c=-16
\end{align}
And equation (11)  holds only when $c=-16$.\\

Therefore the two tangent lines are $\myvec{0 & 1}\vec{x}=0$
and $\myvec{-24 & 7}\vec{x}=-16$
