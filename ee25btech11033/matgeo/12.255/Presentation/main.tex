\begin{frame}{Question}
The least squares generalized inverse of an overdetermined problem is expressed as  


\begin{enumerate}
	\item \ $\brak{\vec{G}^\top \vec{G}}^{-1} \vec{G}^\top$  
	\item \ $\brak{\vec{G}^\top \vec{G}}^{-1}$  
	\item \ $\vec{G}^\top \brak{\vec{G} \vec{G}^\top}^{-1} $ 
	\item \ $\brak{\vec{G} \vec{G}^\top}^{-1}$  
\end{enumerate}\\  
\bigskip

\end{frame}



\begin{frame}{Theoretical Solution}
Let us consider the overdetermined system of equations,
\begin{align}
\vec{G}\vec{x} = \vec{b}
\end{align}
Define the residual vector as,
\begin{align}
\vec{r} = \vec{b} - \vec{G}\vec{x}
\end{align}
The least--squares problem is to minimize the squared error:
\begin{align}
\vec{r}^{\top} \vec{r} = (\vec{b} - \vec{G}\vec{x})^{\top}(\vec{b} - \vec{G}\vec{x})
\end{align}

\begin{align}
\vec{r}^{\top} \vec{r}= (\vec{b} ^{\top} - \vec{x}^{\top}\vec{G}^{\top})(\vec{b} - \vec{G}\vec{x})
\end{align}

\begin{align}
\vec{r}^{\top} \vec{r}= \vec{x}^{\top}\vec{G}^{\top}\vec{G}\vec{x} - \vec{x}^{\top}\vec{G}^{\top}\vec{b} - \vec{b} ^{\top}\vec{G}\vec{x} + \vec{b} ^{\top}\vec{b} 
\end{align}
\end{frame}

\begin{frame}{Theoretical Solution}
Now, to minimize \(\vec{r}^{\top}\vec{r}\), define
\begin{align}
f(x) = \vec{x}^{\top}\vec{G}^{\top}\vec{G}\vec{x} - 2\vec{x}^{\top}\vec{G}^{\top}\vec{b} + \vec{b} ^{\top}\vec{b} 
\end{align}

Taking gradient with respect to \(\vec{x}\):
\begin{align}
\nabla_x f(x) = 2\vec{G}^{\top}\vec{G}\vec{x} - 2\vec{G}^{\top}\vec{b} 
\end{align}

Setting \(\nabla_x f(x) = 0\):
\begin{align}
\vec{G}^{\top}\vec{G}\vec{x} = \vec{G}^{\top}\vec{b} 
\end{align}

Therefore, the least squares solution is
\begin{align}
\vec{x} = \boxed{(\vec{G}^{\top}\vec{G})^{-1}\vec{G}^{\top}\vec{b} }
\end{align}\\
Therefore, $\brak{\vec{G}^{\top}\vec{G}}^{-1}\vec{G}^{\top}$ is the least squares generalized inverse of an overdetermined problem.
\end{frame}
