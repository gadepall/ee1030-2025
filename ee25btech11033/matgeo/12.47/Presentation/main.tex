\begin{frame}{Question}
The following simultaneous equations
\begin{align*}
x + y + z &= 3 \\
x + 2y + 3z &= 4 \\
x + 4y + kz &= 6
\end{align*}
will NOT have a unique solution for $k$ equal to 
\begin{enumerate}

\item $0$
\item $5$
\item $6$
\item $7$

\end{enumerate}\\    
\bigskip
\end{frame}



\begin{frame}{Theoretical Solution}
The equations can be written as,
\begin{align}
    \myvec{1 & 1 & 1 \\ 1 & 2 & 3 \\ 1 & 4 & k }\vec{x}=\myvec{3\\4\\6}
\end{align}
where
\begin{align*}
    \vec{x} = \myvec{x\\y\\z}
\end{align*}\\
\bigskip 


\end{frame}

\begin{frame}{Theoretical Solution}
The augmented matrix is given by,
\begin{align}
    \augvec{3}{1}{1 & 1 & 1 & 3 \\
1 & 2 & 3 & 4 \\
1 & 4 & k & 6}
\end{align}
\begin{align}
    R_2 \rightarrow R_2 - R_1\text{ and }R_3 \rightarrow R_3 - R_1\implies \augvec{3}{1}{1 & 1 & 1 & 3 \\
0 & 1 & 2 & 1 \\
0 & 3 & k-1 & 3}
\end{align}
\begin{align}
    R_3 \rightarrow R_3 - 3R_2\implies\augvec{3}{1}{1 & 1 & 1 & 3 \\
0 & 1 & 2 & 1 \\
0 & 0 & k-7 & 0}
\end{align}\\
\bigskip
\end{frame}


\begin{frame}{Theoretical Solution}
\begin{align}
    \myvec{1 & 1 & 1\\0 & 1 & 2\\0 & 0 & k-7}\vec{x}=\myvec{3\\1\\0}
\end{align}
For the system to \textbf{NOT} have a unique solution, the coefficient of $z$ in the final row must be zero.
\begin{align}
    k-7=0   
\end{align}
\begin{align}
    \implies k=7
\end{align}
\end{frame}
