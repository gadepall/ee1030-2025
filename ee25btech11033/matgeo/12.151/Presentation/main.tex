\begin{frame}{Question}
A point $\vec{P}(10,3)$ MPa on the Mohr's circle represents normal and shear stresses. If the centre of the Mohr's circle is $\vec{C}(6,0)$ MPa, the normal and shear stresses in MPa on the point diametrically opposite to $\vec{P}$ are

\begin{enumerate}

    \item 2, $-3$
    \item 4, $-3$
    \item 2, 3
    \item 4, 3

\end{enumerate}\\    
\bigskip
\end{frame}



\begin{frame}{Theoretical Solution}
A point on the Mohr's circle, 
\begin{align}
    \vec{P} = \myvec{\sigma_P\\ \tau_P} = \myvec{10\\3} \text{ MPa}
\end{align}
where,
\begin{align*}
    \sigma = \text{Normal Stress} \text{ and } \tau = \text{Shear Stress}
\end{align*}
The center of the Mohr's circle,
\begin{align}
    \vec{C} = \myvec{\sigma_C\\ \tau_C} = \myvec{6\\0} \text{ MPa}
\end{align}
Let P' be the point diametrically opposite to P ,
\begin{align}
    \vec{P'} = \myvec{\sigma_P'\\ \tau_P'} \text{ MPa}
\end{align}
\end{frame}

\begin{frame}{Theoretical Solution}
The center $\vec{C}$ is the midpoint of the line segment connecting $\vec{P}$ and $\vec{P'}$. We can use the midpoint formula for both the horizontal (normal stress) and vertical (shear stress) coordinates.
\begin{align}
    \vec{C} = \frac{\vec{P} + \vec{P'}}{2}
\end{align}
\begin{align}
    \vec{P'} = 2\vec{C}-\vec{P}
\end{align}
\begin{align}
    \vec{P'} = 2 \myvec{6\\0}-\myvec{10\\3}
\end{align}
\begin{align}
    \implies \vec{P'} = \myvec{\sigma_P'\\ \tau_P'} = \myvec{2\\-3}\text{ MPa}
\end{align}\\
\bigskip

\end{frame}
