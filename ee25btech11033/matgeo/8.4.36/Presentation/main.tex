\begin{frame}{Question}
Two sets $A$ and $B$ are as under:
$A=\cbrak{\brak{a,b}\in\mathbb{R}\times\mathbb{R}:\abs{a-5}<1\text{ and} \abs{b-5}<1}$
$B=\cbrak{\brak{a,b}\in\mathbb{R}\times\mathbb{R}:4(a-6)^2\text{+}9(b-5)^2\leq36}$

			\begin{multicols}
	\begin{enumerate}
		\item $A\subset B$ 
		\item $A\cap B$
		\item neither $A\subset B$ nor $B\subset A$
		\item $B\subset A$
	\end{enumerate}\end{multicols}\\    
\bigskip
\end{frame}



\begin{frame}{Theoretical Solution}
Set A:
\begin{align}
    A=\cbrak{\brak{a,b}\in\mathbb{R}\times\mathbb{R}:\abs{a-5}<1\text{ and} \abs{b-5}<1}
\end{align}
\begin{align}
    \abs{a-5}<1\implies4<a<6 \text{ and } \abs{b-5}<1\implies4<b<6
\end{align}
This describes the interior of a square in the ab-plane. The square is centered at $\brak{5,5}$ and its sides are the lines $a=4$, $a=6$, $b=4$, and $b=6$.\\
\bigskip 


\end{frame}

\begin{frame}{Theoretical Solution}
Set B:
\begin{align}
    B=\cbrak{\brak{a,b}\in\mathbb{R}\times\mathbb{R}:4(a-6)^2\text{+}9(b-5)^2\leq36}
\end{align}
\begin{align}
    \frac{(a-6)^2}{3^2} + \frac{(b-5)^2}{2^2} \leq 1
\end{align}
This can be expressed in the form ,
\begin{align*}
\vec{x}^{\top}\vec{V}\vec{x}+2\vec{u}^{\top}\vec{x}+f\leq0
\end{align*}
where,
\begin{align}
    \vec{V} = \myvec{4 & 0\\0 & 9} , \vec{u} = \myvec{-24\\-45} \text{ and } f=333.  
\end{align}
This describes the region of an ellipse, including its interior, in the ab-plane. The ellipse is centered at (6,5) and has a horizontal semi-major axis of length 3 and a vertical semi-minor axis of length 2.\\
\bigskip
\end{frame}


\begin{frame}{Theoretical Solution}
Now, we substitute the maximum possible values for these terms into the expression for the ellipse to see if the condition for Set B is met:\\
On substitutting $\vec{x} = \myvec{4 \\ 6}$ ,
\begin{align}
    \myvec{4 & 6}\myvec{4 & 0\\0 & 9}\myvec{4 \\ 6} + 2\myvec{-24&-45}\myvec{4 \\ 6}  + 333
\end{align}
\begin{align}
    \implies -11 \leq 0
\end{align}
Therefore, every point in Set A is contained within Set B.\\
This means that A is a subset of B.\\
\bigskip

Therefore, $A\subset B$ \\
\bigskip
\end{frame}
