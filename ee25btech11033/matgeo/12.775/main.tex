\textbf{Question}:\\
Choose solution set $S$ corresponding to the systems of two equations
    \begin{align*}
    x - 2y + z &= 0 \\
    x - z &= 0
    \end{align*}
    \begin{multicols}{2}
    \begin{enumerate}
        \item $S = \left\{ \alpha \myvec{ 1 \\ 1 \\ 1 } \middle| \alpha \in \mathbb{R} \right\}$
        \item $S = \left\{ \alpha \myvec{ 1 \\ 1 \\ 1 } + \beta \myvec{ 1 \\ 0 \\ 1 } \middle| \alpha, \beta \in \mathbb{R} \right\}$
        \item $S = \left\{ \alpha \myvec{ 1 \\ 1 \\ 1 } + \beta \myvec{ 2 \\ 1 \\ 2 } \middle| \alpha, \beta \in \mathbb{R} \right\}$
        \item $S = \left\{ \alpha \myvec{ 1 \\ 0 \\ 1 } \middle| \alpha \in \mathbb{R} \right\}$
    \end{enumerate}
\end{multicols}\\    
\bigskip

\textbf{Solution}:\\
The solution set $S$ corresponds to the points lying on the intersection of two planes.\\

The parameters of the given planes are
\begin{align}
	\label{eq:input}
 \vec{n}_1 = \myvec{1\\-2\\1},\,
 \vec{n}_2 = \myvec{1\\0\\-1},\,
	c_1 = 0, c_2 = 0.
\end{align}
The direction vector of the line of intersection is given by
\begin{align}
    \vec{m} = \vec{n_1} \times \vec{n_2} = \myvec{1\\-2\\1} \times \myvec{1\\0\\-1}
\end{align}
\begin{align}
	\vec{n_1} \times \vec{n_2} 
	 = \myvec{ \mydet{\vec{n_1}__{23} & \vec{n_2}__{23}} \\[1ex] \mydet{\vec{n_1}__{31} & \vec{n_2}__{31}} \\[1ex] \mydet{\vec{n_1}__{12}  & \vec{n_2}__{12}}}
\end{align}

\begin{align}
    \mydet{\vec{n_1}__{23} & \vec{n_2}__{23} }= \mydet{-2 & 0\\1 & -1} = 2
\end{align}

\begin{align}
    \mydet{\vec{n_1}__{31} & \vec{n_2}__{31}}= \mydet{1 & -1\\1 & 1} = 2
\end{align}

\begin{align}
    \mydet{\vec{n_1}__{12}  & \vec{n_2}__{12}} =  \mydet{1 & 1 \\-2 & 0} = 2
\end{align}

\begin{align}
\vec{m} = 2\myvec{1\\1\\1}
\end{align}

The equation of the line is given by,
\begin{align}
    \vec{x} = \vec{h} + \alpha\vec{m}
\end{align}
here
\begin{align}
    \vec{h} = \myvec{0\\0\\0}
\end{align}

\begin{align}
    \implies \vec{x} = \alpha \myvec{1\\1\\1}
\end{align}

The solution set $S$ consists of all points $\vec{x}$ on the line:
\begin{align}
    S = \left\{ \alpha \myvec{ 1 \\ 1 \\ 1 } \middle| \alpha \in \mathbb{R} \right\}
\end{align}
