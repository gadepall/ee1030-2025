\textbf{Question}:\\
Let $ABC$ be a right triangle in which $AB = 6 cm, BC = 8 cm$ and $\angle B = 90\degree$. $BD$ is the perpendicular from $\vec{B}$ on $AC$. The circle through $\vec{B}, \vec{C}, \vec{D}$ is drawn. Construct the tangents from $\vec{A}$ to this circle.\\    
\bigskip

\textbf{Solution}:\\

Let,
\begin{align}
    \vec{B} = \myvec{0\\0} , \vec{A}=\myvec{0\\6} \text{ and } \vec{C} = \myvec{8\\0}
\end{align}

Let the equation of the line passing through $\vec{A}$ and $\vec{C}$ be,
\begin{align}
    \vec{n}^\top\vec{x}=c
\end{align}
The direction vector of line $AC$ is given by,
\begin{align}
\vec{m}=\vec{A}-\vec{C} \equiv \myvec{-4\\3}
\end{align}
Therefore, the normal vector of the desired line is given by,
\begin{align}
    \vec{n}=\myvec{3\\4}
\end{align}
\begin{align}
    c = \vec{n}^\top\vec{A} = \myvec{3&4}\myvec{0\\6} = 24
\end{align}
Let $\vec{D}$ be the foot of the perpendicular from $\vec{B}$ to the line
\begin{align}
    \myvec{\vec{m} & \vec{n}}^\top\vec{D} &= 
	   \myvec{\vec{m}^\top\vec{B}\\c}
\end{align}
\begin{align}
    \myvec{-4 & 3\\3 & 4}\vec{D}=\myvec{\vec{m}^\top\vec{B}\\24}
\end{align}
\begin{align}
    \vec{m}^\top\vec{B} =  \myvec{-4 & 3}\myvec{0 \\ 0} = 0
\end{align}
\begin{align}
    \implies\myvec{-4 & 3\\3 & 4}\vec{D}=\myvec{0\\24}
\end{align}
The augmented matrix  is given by,
\begin{align}
    \augvec{2}{1}{-4 & 3 & 0\\3 & 3 & 24}
\end{align}
\begin{align}
    R_1\rightarrow \frac{-1}{4}R_1\implies\augvec{2}{1}{1 & -3/4 & 0\\3 & 3 & 24}
\end{align}
\begin{align}
    R_2 \rightarrow R_2-3R_1 \implies\augvec{2}{1}{1 & -3/4 & 0\\0 & 25/4 & 24}
\end{align}
\begin{align}
    R_2 \rightarrow \frac{4}{25}R_2\implies\augvec{2}{1}{1 & -3/4 & 0\\0 & 1 & 96/25}
\end{align}
\begin{align}
    R_1 \rightarrow R_1 + \frac{3}{4}R_2\implies\augvec{2}{1}{1 & 0 & 72/25\\0 & 1 & 96/25}
\end{align}
\begin{align}
    \vec{D} = \myvec{72/25 \\ 96/25}
\end{align}
To find the circle through $\vec{B}, \vec{C}, \vec{D}$ the parameters are given by
\begin{align}
\myvec{
 2 \vec{B} & 2 \vec{C} & 2\vec{D}
 \\
 1 & 1 & 1
 }^\top 
	\myvec{\vec{u} \\ f}
	=
-\myvec{\norm{\vec{B}}^2 \\\norm{\vec{C}}^2 \\\norm{\vec{D}}^2 	}
\end{align}
\begin{align}
    \myvec{0 & 0 & 1\\16 & 0 & 1\\144/25 & 192/25 & 1}\myvec{\vec{u} \\ f} = \myvec{0\\-64\\-576/25}
\end{align}
The augmented matrix  is given by,
\begin{align}
    \augvec{3}{1}{0 & 0 & 1 & 0\\16 & 0 & 1 & -64\\144/25 & 192/25 & 1 & -576/25}
\end{align}
\begin{align}
    R_1\leftrightarrow R_2\implies\augvec{3}{1}{16 & 0 & 1 & -64\\0 & 0 & 1 & 0\\144/25 & 192/25 & 1 & -576/25}
\end{align}
\begin{align}
    R_1 \rightarrow \frac{1}{16}R_1 \implies\augvec{3}{1}{1 & 0 & 1/16 & -4\\0 & 0 & 1 & 0\\144/25 & 192/25 & 1 & -576/25}
\end{align}
\begin{align}
    R_3 \rightarrow R_3 - \frac{144}{25}R_1\implies\augvec{3}{1}{1 & 0 & 1/16 & -4\\0 & 0 & 1 & 0\\0 & 192/25 & 16/25 & -504/25}
\end{align}
\begin{align}
    R_2\leftrightarrow R_3\implies\augvec{3}{1}{1 & 0 & 1/16 & -4\\0 & 192/25 & 16/25 & -504/25\\0 & 0 & 1 & 0}
\end{align}
\begin{align}
    R_2 \rightarrow \frac{25}{192}R_2 \implies\augvec{3}{1}{1 & 0 & 1/16 & -4\\0 & 1 & 1/12 & 0\\0 & 0 & 1 & 0}
\end{align}
\begin{align}
    R_2 \rightarrow R_2 - \frac{1}{12}R_3\ ,\ R_1 \rightarrow R_1 - \frac{1}{16}R_3 \implies\augvec{3}{1}{1 & 0 & 0 & -4\\0 & 1 & 0 & 0\\0 & 0 & 1 & 0}
\end{align}

\begin{align}
    \implies\vec{u} = \myvec{-4 \\ 0} \text{ and } f=0
\end{align}\\

Let $\vec{n_1}^\top \vec{x}=c_1$ and $\vec{n_2}^\top \vec{x}=c_2$ be the equation of tangents to the circle.\\
The external point from which the tangents are drawn is $\vec{A}$\\
Compute $\vec{\Sigma}$
\begin{align}
    \vec{\Sigma}=\brak{\vec{V}\vec{h} + \vec{u}}\brak{\vec{V}\vec{h} + \vec{u}}^{\top} - g\brak{\vec{h}}\vec{V}
\end{align}
for $\vec{h} = \vec{A}$
\begin{align}    
    g\brak{\vec{h}}=\vec{h}^{\top}\vec{V}\vec{h} + 2\vec{u}^{\top}\vec{h} + f =\norm{\vec{h}}^2 +f =36+0 =36
\end{align}
\begin{align}
    \vec{\Sigma} = \brak{\vec{V}\vec{h} + \vec{u}}\brak{\vec{V}\vec{h} + \vec{u}}^{\top} - g\brak{\vec{h}}\vec{V}=\myvec{-4 \\ 6}\myvec{-4 & 6} - 36\myvec{ 1& 0 \\ 0 &1}
\end{align}
\begin{align}
    \implies \vec{\Sigma}=\myvec{-20 & -24 \\ -24 & 0}
\end{align}
The eigenvalues of the matrix $\vec{\Sigma}$ are given by
\begin{align}
    \mydet{\vec{\Sigma} - \lambda\vec{I}} = 0
\end{align}
\begin{align}
    \implies \mydet{-20-\lambda & -24 \\ -24 & -\lambda}
\end{align}
\begin{align}
    \implies \brak{-20-\lambda}\brak{-\lambda}-(-24)(-24)=0
\end{align}
\begin{align}
    \implies \lambda^2 + 20 \lambda - 576 = 0
\end{align}
\begin{align}
    \implies \lambda_1 = 16 \text{ and } \lambda_2 = -36
\end{align}
The eigenvectors of the matrix $\vec{\Sigma}$  are,

\begin{align}
     \vec{p}_1 = \myvec{2\\-3} \text{ and } \vec{p}_2 = \myvec{3\\2}
\end{align}
Define,
\begin{align}
    \vec{P} &= \myvec{\frac{\vec{p}_1}{\norm{\vec{p}_1}} & \frac{\vec{p}_2}{\norm{\vec{p}_2}}} = \myvec{\frac{2}{\sqrt{13}} & \frac{3}{\sqrt{13}}\\\frac{-3}{\sqrt{13}} & \frac{2}{\sqrt{13}}}
\end{align}
The direction vectors of the two tangents are,
\begin{align}
    \vec{m}=\vec{P}\myvec{\sqrt{\abs{\lambda_2}} \\ \pm\sqrt{\abs{\lambda_1}}}=\myvec{\frac{2}{\sqrt{13}} & \frac{3}{\sqrt{13}}\\\frac{-3}{\sqrt{13}} & \frac{2}{\sqrt{13}}}\myvec{6 \\ \pm 4}
\end{align}
\begin{align}
\implies \vec{m}_1 = \myvec{0 \\ 1} \text{ and } \vec{m}_2 = \myvec{12 \\ -5}
\end{align}

\begin{align}
\implies \vec{n}_1 = \myvec{1 \\ 0} \text{ and } \vec{n}_2 = \myvec{5 \\ 12}
\end{align}
Also,
\begin{align}
    c_1 = \vec{n_1}^\top \vec{A} \text{ and } c_2 = \vec{n_2}^\top \vec{A}
\end{align}

\begin{align}
    c_1 = 0 \text{ and } c_2 = 72
\end{align}\\
\bigskip 

Therefore, the line $\myvec{5 & 12}\vec{x} = 72$ and the $\myvec{1 & 0}\vec{x} = 0$ are the two tangents.\\
