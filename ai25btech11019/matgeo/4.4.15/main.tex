\textbf{Question}:\\
Find the sum of the intercepts cut off by the plane  
$-\vec{r}\cdot(2\hat{i}+\hat{j}-\hat{k})-5=0$ on the three coordinate axes.
\\
\bigskip
\textbf{Solution: }

The given plane is
\begin{align}
-\vec{r}\cdot\myvec{2\\1\\-1}-5=0 
&\implies \vec{r}\cdot\myvec{2\\1\\-1}+5=0
\end{align}

\textbf{X-intercept:} $(x,0,0)$  
\begin{align}
\vec{r}=\myvec{x\\0\\0},\quad 
\vec{r}\cdot\myvec{2\\1\\-1}+5=0 
&\implies 2x+5=0 \\
x&=-\frac{5}{2}
\end{align}

\textbf{Y-intercept:} $(0,y,0)$  
\begin{align}
\vec{r}=\myvec{0\\y\\0},\quad 
\vec{r}\cdot\myvec{2\\1\\-1}+5=0 
&\implies y+5=0 \\
y&=-5
\end{align}

\textbf{Z-intercept:} $(0,0,z)$  
\begin{align}
\vec{r}=\myvec{0\\0\\z},\quad 
\vec{r}\cdot\myvec{2\\1\\-1}+5=0 
&\implies -z+5=0 \\
z&=5
\end{align}

\vspace{1cm}

\noindent\textbf{Conclusion:}  
The intercepts cut off are
\begin{align}
x=-\frac{5}{2},\quad
y=-5,\quad
z=5.
\end{align}

Hence, the sum of intercepts is  
\begin{align}
-\frac{5}{2} - 5 + 5 \;=\; -\frac{5}{2}.
\end{align}
