\textbf{Question :}The area between $x = y^2$ and $x = 4$ is divided into two equal parts by the line $x = a$, find the value of $a$.

\textbf{Solution :}
\begin{table}[h!]
  \centering

  \label{9.2.20}
\end{table}

On comparing Parabola  with
\begin{align}
\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f &= 0 
\end{align}
The parameters of the Parabola are 
\begin{align}
\vec{V}=\myvec{0&0\\0&1} , \vec{u}=\frac{-1}{2}\myvec{1\\0} , f=0
\end{align}
For the line $x-a = 0$, the parameters are 
\begin{align}
\vec{h_1}=\myvec{a\\0} , \vec{m_1}=\myvec{0\\1}
\end{align}
For the line $x-4 = 0$, the parameters are
\begin{align}
\vec{h_2}=\myvec{4\\0} , \vec{m_2}=\myvec{0\\1}
\end{align}
The points of intersection for a conic and a line is given by
\begin{align}
\kappa_i
  &= \frac{1}{\vec{m}^\top \vec{V}\vec{m}}
     \left(
       -\,\vec{m}^\top\big(\vec{V}\vec{h}+\vec{u}\big)
       \;\pm\;
       \sqrt{ \big[\vec{m}^\top(\vec{V}\vec{h}+\vec{u})\big]^2
       - g(\vec{h})\,\big(\vec{m}^\top \vec{V}\vec{m}\big)}
     \right)
\end{align}
\begin{align}
g(\vec{x}) = \vec{x}^\top\myvec{0&0\\0&1}\vec{x} - \myvec{1 & 0}\vec{x} \\
g(\vec{h}) = \vec{h}^\top\myvec{0&0\\0&1}\vec{h} - \myvec{1 & 0}\vec{h}
\end{align}
\begin{align}
\kappa_i &=\left(
            0
       \;\pm\;
       \sqrt{\myvec{1 & 0}\vec{h} - \vec{h}^\top\myvec{0&0\\0&1}\vec{h}}
     \right)
\end{align}
So
\begin{align}
\kappa_1 = \sqrt{a} , -\sqrt{a} \ and \ \kappa_2 = 2 , -2
\end{align}
Therefore the points of intersection are 

For line 1
\begin{align}
\vec{P_1} = \myvec{a \\ \sqrt{a}} , \vec{P_2} = \myvec{a \\ -\sqrt{a}}
\end{align}
For line 2
\begin{align}
\vec{P_3} = \myvec{4 \\ 2} , \vec{P_4} = \myvec{4 \\ -2}
\end{align}
The area of the parabola in between the lines $x = a$ and $x = 4$ is given by
\begin{align}
2 \left(\int\limits_{0}^{4}\! \sqrt{x}\, dx - \int\limits_{0}^{a}\! \sqrt{x}\, dx \right) = \frac{4}{3}\left(8 - a^\frac{3}{2}\right)
\end{align}
The area between parabola and line $x = a$ is given by
\begin{align}
2\left(\int\limits_{0}^{a}\! \sqrt{x}\, dx \right) = \frac{4}{3}a^\frac{3}{2}
\end{align}
Given both the areas are equal
\begin{align}
\frac{4}{3}\left(8 - a^\frac{3}{2}\right) = \frac{4}{3}a^\frac{3}{2} \\
4 = a^\frac{3}{2} \\
a = 4^\frac{2}{3} = 16^{\tfrac{1}{3}}
\end{align}
