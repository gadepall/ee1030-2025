\subsection{Equating Area and Finding a}
\begin{frame}
\frametitle{Equating Area and Finding a}
%\framesubtitle{Literature}
On comparing Parabola  with
\begin{align}
\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f &= 0 
\end{align}
The parameters of the Parabola are 
\begin{align}
\vec{V}=\myvec{0&0\\0&1} , \vec{u}=\frac{-1}{2}\myvec{1\\0} , f=0
\end{align}
For the line $x-a = 0$, the parameters are 
\begin{align}
\vec{h_1}=\myvec{a\\0} , \vec{m_1}=\myvec{0\\1}
\end{align}
For the line $x-4 = 0$, the parameters are
\begin{align}
\vec{h_2}=\myvec{4\\0} , \vec{m_2}=\myvec{0\\1}
\end{align}
The points of intersection for a conic and a line is given by
\begin{align}
\kappa_i
  &= \frac{1}{\vec{m}^\top \vec{V}\vec{m}}
     \left(
       -\,\vec{m}^\top\big(\vec{V}\vec{h}+\vec{u}\big)
       \;\pm\;
       \sqrt{ \big[\vec{m}^\top(\vec{V}\vec{h}+\vec{u})\big]^2
       - g(\vec{h})\,\big(\vec{m}^\top \vec{V}\vec{m}\big)}
     \right)
\end{align}
\end{frame}
\begin{frame}
\begin{align}
g(\vec{x}) = \vec{x}^\top\myvec{0&0\\0&1}\vec{x} - \myvec{1 & 0}\vec{x} \\
g(\vec{h}) = \vec{h}^\top\myvec{0&0\\0&1}\vec{h} - \myvec{1 & 0}\vec{h}
\end{align}
\begin{align}
\kappa_i &=\left(
            0
       \;\pm\;
       \sqrt{\myvec{1 & 0}\vec{h} - \vec{h}^\top\myvec{0&0\\0&1}\vec{h}}
     \right)
\end{align}
\begin{align}
\kappa_1 = \sqrt{a} , -\sqrt{a} \ and \ \kappa_2 = 2 , -2
\end{align}
Therefore the points of intersection are 

For line 1
\begin{align}
\vec{P_1} = \myvec{a \\ \sqrt{a}} , \vec{P_2} = \myvec{a \\ -\sqrt{a}}
\end{align}
For line 2
\begin{align}
\vec{P_3} = \myvec{4 \\ 2} , \vec{P_4} = \myvec{4 \\ -2}
\end{align}
\end{frame}
\begin{frame}
The area of the parabola in between the lines $x = a$ and $x = 4$ is given by
\begin{align}
2 \left(\int\limits_{0}^{4}\! \sqrt{x}\, dx - \int\limits_{0}^{a}\! \sqrt{x}\, dx \right) = \frac{4}{3}\left(8 - a^\frac{3}{2}\right)
\end{align}
The area between parabola and line $x = a$ is given by
\begin{align}
2\left(\int\limits_{0}^{a}\! \sqrt{x}\, dx \right) = \frac{4}{3}a^\frac{3}{2}
\end{align}
Given both the areas are equal
\begin{align}
\frac{4}{3}\left(8 - a^\frac{3}{2}\right) = \frac{4}{3}a^\frac{3}{2} \\
4 = a^\frac{3}{2} \\
a = 4^\frac{2}{3} = 16^{\frac{1}{3}}
\end{align}

\end{frame}

#include <stdio.h>
#include <math.h>

int main() {
    double total_area, half_area, a, lhs;

    // Step 1: Find total area between x = y^2 and x = 4
    // Limits are y = -2 to y = 2
    total_area = (4*2 - pow(2,3)/3) - (4*(-2) - pow(-2,3)/3);
    // Simplify the above: total_area = 32/3
    printf("Total area between x = y^2 and x = 4 : %.6lf\n", total_area);

    // Step 2: Half of the total area
    half_area = total_area / 2.0;
    printf("Half of total area : %.6lf\n", half_area);

    // Step 3: For x = a dividing the region into equal parts:
    \end{lstlisting}
    \end{frame}
    \begin{frame}[fragile]
    \begin{lstlisting}
    // Area between x = y^2 and x = a is (4/3) * a^(3/2)
    // We set (4/3)*a^(3/2) = half_area
    // => a^(3/2) = (3/4)*half_area
    double rhs = (3.0/4.0) * half_area;

    // Now find a = (rhs)^(2/3)
    a = pow(rhs, 2.0/3.0);

    // Step 4: Display results
    printf("Equation used: (4/3)*a^(3/2) = %.6lf\n", half_area);
    printf("=> a^(3/2) = %.6lf\n", rhs);
    printf("=> a = (%.6lf)^(2/3)\n", rhs);
    printf("Value of a = %.6lf\n", a);

    return 0;
}
