\section{Problem}
\begin{frame}
\frametitle{Problem Statement}
\textbf{Question} : Let P be the plane 3x+2y+3z=16 and let S: $\alpha\hat{i} + \beta\hat{j} + \gamma\hat{k} $ ,where $\alpha+\beta+\gamma=7$ and the distance of $(\alpha,\beta,\gamma)$ from the plane is $2/{\sqrt{22}}$ .Let $\vec{u},\vec{v},\vec{w}$ be three distinct vectors in S such that $|\vec{u}-\vec{v}|=|\vec{v}-\vec{w}|=|\vec{w}-\vec{u}|$. Let V be the volume of the parallelopiped determined by vectors $\vec{u},\vec{v},\vec{w}$. Then the value of (80/3)V is 

\end{frame}
\section{Solution}
\subsection{Distance between point and plane}
\begin{frame}
\frametitle{Distance between point and plane}
%\framesubtitle{Literature}
\textbf{Solution} :
\begin{align}
P:\; \vec{n}^\top \vec{x} = c,\qquad \vec{n}=\myvec {3\\2\\3}\;c=16.
\end{align}

The distance of point $\vec{P_0}=\myvec{\alpha\\ \beta\\ \gamma}$ from the plane is
\begin{align}
\operatorname{dist}(\vec{P_0},P)=\frac{|\vec{n}^\top \vec{P_0} - c|}{\norm{\vec{n}}}
\end{align}
Given $\alpha+\beta+\gamma=7$ and $\operatorname{dist}(\vec{P_0},P)=\tfrac{2}{\sqrt{22}}$, we have
\begin{align}
\frac{|3\alpha+2\beta+3\gamma-16|}{\sqrt{22}}=\frac{2}{\sqrt{22}}
\;\;\Longrightarrow\;\; \vec{n}^\top \vec{P_0}=18 \;\;\text{or}\;\; \vec{n}^\top \vec{P_0}=14.
\end{align}

Thus $S$ lies on the intersections
\end{frame}
\begin{frame}
\begin{align}
\Pi:\; \vec{m}^\top \vec{x}=7,\qquad \vec{m}=\myvec{1\\1\\1},
\end{align}
with
\begin{align}
P_+: \vec{n}^\top \vec{x}=18,\qquad P_-:\vec{n}^\top \vec{x}=14.
\end{align}

Each intersection is a line.  
The direction vector of the line is
\begin{align}
\vec{d} = \vec{m}\times \vec{n} =
\myvec{
\mydet
{m_{23} & n_{23}} \\
 \mydet
 {m_{31} & n_{31}} \\
\mydet
{m_{12} & n_{12}}
}
=
\myvec{1\\0\\-1}.
\end{align}

\end{frame}
\subsection{Equation of line and finding u,v,w}
\begin{frame}
\frametitle{Equation of line and finding u,v,w}
Pick points:  
For $\Pi\cap P_+$, solve
\begin{align}
\begin{cases}
\vec{m}^\top \vec{x} = 7,\\
\vec{n}^\top \vec{x} = 18
\end{cases}
\quad\Longrightarrow\quad  \vec{p} = \myvec{0\\3\\4}
\end{align}
So
\begin{align}
L_+:\; \vec{r}=\vec{p} + t\vec{d}.
\end{align}

For $\Pi\cap P_-$, solve
\begin{align}
\begin{cases}
\vec{m}^\top \vec{x}= 7,\\
\vec{n}^\top \vec{x}= 14
\end{cases}
\quad\Longrightarrow\quad  \vec{q} = \myvec{0\\7\\0}
\end{align}
So
\begin{align}
L_-:\; \vec{r}=\vec{q} +t\vec{d}.
\end{align}

\textbf{Construct equilateral triangle:}
\end{frame}
\begin{frame}
Take
\begin{align}
\vec{u}=\vec{p}+s\vec{d},\quad \vec{v}=\vec{p}-s\vec{d},\quad \vec{w}=\vec{q}+t\vec{d}.
\end{align}

\begin{align}
{\norm{\vec{u}-\vec{v}}}^2={\norm{\vec{u}-\vec{w}}}^2={\norm{\vec{v}-\vec{w}}}^2.
\end{align}

\begin{align}
\vec{u}-\vec{v}=2s\vec{d} \;\;\Rightarrow\;\; {\norm{\vec{u}-\vec{v}}}^2=4s^2{\norm{\vec{d}}}^2.
\end{align}

\begin{align}
{\norm{\vec{d}}}^2=\vec{d}^\top\vec{d}=1^2+0^2+(-1)^2=2,
{\norm{\vec{u}-\vec{v}}}^2=8s^2.
\end{align}

\begin{align}
\vec{r}=\vec{p}-\vec{q}=\myvec{0\\-4\\4},\quad
\vec{u}-\vec{w}=\vec{r}+(s-t)\vec{d}.
\end{align}

\begin{align}
{\norm{\vec{u}-\vec{w}}}^2={\norm{\vec{r}}}^2+2(s-t)(\vec{r}^\top\vec{d})
+(s-t)^2{\norm{\vec{d}}}^2.
\end{align}
\end{frame}
\begin{frame}
\begin{align}
{\norm{\vec{r}}}^2=32,\quad \vec{r}^\top\vec{d}=-4,\quad {\norm{\vec{d}}}^2=2.
\end{align}

\begin{align}
{\norm{\vec{u}-\vec{w}}}^2=32-8(s-t)+2(s-t)^2.
\end{align}

\begin{align}
\vec{v}-\vec{w}=\vec{r}+(-s-t)\vec{d} \;\;\Rightarrow\;\;
{\norm{\vec{v}-\vec{w}}}^2=32+8(s+t)+2(s+t)^2.
\end{align}
Equating
\begin{align}
32-8(s-t)+2(s-t)^2=8s^2, \quad
32+8(s+t)+2(s+t)^2=8s^2.
\end{align}
\begin{align}
16s+8st=0 \quad\Rightarrow\quad t=-2,\; s^2=4.
\end{align}
Take $ s=2,t=-2:$
\begin{align}
\vec{u}=\vec{p}+2\vec{d}=\myvec {2\\3\\2},\quad
\vec{v}=\vec{p}-2\vec{d}=\myvec {-2\\3\\6},\quad
\vec{w}=\vec{q}-2\vec{d}=\myvec{-2\\7\\2}.
\end{align}

\end{frame}
\subsection{Gram matrix and Volume}
\begin{frame}
\frametitle{Gram matrix and Volume}

\textbf{Volume of parallelepiped:}

\textbf{Gram matrix }\(G\) (matrix of pairwise inner products):
\begin{align}
G=\myvec
{{\norm{\vec{u}}}^2 & \vec{u}^\top\vec{v} & \vec{u}^\top\vec{w}\\[6pt]
\vec{v}^\top\vec{u} & {\norm{\vec{v}}}^2 & \vec{v}^\top\vec{w}\\[6pt]
\vec{w}^\top\vec{u} & \vec{w}^\top\vec{v} & {\norm{\vec{w}}}^2}
=
\myvec
{17 & 17 & 21\\[4pt]
17 & 49 & 37\\[4pt]
21 & 37 & 57},
\end{align}
\textbf{Determinant of the Gram matrix (gives the squared volume):}
\begin{align}
\det G =det \myvec
{17 & 17 & 21\\[4pt]
17 & 49 & 37\\[4pt]
21 & 37 & 57} = 12544
\end{align}
Hence the volume itself is
\begin{align}
V=\sqrt{\det G}=\sqrt{12544}=112.
\end{align}

\begin{align}
\frac{80V}{3}= \frac{80x112}{3} = \frac{8960}{3}.
\end{align}

\end{frame}

// Function to compute determinant of 3x3 matrix
double determinant(double a[3][3]) {
    return a[0][0]*(a[1][1]*a[2][2] - a[1][2]*a[2][1])
         - a[0][1]*(a[1][0]*a[2][2] - a[1][2]*a[2][0])
         + a[0][2]*(a[1][0]*a[2][1] - a[1][1]*a[2][0]);}
int main() {
    // Two planes:
    // Plane P: 3x+2y+3z=16
    // Plane S: x+y+z=7
    // Solve intersection line between P and S
    // Choose basis for S: let
    // a = (1,-1,0), b = (1,0,-1) → both satisfy x+y+z=0
    \end{lstlisting}

    // So general point on S: (7,0,0) + s*a + t*b
    // Constraint from P: 3x+2y+3z = 14 or 18 (distance condition)
    
    // Pick "14" first.
    // Solve quickly: Intersection point
    double u[3] = {2,3,2};
    double v[3] = {-2,3,6};
    double w[3] = {-2,7,2};
    
    // Put in matrix
    double M[3][3] = {
        {u[0], v[0], w[0]},
        {u[1], v[1], w[1]},
        {u[2], v[2], w[2]}};
    // Determinant
    double det = determinant(M);
    double V = fabs(det);
    double result = (80.0/3.0) * V;
