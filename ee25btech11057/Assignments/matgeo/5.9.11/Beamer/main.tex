\section{Problem}
\begin{frame}
\frametitle{Problem Statement}
\textbf{Question} : 
Sumit is 3 times as old as his son. Five years later, he shall be two and a half times as old as his son. How old is Sumit at present ?
\end{frame}
\section{Solution}
\subsection{ Intersection of lines and Age}
\begin{frame}
\frametitle{Intersection of lines and Age}
%\framesubtitle{Literature}
\textbf{Solution:}

 Let Age of son = x
 
Age of Father = y =3x
\begin{align}
3x-y=0
\end{align}
Five years later, father shall be two and a half times as old as his son, So
\begin{align}
y+5 = \frac{5}{2}(x+5)
\end{align}
\begin{align}
5x-2y = -15
\end{align}
The system of equations formed is :
\begin{align}
3x-y = 0
\end{align}
\begin{align}
5x-2y = -15
\end{align}
Writing it in the matrix form,
\end{frame}
\begin{frame}
\begin{align}
  \myvec{3 & -1\\5 & -2}\myvec{x\\y} &= \myvec{0\\-15}
\end{align}

Forming the augmented matrix to solve the system of equations,

\begin{align}
  \myaugvec{2}{3 & -1 & 0\\5 & -2 & -15}
\end{align}

Using Gaussian Elimination,

\begin{align}
 \myaugvec{2}{3 & -1 & 0\\5 & -2 & -15} 
 \xleftrightarrow{\;R_2 \to 3R_2 -5R_1}
 \myaugvec{2}{3 & -1 & 0\\0 & -1 & -45} 
\end{align}

Using back substitution we get,

\begin{align}
  -y = -45
  \end{align}
  \begin{align}
  y = 45
  \end{align}
  \end{frame}
  \begin{frame}
\begin{align}
3x-y = 0
\end{align}
\begin{align}
  3x = 45
  \end{align}
  \begin{align}
  x = 15
\end{align}


The solution for the system of equations is :

\begin{align}
  \myvec{x\\y} &= \myvec{15\\45}
\end{align}

Therefore the present age of father = y = 45

\end{frame}

int main() {
    float son, sumit;

    // Problem:
    // Sumit is 3 times as old as his son
    // => sumit = 3 * son
    //
    // After 5 years:
    // sumit + 5 = 2.5 * (son + 5)

    // Substituting sumit = 3 * son:
    // 3*son + 5 = 2.5*(son + 5)
    // 3*son + 5 = 2.5*son + 12.5
    // 0.5*son = 7.5
    // son = 15
    // sumit = 45

x, y = solution
