\section{Problem}
\begin{frame}
\frametitle{Problem Statement}
\textbf{Question} : 
A rectangle $PQRS$ has its side $\vec{P}\vec{Q}$ parallel to the line $y = mx$ and vertices $\vec{P}, \vec{Q} \ and \ \vec{S}$ on the lines $y = a, x = b \ and \ x = -b $ ,respectively. Find the locus of the vertex $\vec{R}$.


\end{frame}
\section{Solution}
\subsection{ Dot Product  and Locus}
\begin{frame}
\frametitle{Dot Product and Locus}
%\framesubtitle{Literature}
\textbf{Solution:}

Given,
$\vec{P}, \vec{Q} \ and \ \vec{S}$ lie on the lines $y = a, x = b \ and \ x = -b .$
\begin{align}
\vec{n_1}^\top\vec{P}=a , \vec{n_1}=\myvec{0\\1}
\end{align}
\begin{align}
\vec{n_2}^\top\vec{Q}=b , \vec{n_2}^\top\vec{S}=-b , \vec{n_2}=\myvec{1\\0}
\end{align}
\begin{align}
\vec{m}=\myvec{1\\m} , \vec{n}=\myvec{-m\\1} , \vec{m} \perp \vec{n}
\end{align}
As $\vec{P}\vec{Q}$ is parallel to $\vec{m}$ and $\vec{P}\vec{S}$ is parallel to $\vec{n}$
\begin{align}
\vec{Q}-\vec{P}=\vec{R}-\vec{S}=k_1\vec{m} , \vec{S}-\vec{P}=k_2\vec{n}
\end{align}
\begin{align}
\vec{n_2}^\top\vec{m}=1 , \vec{n_2}^\top\vec{n}=-m 
\end{align}
\begin{align}
\vec{n_1}^\top\vec{m}=m , \vec{n_1}^\top\vec{n}=1
\end{align}
\end{frame}
\begin{frame}
\begin{align}
\vec{Q}=\vec{P}+k_1\vec{m} \Longrightarrow
\vec{n_2}^\top\vec{Q}=\vec{n_2}^\top\vec{P}+k_1\vec{n_2}^\top\vec{m}=b
\end{align}
\begin{align}
\vec{S}=\vec{P}+k_2\vec{n} \Longrightarrow 
\vec{n_2}^\top\vec{S}=\vec{n_2}^\top\vec{P}+k_2\vec{n_2}^\top\vec{n}=-b
\end{align}
\begin{align}
k_1+mk_2=2b
\end{align}
\begin{align}
\vec{R}=\vec{P}+k_2\vec{n}+k_1\vec{m} \Longrightarrow
\vec{n_1}^\top\vec{R}=\vec{n_1}^\top\vec{P}+k_2\vec{n_1}^\top\vec{n}+k_1\vec{n_1}^\top\vec{m}
\end{align}
\begin{align}
\vec{n_1}^\top\vec{R}=a+k_2+k_1m
\end{align}
\begin{align}
\vec{R}=\vec{P}+k_2\vec{n}+k_1\vec{m} \Longrightarrow
\vec{n_2}^\top\vec{R}=\vec{n_2}^\top\vec{P}+k_2\vec{n_2}^\top\vec{n}+k_1\vec{n_2}^\top\vec{m}
\end{align}
\begin{align}
\vec{n_2}^\top\vec{P}+k_1\vec{n_2}^\top\vec{m}=b
\end{align}
\begin{align}
\vec{n_2}^\top\vec{R}=b-mk_2 \Longrightarrow k_2=\frac{b-\vec{n_2}^\top\vec{R}}{m}
\end{align}
\begin{align}
k_1+mk_2=2b \Longrightarrow k_1+b-\vec{n_2}^\top\vec{R}=2b
\end{align}
\end{frame}
\begin{frame}
\begin{align}
k_1=b+\vec{n_2}^\top\vec{R}
\end{align}
\begin{align}
\vec{n_1}^\top\vec{R}=a+k_2+k_1m \Longrightarrow \vec{n_1}^\top\vec{R}=a+\frac{b-\vec{n_2}^\top\vec{R}}{m}+mb+m\vec{n_2}^\top\vec{R}
\end{align}
\begin{align}
\vec{n_1}^\top\vec{R}=a+b\left( m + \frac{1}{m} \right)
+\vec{n_2}^\top\vec{R}\left( m - \frac{1}{m} \right)
\end{align}
\begin{align}
\vec{n_2}^\top\vec{R}\left(\frac{1}{m} - m \right)+\vec{n_1}^\top\vec{R}=a+b\left( m + \frac{1}{m} \right)
\end{align}
\begin{align}
\myvec{ \frac{1}{m} - m & 1}\myvec{\vec{n_2}^\top\vec{R} \\ \vec{n_1}^\top\vec{R}}=a+b\left( m + \frac{1}{m} \right)
\end{align}
\begin{align}
\vec{R}=\myvec{\vec{n_2}^\top\vec{R} \\ \vec{n_1}^\top\vec{R}}
\end{align}
\end{frame}
\begin{frame}
Let
\begin{align}
c = a+b\left( m + \frac{1}{m} \right) \ and \ \vec{n_3}=\myvec{\frac{1}{m} - m \\ 1}
\end{align}
The locus of the vertex $\vec{R}$ is
\begin{align}
\vec{n_3}^\top\vec{R}=c
\end{align}

\end{frame}

   #include <stdio.h>
/*
   A rectangle PQRS has its side PQ parallel to the line y = mx
   and vertices P, Q, S on lines y=a, x=b, x=-b respectively.
   Find the locus of vertex R.
   Result: y = m*x - (a*(1+m^2))/m,  with x = ±b
*/

double find_locus(double m, double a, double x) {
    return m * x - (a * (1 + m * m)) / m;}

int main() {
    double m, a, b;

    printf("Enter slope m: ");
    scanf("%lf", &m);
    \end{lstlisting}

    // For x = +b
    double y1 = find_locus(m, a, b);
    printf("For x = %.2f, R = (%.2f, %.2f)\n", b, b, y1);

    // For x = -b
    double y2 = find_locus(m, a, -b);
    printf("For x = %.2f, R = (%.2f, %.2f)\n", -b, -b, y2);

    return 0;
}

import ctypes

# Load the compiled shared object
# Make sure rectangle_locus.so is in the same directory
lib = ctypes.CDLL('./rectangle_locus.so')
# Define the argument and return types of the C function
# double find_locus(double m, double a, double x)
lib.find_locus.argtypes = [ctypes.c_double, ctypes.c_double, ctypes.c_double]
lib.find_locus.restype = ctypes.c_double

def call_find_locus(m, a, b):
    results = [ ]
    for x in [b, -b]:
        y = lib.find_locus(m, a, x)
        results.append((x, y))
    return results
