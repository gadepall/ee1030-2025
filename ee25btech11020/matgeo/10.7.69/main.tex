Question:\\

Let $x^2+y^2-4x-2y-11=0$ be a circle. A pair of tangents from the point $\brak{4,5}$ with a pair of radii form a quadrilateral with area\rule{1cm}{0.4pt}.


\solution
\begin{align}
\text{Conic: }\quad \vec{x}^\top V\vec{x}+2\vec{u}^\top\vec{x}+f=0
\end{align}

\begin{align}
V=\myvec{1 & 0\\0 & 1},\quad \vec{u}=\myvec{-2\\-1},\quad f=-11
\end{align}
Equation of Tangent,
\begin{align}
	\brak{\vec{V}\vec{q}+\vec{u}}^\top\vec{x}+\vec{u}^\top\vec{q}+f=0
\end{align}
Point $\vec{a}=\myvec{4\\5}$ satisfies the given equation,
\begin{align}
	\brak{\vec{V}\vec{q}+\vec{u}}^\top\vec{a}+\vec{u}^\top\vec{q}+f=0
\end{align}
