\begin{frame}{Question}
Find the equation of the conic, that satisfies the given conditions.
vertex (-3, 0), directrix x + 5 = 0.
\end{frame}


\begin{frame}[fragile]
    \frametitle{Theoretical Solution}
Given conditions:\\
Vertex: $(-3, 0)$\\
Directrix: $x + 5 = 0$\\

$\Rightarrow$ Since we have a vertex and directrix, this conic is a parabola with eccentricity $e = 1$.\\

Using the formulation for a conic with directrix $\vec{n}^T\vec{x} = c$, eccentricity $e$, and focus $\vec{F}$:\\

Given Parameters:

Normal vector: $\vec{n} = \myvec{1\\0}$

Directrix constant: $c = -5$

Eccentricity: $e = 1$
\end{frame}


\begin{frame}[fragile]
    \frametitle{Theoretical Solution}
 
Focus: $\vec{F} = \myvec{-1\\0}$ [derived from equidistance property]\\

Calculating matrix $\vec{V}$
\begin{equation}
\vec{V} = \norm{\vec{n}}^2\vec{I} - e^2(\vec{n}\vec{n}^T) = \vec{I} -\myvec{1\\0}\myvec{1 & 0} = \myvec{0 & 0\\0 & 1}
\end{equation}

Calculating $\vec{u}$
\begin{equation}
\vec{u} = ce^2\vec{n} - \norm{\vec{n}}^2\vec{F} = (-5)(1)\myvec{1\\0} - 1\myvec{-1\\0} = \myvec{-4\\0}
\end{equation}

Calculating f:
\begin{equation}
f = \norm{\vec{n}}^2\norm{\vec{F}}^2 - c^2e^2 = 1 - 25 = -24
\end{equation}
\end{frame}


\begin{frame}[fragile]
    \frametitle{Theoretical Solution}
\begin{equation}
\text{The eigenvalues of } \vec{V} \text{ is } \lambda_1 = 0
\end{equation}

Since the eigenvalue is zero, we confirm this is a parabola $(e = 1)$.

The general conic equation is:
\begin{equation}
\vec{x}^T\vec{V}\vec{x} + 2\vec{u}^T\vec{x} + f = 0
\end{equation}
\end{frame}


\begin{frame}[fragile]
    \frametitle{Theoretical Solution}
Substituting our values in Equation 5:
\begin{align}
\myvec{x & y}\myvec{0 & 0\\0 & 1}\myvec{x\\y} + 2\myvec{-4 & 0}\myvec{x\\y} + (-24) &= 0
\end{align}

We Get,
\begin{equation}
    \boxed{y^2 - 8x - 24 = 0}
\end{equation}\\\\

Therefore, the Equation of the Conic is:
\begin{equation}
\centering
\boxed{y^2 = 8(x + 3)}
\end{equation}
\end{frame}
