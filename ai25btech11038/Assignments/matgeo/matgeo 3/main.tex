\textbf{Question:}

\noindent The angle between the vectors $\hat{i} - \hat{j}$ and $\hat{j} - \hat{k}$ is

\textbf{Solution:}

\noindent The cosine of the angle between two vectors $\mathbf{n_1}$ and $\mathbf{n_2}$

\begin{equation}
    cos\theta = \frac{\mathbf{n_1}^T \mathbf{n_2}}{||\mathbf{n_1}|| \cdot ||\mathbf{n_2}||}
\end{equation}

\noindent Here, 

\begin{equation}
    \mathbf{n_1} = \myvec{1 \\ -1 \\ 0}, \mathbf{n_2} = \myvec{0 \\ 1 \\ -1}
\end{equation}

\noindent For the given two vectors, the value of their norms is $\sqrt{2}$ and 

\begin{equation}
    \mathbf{n_1}^T \mathbf{n_2} = \myvec{1 -1 0} \myvec{0 \\ 1 \\ -1}
\end{equation}

\begin{equation}
    \mathbf{n_1}^T \mathbf{n_2} = -1
\end{equation}

\begin{equation}
    cos\theta = \frac{-1}{(\sqrt{2})^2} = \frac{-1}{2}
\end{equation}

\noindent Therefore the angle $\theta$ between the given vectors can be
$$\theta = \frac{\pi}{3} or \frac{2\pi}{3}$$
