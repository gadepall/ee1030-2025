\textbf{Question:}

\noindent Find the area of the triangle whose vertices are $\brak{-1,1}$, $\brak{0, 5}$, and $\brak{3, 2}$

\textbf{Solution:}

\noindent The area of the triangle formed by three point vectors $\mathbf{A}$, $\mathbf{B}$ and $\mathbf{C}$ is 

\begin{equation}
    area\brak{ABC} = \frac{||(\mathbf{B} - \mathbf{A}) \times (\mathbf{C} - \mathbf{A})||}{2}\
\end{equation}

\noindent Here, 

\begin{equation}
    \mathbf{A} = \myvec{-1 \\ 1}, \mathbf{B} = \myvec{0 \\ 5}, \mathbf{C} = \myvec{3 \\ 2} 
\end{equation}

\begin{equation}
    (\mathbf{B} - \mathbf{A}) = \myvec{1 \\ 4}, (\mathbf{C} - \mathbf{A}) = \myvec{4 \\ 1}
\end{equation}

\begin{equation}
    ||(\mathbf{B} - \mathbf{A}) \times (\mathbf{C} - \mathbf{A})|| = 17
\end{equation}

\noindent Therefore,

\begin{equation}
    area\brak{ABC} = \frac{17}{2}
\end{equation}
