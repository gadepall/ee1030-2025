\textbf{Question:}

\noindent The scalar $\textbf{A}\cdot\brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}}$ equals

\begin{multicols}{2}
\begin{enumerate}
    \item $0$
    \item $[\textbf{A} \textbf{B} \textbf{C}]$
    \item $[\textbf{A} \textbf{B} \textbf{C}] + [\textbf{B} \textbf{C} \textbf{A}]$
    \item None of these
\end{enumerate}
\end{multicols}

\textbf{Solution:}

\noindent Using the \textbf{distributive property} of cross product over addition $\brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}}$ can be expanded to,
\begin{equation}
    \brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = \brak{\brak{\textbf{B} + \textbf{C}}\times\textbf{A}} + \brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{B} + \textbf{C}}}
\end{equation}

\noindent The cross product of two parallel vectors is a zero vector,

\begin{equation}
    \brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = \brak{\brak{\textbf{B} + \textbf{C}}\times\textbf{A}}
\end{equation}

\begin{equation}
    \brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = \brak{\textbf{B}\times\textbf{A}} + \brak{\textbf{C}\times\textbf{A}}
\end{equation}

\noindent Now, 

\begin{equation}
    \textbf{A}\cdot\brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = \textbf{A}\cdot\brak{\brak{\textbf{B}\times\textbf{A}} + \brak{\textbf{C}\times\textbf{A}}}
\end{equation}

\begin{equation}
    \textbf{A}\cdot\brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = \brak{\textbf{A}\cdot\brak{\textbf{B}\times\textbf{A}}} + \brak{\textbf{A}\cdot\brak{\textbf{C}\times\textbf{A}}}
\end{equation}

\noindent Since the two vectors \textbf{A} and $\textbf{B}\times\textbf{A}$ are perpendicular and so are \textbf{A} and $\textbf{C}\times\textbf{A}$, their dot product is zero

\begin{equation}
    \textbf{A}\cdot\brak{\brak{\textbf{B} + \textbf{C}}\times\brak{\textbf{A} + \textbf{B} + \textbf{C}}} = 0
\end{equation}
