\textbf{Question:}\\
The distance between the points $\brak{0,2\sqrt{5}}$ and $\brak{-2\sqrt{5},0}$ is \underline{\hspace{2cm}}.

\textbf{Solution:}\\
The input parameters for this problem are available in Table~\ref{tab:1}.

\begin{table}[h!]
\centering
\begin{tabular}{|c|c|c|}
\hline
\textbf{Symbol} & \textbf{Value} & \textbf{Description} \\ \hline
$\vec{P}$ & $\myvec{0 \\ 2\sqrt{5}}$ & First point \\ \hline
$\vec{Q}$ & $\myvec{-2\sqrt{5} \\ 0}$ & Second point \\ \hline
$d$ & ? & Required distance \\ \hline
\end{tabular}
\caption{Input parameters}
\label{tab:1}
\end{table}

The distance between $\vec{P}$ and $\vec{Q}$ is given by
\begin{align}
    d &= \norm{\vec{P}-\vec{Q}} \label{eq:dist}
\end{align}

\begin{align*}
    \vec{P} - \vec{Q} &= \myvec{0 \\ 2\sqrt{5}} - \myvec{-2\sqrt{5} \\ 0} \\
    &= \myvec{2\sqrt{5} \\ 2\sqrt{5}}
\end{align*}

Substituting in \eqref{eq:dist},
\begin{align}
    d &= \sqrt{(2\sqrt{5})^2 + (2\sqrt{5})^2} \\
      &= \sqrt{20 + 20} \\
      &= \sqrt{40} \\
      &= 2\sqrt{10}
\end{align}

\begin{align*}
    \therefore \text{The required distance is } \mathbf{2\sqrt{10}}
\end{align*}
